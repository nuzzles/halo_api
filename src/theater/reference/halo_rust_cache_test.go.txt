func TestHaloRustCache(t *testing.T) {
	inputs := []string{`null`, `{}`, `{"chunks":null}`, `{"chunks":[null,{}]}`, `{"chunks":[{"index":2,"INDEX":4,"index":null,"chunk_type":3,"start_ms":-4}]}`, `{"CHUNKS":[{"INDEX":100}]}`, `{"chunks":[],"chunks":[{"index":1}]}`, `{"chunks":[{"index":1}],"chunks":null}`, `{"chunks":[{"index":1.0}]}`, `{"chunks":[{"index":"1"}]}`, `{"chunks":[{"index":9223372036854775807,"chunk_type":-9223372036854775808}]}`, `{"chunks":[{"index":9223372036854775808}]}`, `[]`, `{"chunks":{}}`, `{"blob_prefix":"keep","chunks":[{"index":4},{"index":1},{"index":4}]}`, `{`}
	rows := []any{}
	for _, input := range inputs {
		root := t.TempDir()
		if err := os.MkdirAll(ManifestsRoot(root), 0755); err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(ManifestPath(root, "test"), []byte(input), 0600); err != nil {
			t.Fatal(err)
		}
		source, found, err := Open(root, "test")
		metas := []any{}
		if err == nil {
			for _, m := range source.Meta() {
				metas = append(metas, map[string]any{"index": m.Index, "chunk_type": m.ChunkType, "start_ms": m.StartMS})
			}
		}
		rows = append(rows, map[string]any{"json": input, "found": found, "error": err != nil, "metadata": metas})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-cache-reader.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}

func TestHaloRustCacheWrite(t *testing.T) {
	rows := []any{}
	for mode := 0; mode < 8; mode++ {
		root := t.TempDir()
		must := func(err error) {
			if err != nil {
				t.Fatal(err)
			}
		}
		must(EnsureDirs(root))
		dir := ChunkDir(root, "test")
		switch mode {
		case 3:
			must(os.WriteFile(ManifestPath(root, "test"), []byte(`{"blob_prefix":"keep"}`), 0600))
		case 4:
			must(os.MkdirAll(dir, 0755))
			must(os.WriteFile(dir+"/chunk_04.bin", []byte("old"), 0600))
		case 5:
			must(os.WriteFile(dir, []byte("block"), 0600))
		case 6:
			must(os.Remove(ManifestsRoot(root)))
			must(os.WriteFile(ManifestsRoot(root), []byte("block"), 0600))
		case 7:
			must(os.MkdirAll(dir+"/chunk_04.bin", 0755))
		}
		chunks := []WriteChunk{{Index: 4, ChunkType: 2, StartMS: 123, DurationMS: 456, Data: []byte{0, 255, 1}}, {Index: 1, Data: []byte("second")}}
		if mode == 0 {
			chunks = nil
		}
		if mode == 2 {
			chunks = append(chunks, WriteChunk{Index: 4, ChunkType: 3, StartMS: -9, DurationMS: 7, Data: []byte("replacement")})
		}
		err := Write(root, "test", chunks)
		paths := []string{"film_chunks/test", "film_chunks/test/chunk_04.bin", "film_chunks/test/chunk_01.bin", "film_manifests", "film_manifests/test.json"}
		snapshot := []any{}
		for _, path := range paths {
			info, e := os.Stat(root + "/" + path)
			if e != nil {
				snapshot = append(snapshot, map[string]any{"path": path, "kind": "missing"})
				continue
			}
			if info.IsDir() {
				snapshot = append(snapshot, map[string]any{"path": path, "kind": "directory"})
				continue
			}
			data, e := os.ReadFile(root + "/" + path)
			must(e)
			snapshot = append(snapshot, map[string]any{"path": path, "kind": "file", "data": hex.EncodeToString(data)})
		}
		rows = append(rows, map[string]any{"mode": mode, "error": err != nil, "snapshot": snapshot})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-cache-writer.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}

func TestHaloRustCachePartialMetadata(t *testing.T) {
 rows := []any{}
 for _, input := range []string{
  `{"chunks":[{"index":4,"chunk_type":2,"start_ms":900},{"index":9,"chunk_type":9223372036854775807}]}`,
  `{"chunks":[{"index":4,"chunk_type":2,"start_ms":900},{"index":4,"chunk_type":-9223372036854775808}]}`,
  `{"chunks":[{"index":-1,"chunk_type":9223372036854775807},{"index":4,"chunk_type":2,"start_ms":900}]}`,
  `{"chunks":[{"index":4,"chunk_type":9223372036854775807}]}`,
  `{"chunks":[{"index":4,"chunk_type":-9223372036854775808}]}`,
  `{"chunks":[{"index":4,"chunk_type":4294967296}]}`,
 } {
  root := t.TempDir()
  must := func(e error) { if e != nil { t.Fatal(e) } }
  must(os.MkdirAll(ManifestsRoot(root),0755))
  must(os.MkdirAll(ChunkDir(root,"test"),0755))
  must(os.WriteFile(ManifestPath(root,"test"),[]byte(input),0600))
  must(os.WriteFile(ChunkDir(root,"test")+"/chunk_04.bin",[]byte("raw"),0600))
  must(os.WriteFile(ChunkDir(root,"test")+"/chunk_unknown.bin",[]byte("opaque"),0600))
  f,found,err := LoadFilm(root,"test")
  if err != nil { t.Fatal(err) }
  metas := []any{}; chunks := []string{}
  for i,m := range f.Meta() {
   metas=append(metas,map[string]any{"index":m.Index,"chunk_type":m.ChunkType,"start_ms":m.StartMS})
   chunks=append(chunks,hex.EncodeToString(f.Chunk(i)))
  }
  rows=append(rows,map[string]any{"json":input,"found":found,"metadata":metas,"chunks":chunks})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)}
 if e=os.WriteFile("/private/tmp/halo-cache-partial-metadata.json",b,0600);e!=nil{t.Fatal(e)}
}
