func haloRustCalloutsRetained(v reflect.Value) any {
 if v.Kind()==reflect.Pointer {if v.IsNil(){return nil};return haloRustCalloutsRetained(v.Elem())}
 switch v.Kind(){
 case reflect.Struct:
  out:=map[string]any{};for i:=0;i<v.NumField();i++{f:=v.Type().Field(i);name:=strings.Split(f.Tag.Get("json"),",")[0];if name==""{name=f.Name};out[name]=haloRustCalloutsRetained(v.Field(i))};return out
 case reflect.Map:
  if v.IsNil(){return nil};out:=map[string]any{};for _,k:=range v.MapKeys(){out[k.String()]=haloRustCalloutsRetained(v.MapIndex(k))};return out
 case reflect.Slice,reflect.Array:
  if v.Kind()==reflect.Slice&&v.IsNil(){return nil};out:=[]any{};for i:=0;i<v.Len();i++{out=append(out,haloRustCalloutsRetained(v.Index(i)))};return out
 case reflect.String:return v.String()
 case reflect.Int:return v.Int()
 case reflect.Float64:return v.Float()
 case reflect.Bool:return v.Bool()
 }
 panic(v.Kind())
}
