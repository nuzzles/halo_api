package grammar

import (
	"encoding/hex"
	"encoding/json"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustManagedPropertyScan(t *testing.T) {
	rng := rand.New(rand.NewSource(4113))
	rows := []any{}
	for run := 0; run < 512; run++ {
		names := make([]string, 36)
		for i := range names {
			names[i] = "managed-object-player-masked-property-component"
			if i%5 == 0 {
				names[i] = "managed-object-property-component"
			}
		}
		if run%3 == 0 {
			names[8] = "unsupported"
		}
		if run%7 == 0 {
			names[20] = ""
		}
		arch := Archetype{Index: 13, Components: names, Levels: make([]uint32, 36)}
		prof := ProfilDeBalayageParDefaut()
		w := managedPropertyWalk{prof: prof, arch: arch}
		w.obs = w.install()
        emissions := []any{}
        currentTS := uint64(0)
        original := w.obs.ManagedPropertyHook
        w.obs.ManagedPropertyHook = func(f ManagedPropertyField, values []uint64) {
            emissions = append(emissions,map[string]any{"timestamp_us":currentTS,"field":f,"values":append([]uint64{},values...)})
            original(f,values)
        }
		band := map[uint32]bool{512: true, 514: true}
		sc := ManagedPropertyScan{Slots: 2}
		inputs := []any{}
		for part := 0; part < 4; part++ {
			bw := &translocWriter{}
			bw.put(run%8, 0)
			for j := 0; j < 16; j++ {
				bw.put(1, 1)
				bw.put(13, uint64(512+rng.Intn(3)))
				bw.put(2, uint64(rng.Intn(4)))
				bw.put(2, 0)
				n := 1 + rng.Intn(4)
				bw.put(3, uint64(n))
				base := rng.Intn(36)
				for k := 0; k < n; k++ {
					bw.put(6, uint64(base+k))
				}
				for k := 0; k < n; k++ {
					tag := rng.Intn(16)
					bw.put(4, uint64(tag))
					a := (base+k)%5 == 0
					width := 0
					switch {
					case a && tag == 1 || !a && tag >= 11:
						width = 4
					case a && tag == 2 || !a && tag == 10:
						width = 1
					case a && tag == 3 || !a && tag == 7:
						width = 24
					case a && tag >= 4 && tag <= 6 || !a && (tag == 8 || tag == 9):
						width = 32
					}
					if width > 0 {
						bw.put(width, rng.Uint64())
					}
				}
				bw.put(rng.Intn(5), 0)
			}
			if part == 3 && run%2 == 0 {
				bw.pay = bw.pay[:len(bw.pay)-2]
			}
			ts := uint64(part*1000000 + run)
			currentTS = ts
            w.scanPayload(bw.pay, band, ts, &sc)
			inputs = append(inputs, map[string]any{"hex": hex.EncodeToString(bw.pay), "timestamp_us": ts})
		}
		if sc.Reads == nil {
			sc.Reads = []ManagedPropertyRead{}
		}
		rows = append(rows, map[string]any{"names": names, "inputs": inputs, "output": sc, "emissions":emissions})
	}

	for run := 0; run < 512; run++ {
		names := []string{"managed-object-property-component", "simulation-state", "managed-object-property-component"}
		arch := Archetype{Index: 13, Components: names, Levels: []uint32{0, 0, 0}}
		prof := ProfilDeBalayageParDefaut()
		prof.Grammaire.SimStateComplet = run%2 != 0
		sc := ManagedPropertyScan{Slots: 2}
		walk := managedPropertyWalk{prof: prof, arch: arch}
		walk.obs = walk.install()
        emissions := []any{}
        original := walk.obs.ManagedPropertyHook
        walk.obs.ManagedPropertyHook = func(f ManagedPropertyField, values []uint64) {
            emissions = append(emissions,map[string]any{"timestamp_us":run,"field":f,"values":append([]uint64{},values...)})
            original(f,values)
        }
		bw := &translocWriter{}
		bw.put(run%8, 0)
		for j := 0; j < 4; j++ {
			bw.put(1, 1)
			bw.put(13, uint64(512+2*(j%2)))
			bw.put(2, 1)
			bw.put(2, 0)
			bw.put(3, 3)
			bw.put(6, 0)
			bw.put(6, 1)
			bw.put(6, 2)
			bw.put(4, 1)
			bw.put(4, uint64((run+j)%16))
			bw.put(1, 0)
			bw.put(4, 1)
			bw.put(4, uint64((run+j+7)%16))
		}
		if run%7 == 0 {
			bw.pay = bw.pay[:len(bw.pay)-1]
		}
		walk.scanPayload(bw.pay, map[uint32]bool{512: true, 514: true}, uint64(run), &sc)
		if sc.Reads == nil {
			sc.Reads = []ManagedPropertyRead{}
		}
		rows = append(rows, map[string]any{"names": names, "simulation_complete": prof.Grammaire.SimStateComplet, "inputs": []any{map[string]any{"timestamp_us": run, "hex": hex.EncodeToString(bw.pay)}}, "output": sc, "emissions":emissions})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-managed-property.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
