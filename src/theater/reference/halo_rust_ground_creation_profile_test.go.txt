package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"sort"
	"testing"
)

func TestHaloRustGroundCreationProfile(t *testing.T) {
	cat, e := profile.LoadMapQuantCatalog("/private/tmp/halo-levelup-v75/data/titles/halo_infinite/reference/map_quant_bounds.json")
	if e != nil {
		t.Fatal(e)
	}
	entry, e := cat.Lookup("bazaar")
	if e != nil {
		t.Fatal(e)
	}
	wr := entry.Range()
	rng := rand.New(rand.NewSource(373242))
	rows := []any{}
	for i := 0; i < 512; i++ {
		p := ProfilDeBalayageParDefaut()
		p.MPP = profile.MPPWidths{Lead: 6 + i%8, Index: 2 + i%6}
		iw := i % 4
		region := uint32((i / 4) % (1 << uint(iw)))
		axes := [3]uint{uint(6 + i%15), uint(7 + i%13), uint(8 + i%11)}
		if i%5 == 0 {
			axes = [3]uint{31, 32, 31}
		}
		p.PoserLargeursObjetDuMonde(profile.PrecisionDescriptor{IndexW: uint(iw), AxisW: axes, Region: region})
		ti := uint32(42)
		deser := consumeDefaultStateTI42
		p.Grammaire.SimStateComplet = i%3 != 0
		p.Grammaire.ControleDeCorruption = i%2 == 0
		p.Grammaire.LargeursCalibrees = map[string]int{}
		p.Grammaire.LargeursBouchon = map[string]int{}
		arch := Archetype{Index: 42, Components: make([]string, 22), Levels: make([]uint32, 22)}
		arch.Components[0] = "object-position-dynamic-precision-component"
		arch.Components[1] = "simulation-state-component"
		arch.Components[2] = "not-ported"
		arch.Components[9] = "object-multiplayer-properties-component"
		arch.Components[20] = "weapon-ammo-component"
		width := []int{0, 1, 17, 65}[i%4]
		switch (i / 4) % 4 {
		case 1:
			p.Grammaire.LargeursCalibrees[arch.Components[0]] = width
		case 2:
			p.Grammaire.LargeursBouchon[arch.Components[2]] = width
		case 3:
			p.Grammaire.LargeursCalibrees[arch.Components[0]] = width
			p.Grammaire.LargeursBouchon[arch.Components[2]] = width
		}
		pay := make([]byte, 512)
		rng.Read(pay)
		put := func(at, n int, v uint64) {
			for j := 0; j < n; j++ {
				m := byte(1 << uint(7-(at+j)%8))
				pay[(at+j)/8] &= ^m
				if v&(uint64(1)<<uint(n-1-j)) != 0 {
					pay[(at+j)/8] |= m
				}
			}
		}
		start := i % 8
		put(start, 3, 1)
		put(start+3, 13, 512)
		put(start+16, 2, uint64(i%4))
		put(start+18, 6, uint64(ti))
		br := LecteurSur(pay)
		br.PoserContexte(ContexteDeLecture{Profil: p})
		br.SetBitPos(start + 24)
		deser(br)
		at := br.BitPos()
		put(at, 1, 0)
		put(at+1, 1, 0)
		mask := []int{0, 20}
		if i%5 == 0 {
			mask = []int{0, 1, 20}
		}
		if i%7 == 0 {
			mask = []int{0, 2, 20}
		}
		if i%11 == 0 {
			mask = append(mask, 9)
			sort.Ints(mask)
		}
		put(at+2, 3, uint64(len(mask)))
		for j, n := range mask {
			put(at+5+j*6, 6, uint64(n))
		}
		at += 5 + len(mask)*6
		put(at, 2, 0)
		encodedRegion := region
		if i%17 == 0 && iw > 0 {
			encodedRegion ^= 1
		}
		put(at+2, iw, uint64(encodedRegion))
		at += 2 + iw
		for a, n := range axes {
			q := uint64(rng.Uint32())%((uint64(1)<<n)-2) + 1
			if i%19 == 0 && a == 1 {
				q = 0
			}
			put(at, int(n), q)
			at += int(n)
		}
		put(at, 2, 0)
		at += 2
		pay = pay[:(at+160+7)/8]
		if i%13 == 0 {
			pay = pay[:len(pay)-2]
		}
		var cur equipCreationRead
		walk := equipCreationWalk{obs: installCreationHooks(&cur), prof: p, ammoArch: &arch, comps: len(arch.Components), wr: &wr, band: map[uint32]bool{512: true}, cur: &cur, ti: ti, deser: deser}
		stats := types.EquipmentCreationStats{Slots: 1}
		records := walk.scanPayload(pay, &stats, FilmPacket{}, 0)
		if records == nil {
			records = []types.EquipmentCreation{}
		}
		rows = append(rows, map[string]any{"map": entry, "ti": ti, "names": arch.Components, "component_widths": map[string]any{"calibrated": p.Grammaire.LargeursCalibrees, "stubs": p.Grammaire.LargeursBouchon}, "simulation": p.Grammaire.SimStateComplet, "corruption": p.Grammaire.ControleDeCorruption, "index": iw, "region": region, "axes": axes, "widths": [2]int{p.MPP.Lead, p.MPP.Index}, "hex": hex.EncodeToString(pay), "records": records, "stats": stats})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-ground-creation-profile.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
