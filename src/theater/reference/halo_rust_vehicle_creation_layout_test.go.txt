package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"sort"
	"testing"
)

func TestHaloRustVehicleCreationLayout(t *testing.T) {
	cat, e := profile.LoadMapQuantCatalog("/private/tmp/halo-levelup-v75/data/titles/halo_infinite/reference/map_quant_bounds.json")
	if e != nil {
		t.Fatal(e)
	}
	names := []string{}
	for k := range cat.Maps {
		names = append(names, k)
	}
	sort.Strings(names)
	rng := rand.New(rand.NewSource(42013))
	rows := []any{}
	for run := 0; run < 512; run++ {
		entry := cat.Maps[names[run%len(names)]]
		lay := entry.Layout()
		wr := entry.Range()
		prof := ProfilDeBalayageParDefaut()
		prof.PoserLargeursObjetDuMonde(profile.PrecisionDescriptor{IndexW: uint(entry.EffectiveRegionIndexBits()), AxisW: lay.AxisW, Region: lay.Region})
		prof.MPP.Lead = 7 + run%4
		prof.MPP.Index = 4 + run%3
		lay.GateBits = 4 + run%4
		lay.Region = 0
		if lay.GateBits > 4 {
			lay.Region = uint32(run) & ((1 << uint(lay.GateBits-4)) - 1)
		}
		band := map[uint32]bool{512: true, 514: true, 515: true}
		w := &translocWriter{}
		for i := 0; i < 8; i++ {
			w.put(rng.Intn(8), 0)
			w.put(3, 1)
			w.put(13, uint64(512+rng.Intn(5)))
			w.put(2, uint64(rng.Intn(4)))
			w.put(6, 40)
			w.put(1, 0)
			w.put(prof.MPP.Lead, uint64(rng.Uint32()))
			w.put(32, uint64(rng.Uint32()))
			w.put(1, 1)
			w.put(1, 0)
			w.put(1, 0)
			w.put(2, 0)
			w.put(prof.MPP.Index, 0)
			w.put(3, 0)
			w.put(1, 0)
			w.put(1, 0)
			if i%2 == 0 {
				w.put(1, 1)
				w.put(1, 0)
				w.put(int(entry.EffectiveRegionIndexBits()), uint64(entry.Layout().Region))
				for _, n := range lay.AxisW {
					w.put(int(n), uint64(rng.Uint32()))
				}
				w.put(1, 0)
				w.put(19, uint64(rng.Uint32()))
				w.put(8, uint64(rng.Intn(256)))
			} else {
				w.put(1, 0)
			}
			w.put(19, uint64(rng.Uint32()))
			w.put(1, 0)
			w.put(1, 0)
			w.put(1, uint64(i%2))
			if i%3 == 0 {
				w.put(1, 1)
				mask := uint64(1) << uint(i)
				if run%11 == 0 {
					mask = 1 << 63
				}
				w.put(64, mask)
			} else {
				w.put(1, 0)
				w.put(3, 1)
				w.put(6, uint64(i%2))
			}
			w.put(4, 0)
			region := lay.Region
			if i == 4 && run%2 == 0 {
				region ^= 1
			}
			w.put(lay.GateBits-4, uint64(region))
			for _, n := range lay.AxisW {
				q := uint64(1 + rng.Intn((1<<n)-2))
				if i == 5 && run%3 == 0 {
					q = 0
				}
				w.put(int(n), q)
			}
			w.put(2, 0)
		}
		for cut := 0; cut < 4; cut++ {
			pay := append([]byte{}, w.pay...)
			if cut == 1 {
				rng.Read(pay)
			}
			if cut >= 2 {
				pay = pay[:rng.Intn(len(pay))]
			}
			var cur equipCreationRead
			walk := equipCreationWalk{obs: installCreationHooks(&cur), prof: prof, comps: 33, wr: &wr, band: band, cur: &cur, ti: 40, deser: consumeDefaultStateTI40, posDecode: func(p []byte, at int) ([3]float32, bool) { return decodeBipedI0Pos(p, at, lay, &wr) }, posBits: lay.TotalBits()}
			st := types.EquipmentCreationStats{Slots: len(band)}
			records := walk.scanPayload(pay, &st, FilmPacket{}, 0)
			if records == nil {
				records = []types.EquipmentCreation{}
			}
			rows = append(rows, map[string]any{"map": entry, "layout": map[string]any{"gate_bits": lay.GateBits, "axis_widths": lay.AxisW, "region": lay.Region}, "widths": [2]int{prof.MPP.Lead, prof.MPP.Index}, "hex": hex.EncodeToString(pay), "records": records, "stats": st})
		}
	}
	data, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-vehicle-creation-layout.json", data, 0600); e != nil {
		t.Fatal(e)
	}
}
