fn main() {
 let input=std::fs::read_to_string("/private/tmp/halo-geometry-distance-probe.txt").unwrap();
 let(mut d3,mut plan,mut go_plan,mut go_fma)=(0,0,0,0);
 for (i,line) in input.lines().enumerate() {
  let v:Vec<u64>=line.split_whitespace().map(|s|s.parse().unwrap()).collect();
  let a:[f32;3]=std::array::from_fn(|j|f32::from_bits(v[j] as u32));
  let b:[f32;3]=std::array::from_fn(|j|f32::from_bits(v[j+3] as u32));
  let [x,y,z]=std::array::from_fn(|j|f64::from(a[j]-b[j]));
  if (x*x+y*y+z*z).sqrt().to_bits()!=v[6]{d3+=1;}
  let p=x.abs().max(y.abs());let q=x.abs().min(y.abs())/p;
  if (p*(1.0+q*q).sqrt()).to_bits()!=v[7]{go_plan+=1;}
  if (p*q.mul_add(q,1.0).sqrt()).to_bits()!=v[7]{go_fma+=1;}
  if x.hypot(y).to_bits()!=v[7]{plan+=1;if plan<=3{println!("plan mismatch {i}: native={} Rust={} input={line}",v[7],x.hypot(y).to_bits());}}
 }
 println!("dist3 mismatches={d3}; std hypot={plan}; scaled unfused={go_plan}; scaled fused={go_fma}");
}
