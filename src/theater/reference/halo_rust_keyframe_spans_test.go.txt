package grammar
import("testing";"encoding/json";"encoding/hex";"math/rand";"os")
func TestHaloRustKeyframeSpans(t *testing.T){
 rows:=[]any{};rng:=rand.New(rand.NewSource(8941))
 for c:=0;c<512;c++{
 pay:=make([]byte,200);rng.Read(pay);at:=1;put:=func(w int,v uint64){for k:=0;k<w;k++{bit:=at+k;mask:=byte(1<<uint(7-bit%8));pay[bit/8]&=^mask;if v&(uint64(1)<<uint(w-1-k))!=0{pay[bit/8]|=mask}};at+=w};for j:=0;j<6;j++{put(32,uint64(1+c%3)<<30|uint64(100+j*(1+c%2)));put(32,uint64(35+j%7));at+=40+c%40};if c%5==0{pay=pay[:c%80]};spans:=KeyframeRecordSpans(pay);if spans==nil{spans=[]KeyframeRecordSpan{}}
 input:=[]KeyframeRecordSpan{};for j:=0;j<c%64;j++{input=append(input,KeyframeRecordSpan{Slot:j%5,TI:39+j%3,Gen:j%4,LengthBits:(j*89+c)%700-10,SlotGap:j%4,TimestampUS:uint64(j%7),Chunk:j%3,PacketIndex:j})};states:=VehicleKeyframeStates(input)
 long:=make([]byte,80);short:=make([]byte,80);rng.Read(long);rng.Read(short);ls,ss:=c%8,c%9;le,se:=ls+(c*7)%800,ss+(c*11)%800;ins:=FindKeyframeBlockInsertion(KeyframeRecordBits{Pay:long,BitStart:ls,BitEnd:le},KeyframeRecordBits{Pay:short,BitStart:ss,BitEnd:se})
 rows=append(rows,map[string]any{"hex":hex.EncodeToString(pay),"spans":spans,"input":input,"states":states,"long":hex.EncodeToString(long),"short":hex.EncodeToString(short),"ranges":[4]int{ls,le,ss,se},"insertion":ins})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-keyframe-spans.json",b,0600);e!=nil{t.Fatal(e)}
}
