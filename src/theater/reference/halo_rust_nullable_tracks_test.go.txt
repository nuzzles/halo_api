package replay
import("testing";"encoding/json";"os")
func TestHaloRustNullableTracks(t *testing.T) {
 rows:=[]any{}
 for n:=0;n<64;n++ {
  ts:=[]Track{};kinds:=[]int{}
  for j:=0;j<3;j++ {mode:=(n>>(j*2))&3;tr:=Track{Slot:uint32(j+1),Team:-1};switch mode {case 1:tr.Points=[]Point{};case 2:tr.Points=[]Point{{T:1,X:float32(j)+0.5,Y:2.5,Z:-1.5}};case 3:tr.Points=[]Point{{T:1,X:float32(j)+0.5,Y:2.5,Z:-1.5},{T:2,X:float32(j)+1.5,Y:3.5,Z:1.5}}};ts=append(ts,tr);kinds=append(kinds,mode)}
  b,rejected:=boundsOf(ts);rows=append(rows,map[string]any{"tracks":ts,"kinds":kinds,"bounds":b,"rejected":rejected})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-null-tracks.json",raw,0600);e!=nil{t.Fatal(e)}
}
