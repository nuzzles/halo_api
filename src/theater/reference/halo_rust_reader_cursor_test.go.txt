package grammar

import (
 "encoding/json"
 "os"
 "testing"
)

func TestHaloRustReaderCursor(t *testing.T) {
 type row struct {
  Start int `json:"start"`
  Width uint64 `json:"width"`
  Variable bool `json:"variable"`
  Panic bool `json:"panic"`
  Value uint64 `json:"value"`
  SignedValue int32 `json:"signed_value"`
  End int `json:"end"`
  Remaining int `json:"remaining"`
  Recovery uint64 `json:"recovery"`
  RecoveryEnd int `json:"recovery_end"`
 }
 data:=[]byte{0xab,0xcd,0xef,0x12,0x34,0x56,0x78,0x90}
 starts:=[]int{-1,-1<<63,1<<63-128,1<<63-65,1<<63-2,1<<63-1}
 widths:=[]uint64{0,1,2,8,64,65,127,128,129,256,1<<32,^uint64(0)}
 var rows []row
 for _,start:=range starts {
  for _,width:=range widths {
   b:=LecteurSur(data);b.SetBitPos(start)
   r:=row{Start:start,Width:width}
   func(){defer func(){if recover()!=nil{r.Panic=true}}();r.Value=b.ReadBits(uint(width))}()
   r.End=b.BitPos();r.Remaining=b.Remaining()
   b.SetBitPos(0);r.Recovery=b.ReadBits(16);r.RecoveryEnd=b.BitPos()
   rows=append(rows,r)
  }
  b:=LecteurSur(data);b.SetBitPos(start)
  r:=row{Start:start,Variable:true}
  func(){defer func(){if recover()!=nil{r.Panic=true}}();r.SignedValue=b.ReadSignedVarWidth()}()
  r.End=b.BitPos();r.Remaining=b.Remaining()
  b.SetBitPos(0);r.Recovery=b.ReadBits(16);r.RecoveryEnd=b.BitPos()
  rows=append(rows,r)
 }
 raw,err:=json.Marshal(struct{Data []int `json:"data"`;Rows []row `json:"rows"`}{[]int{171,205,239,18,52,86,120,144},rows})
 if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-reader-cursor.json",raw,0600);err!=nil{t.Fatal(err)}
}
