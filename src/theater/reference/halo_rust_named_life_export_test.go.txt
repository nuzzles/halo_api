  lives:=[]lifeSpan{};input:=[]any{};count:=run%43

  }

 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-named-life-export.json",raw,0600);e!=nil{t.Fatal(e)}
}
