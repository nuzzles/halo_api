package damagetag

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"sort"
	"strconv"
	"strings"
	"testing"
)

func TestHaloRustDamageCatalog(t *testing.T) {
	rows := []any{}
	whitespace := []string{"", " ", "\t", "\r", "\u0085", "\u00a0", "\u2007", "\u2028", "\u2029", "\u202f", "\u3000", "\u200b"}
	hexes := []string{"0", "ffffffff", "FFFFFFFF", "000000000000000000001", "100000000", "100000000g", "ffffffffg", "+1", "-1", "0x1", "1_0", "", "g", "é", " 1", "1 "}
	for i := 0; i < 513; i++ {
		ids := fmt.Sprintf("# date=first date=ignored\n%s# date=second\n\n000A\na\n%s%s%s\n# date= date=ignored\n", whitespace[i%len(whitespace)], whitespace[(i/4)%len(whitespace)], hexes[(i/12)%len(hexes)], whitespace[(i/8)%len(whitespace)])
		labels := "# date=labels-first date=ignored\n1\tARME\tSOUS_RESERVE\tname\tdetail\treserve\n1\tcustom-class\tcustom-status\tnew name\tnew detail\tnew reserve\n"
		switch i % 8 {
		case 0:
			labels += "# date=labels-last\n2\tMELEE\tVALIDE\t\t\t\r\n"
		case 1:
			labels += " # date=not-comment\n"
		case 2:
			labels += hexes[(i/8)%len(hexes)] + "\tARME\tVALIDE\t\t\t\n"
		case 3:
			labels += "2\tARME\tVALIDE\t\t\textra\tcolumn\n"
		case 4:
			labels += "\u0085\n# date= date=ignored\n"
		case 5:
			labels += "2\tGRENADE\tAMBIGU\tn\td\tr\n"
		case 6:
			labels += "2\tINCONNU\tINCONNU\t\t\t\n"
		case 7:
			labels += "2\tMELEE\tVALIDE\t\t\t"
		}
		// Half the rows always reach label parsing independently of ID errors.
		if i%2 == 1 {
			ids = "# date=first date=ignored\n\u0085 a \r\na\nffffffff\n# date=last\n# date= date=ignored\n"
		}
		if i == 512 {
			ids, labels = rawIDs, rawLabels
		}
		set, list, idDate, err := parseIDs(ids)
		var labs map[uint32]Label
		labelDate := ""
		table := "Ids"
		if err == nil {
			table = "Labels"
			labs, labelDate, err = parseLabels(labels)
		}
		row := map[string]any{"ids_text": ids, "labels_text": labels}
		if err != nil {
			var line int
			fmt.Sscanf(err.Error(), "ligne %d:", &line)
			failure := "columns"
			var number *strconv.NumError
			if errors.As(err, &number) {
				failure = "syntax"
				if errors.Is(number.Err, strconv.ErrRange) {
					failure = "overflow"
				}
			}
			value := ""
			columns := 0
			if number != nil {
				value = number.Num
			} else {
				value = strings.Split(labels, "\n")[line-1]
				fmt.Sscanf(err.Error(), "ligne %d: %d colonnes", &line, &columns)
			}
			row["error"] = map[string]any{"table": table, "line": line, "kind": failure, "value": value, "columns": columns}
		} else {
			if len(set) != len(list) {
				t.Fatal("set/list mismatch")
			}
			if list == nil {
				list = []uint32{}
			}
			sorted := []Label{}
			for _, l := range labs {
				sorted = append(sorted, l)
			}
			sort.Slice(sorted, func(i, j int) bool { return sorted[i].Tag < sorted[j].Tag })
			row["ids"] = list
			row["labels"] = sorted
			row["ids_date"] = idDate
			row["labels_date"] = labelDate
			pub := []bool{}
			for _, l := range sorted {
				pub = append(pub, l.Publishable())
			}
			row["publishable"] = pub
		}
		rows = append(rows, row)
	}
	data, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-damage-catalog.json", data, 0600); err != nil {
		t.Fatal(err)
	}
}
