package replay
import("testing";"os";"encoding/json";"math/rand")
func TestHaloRustReplayVehicleRelays(t *testing.T){
 rng:=rand.New(rand.NewSource(42010));rows:=[]any{}
 for run:=0;run<1024;run++ {
  tracks:=[]VehicleTrack{}
  for i:=0;i<run%30;i++ {
   v:=VehicleTrack{Slot:uint32(i),Gen:uint32(i%4),Chassis:[]string{"","0x123","0x456"}[rng.Intn(3)],Family:"test",T0:rng.Intn(20),T1:rng.Intn(25),T1Max:rng.Intn(30),End:[]string{VehicleEndDestroyed,VehicleEndUnknown,VehicleEndFilmEnd}[rng.Intn(3)]}
   if run%4==0 {v.Chassis="0x123";v.T0=i*2;v.T1=i*2+1;v.T1Max=i*2+5;v.End=VehicleEndUnknown}
   if rng.Intn(4)!=0 {v.Spawn=&VehicleSpawn{X:float32(rng.Intn(5))/10,Y:float32(rng.Intn(5))/10,H:new(float32)}}
   if rng.Intn(3)==0 {end:=rng.Intn(20);v.TEnd=&end}
   for j:=0;j<rng.Intn(5);j++ {v.Samples=append(v.Samples,VehicleSample{T:v.T0+j,X:float32(rng.Intn(5))/10,Y:float32(rng.Intn(5))/10,H:float32(i)})}
   for j:=0;j<rng.Intn(5);j++ {r:=VehicleRide{T0:rng.Intn(25),T1:rng.Intn(30),Slot:uint32(rng.Intn(5)),Src:"event"};seat:=rng.Intn(4);r.Seat=&seat;for k:=0;k<rng.Intn(10);k++ {r.Aim=append(r.Aim,VehicleAim{T:rng.Intn(30),H:float32(k)})};v.Rides=append(v.Rides,r)}
   tracks=append(tracks,v)
  }
  sortVehicleTracks(tracks)
  before,e:=json.Marshal(tracks);if e!=nil{t.Fatal(e)}
  out,n:=mergeVehicleRelays(tracks)
  rows=append(rows,map[string]any{"input":json.RawMessage(before),"output":out,"merged":n})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-vehicle-relays.json",raw,0600);e!=nil{t.Fatal(e)}
}
