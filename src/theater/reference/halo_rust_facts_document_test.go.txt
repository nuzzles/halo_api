package replay

import (
 "encoding/hex"
 "encoding/json"
 "os"
 "testing"
 "log/slog"
 "levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
 "levelup/go-api/internal/games/halo_infinite/film/internal/profile"
 "levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives"
 "levelup/go-api/internal/games/halo_infinite/film/internal/facts/fallback"
 "levelup/go-api/internal/games/halo_infinite/film/types"
)

func TestHaloRustFactsDocument(t *testing.T) {
 data,e:=os.ReadFile("/private/tmp/halo-facts-players.json");if e!=nil{t.Fatal(e)}
 var base []struct{Input string};if e=json.Unmarshal(data,&base);e!=nil{t.Fatal(e)}
 entry:=profile.MapQuantEntry{Module:"map",AxisWidths:[3]uint{13,14,15},Min:[3]float32{-10,-30,-50},Max:[3]float32{100,300,500}}
 old:=slog.Default();defer slog.SetDefault(old)
 rows:=[]any{}
 for n,b:=range base {
  raw,e:=hex.DecodeString(b.Input);if e!=nil{t.Fatal(e)}
  f,e:=DecodeFilmFactsFile(raw,entry);if e!=nil{t.Fatal(e)}
  g:=&f.Facts
  g.Fire=[]grammar.FireEvent{{TimestampUS:2200000,FilmIndex:0,WeaponID:uint64(9)<<32|1},{TimestampUS:2300000,FilmIndex:1,WeaponID:uint64(9)<<32|2}}
  g.Loadouts=[]types.KeyframeLoadout{{TimestampUS:2000000,Slot:519,Families:[]uint32{9}}}
  g.Grenades=[]grammar.GrenadeThrow{{TimestampUS:2300000,FilmIndex:0,TypeID:0}}
  g.Inventory=[]KeyframeInventory{{TimestampUS:2100000,Slot:519,GrenadesRead:true,Grenades:[4]uint32{2,1,0,0},AbilityRank:5,SelectedGrenadeRank:0}}
  g.AbilityRanks=[]types.AbilityRank{{TimestampUS:2100000,Slot:519,Rank:5}}
  g.WeaponChanges=[]types.HeldWeaponChange{{TimestampUS:2200000,Slot:519,Family:9,Kind:"taken"},{TimestampUS:2300000,Slot:519,Previous:9,Kind:"dropped"}}
  g.AbilityImpulseStats.Scanned=n%3==0;g.AbilityChargeStats.Scanned=n%4==0
  g.Pads.Weapons.Scanned=n%2==0;g.Pads.Powerups.Scanned=n%3==0;g.Vehicles.Scanned=n%2!=0
  g.ZoneScanned=n%2==0;g.FlagGaugeScanned=n%3==0
  records:=[]types.StatRecord{}
  for j:=0;j<20;j++ {records=append(records,types.StatRecord{Slot:10,Round:0,TimeMS:1000+j*100,Comps:map[int]types.StatValue{0:{A:int64(j)},1:{A:int64(j)},2:{A:int64(j)},3:{A:int64(j)},21:{B:int64(j)},22:{A:int64(j)},23:{B:int64(j)},25:{A:int64(j)}}})}
  f.Statborg=FilmStatborg{Records:records,BurstMS:[]int{1000,2000},Truncated:true}
  f.Fallbacks=[]fallback.Declenchement{{Nom:"repli_plafond_grenade_par_defaut",Declenchements:n+1}}
  encoded,e:=EncodeFilmFactsFile(f);if e!=nil{t.Fatal(e)}
  decoded,e:=DecodeFilmFactsFile(encoded,entry);if e!=nil{t.Fatal(e)}
  id:=objectives.RoundIdentity{}
  if n%3==1{id=objectives.FlatRoundIdentity(map[int]string{10:"100"})}
  if n%3==2{id=objectives.FlatRoundIdentity(map[int]string{})}
  opt:=Options{FrameIntervalMS:[]int{-1,0,50,100,250}[n%5],MinPoints:[]int{-1,0,1,3}[n%4],StatborgIdentity:id,
    Flag:FlagInput{Scanned:n%2==0,Records:records,Bursts:[]int{1000,2000},Identity:id},
    Vip:VipInput{Scanned:n%3==0,Records:records},Skull:SkullInput{Scanned:n%4==0,Records:records,Identity:id},
    Bomb:BombInput{Scanned:n%5==0,CarryScanned:n%7==0},
    ScoreboardTeams:map[string]int{"100":1,"101":0},
    Fallbacks:fallback.NouveauCompteur(),
  }
  opt.Fallbacks.DeclencheN("repli_plafond_grenade_par_defaut",7)
  if n%2==0 {opt.Score=&ScoreInput{Records:records,TargetScore:50,HoldTicksPerPoint:10,Truncated:n%4==0}}
  opt.Objectives=[]objectives.IdentifiedEvent{{NamedEvent:objectives.NamedEvent{TimeMS:1500,Stat:"flag_grabs",Side:"A"},XUID:"100"}}
  opt.Labels=LabelCatalog{Weapons:map[uint32]WeaponLabel{9:{En:"Fixture",Fr:"Fixture",Fx:"ballistic"}},Keys:map[uint32]string{9:"fixture_weapon"},Effects:map[string]string{"fixture_weapon":"ballistic"}}
  h:=&rustScoreSeriesHandler{records:[]any{}};slog.SetDefault(slog.New(h))
  doc:=BuildFromFacts("fixture-rich","halo_infinite",decoded,opt)
  rows=append(rows,map[string]any{"input":hex.EncodeToString(encoded),"interval":opt.FrameIntervalMS,"min_points":opt.MinPoints,"identity_mode":n%3,"records":records,"identified":opt.Objectives,"flag":opt.Flag.Scanned,"vip":opt.Vip.Scanned,"skull":opt.Skull.Scanned,"bomb":opt.Bomb.Scanned,"carry":opt.Bomb.CarryScanned,"score":opt.Score!=nil,"truncated":n%4==0,"document":doc,"logs":h.records})
 }
 data,e=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-facts-document.json",data,0600);e!=nil{t.Fatal(e)}
}
