package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"reflect"
	"sort"
	"testing"
)

func TestHaloRustZoomPadding(t *testing.T) {
	rows := []any{}
	data := []byte{}
	for i := 0; i < 256; i++ {
		for length := 0; length <= 8; length++ {
			pay := []byte{0xca, byte(i), byte(i * 7), byte(i * 13), byte(i * 23), byte(i * 37), byte(i * 53), byte(i * 71)}[:length]
			if i%7 == 0 && length > 0 {
				pay[0] = 0x4a
			}
			ev, ok := decodeZoomHead(pay, 123)
			br := LecteurSur(pay)
			h := readPacketHead(br)
			if h.More && h.Type == zoomEventType {
				for _, dom := range zoomRefDomains {
					readZoomRef(br, dom)
				}
				br.ReadBits(2)
			}
			rows = append(rows, map[string]any{"hex": hex.EncodeToString(pay), "event": ev, "ok": ok, "end": br.BitPos()})
			if length == 0 {
				continue
			}
			packet := make([]byte, 16)
			kind := uint16(0)
			if i%5 == 0 {
				kind = 6
			}
			binary.LittleEndian.PutUint16(packet, kind)
			binary.LittleEndian.PutUint32(packet[4:], uint32(len(pay)))
			binary.LittleEndian.PutUint64(packet[8:], uint64(i%5))
			data = append(data, append(packet, pay...)...)
		}
	}
	film, err := source.Load(source.MemoryChunks{data}, []types.ChunkMeta{{Index: 1}})
	if err != nil {
		t.Fatal(err)
	}
	expected := ScanZoomEvents(film)
	type item struct {
		Event  ZoomEvent `json:"event"`
		Packet int       `json:"packet"`
		Start  int       `json:"start"`
		Size   int       `json:"size"`
	}
	observed := []item{}
	d, pks, _ := FilmChunkAt(film, 1)
	for pi, pk := range pks {
		if pk.Type != 0 || pk.Size < 2 {
			continue
		}
		pay := pk.Payload(d)
		if pay[0] != zoomFamilyByte {
			continue
		}
		if ev, ok := decodeZoomHead(pay, pk.TimestampUS); ok {
			observed = append(observed, item{ev, pi, pk.Start, pk.Size})
		}
	}
	sort.Slice(observed, func(i, j int) bool { return observed[i].Event.TimestampUS < observed[j].Event.TimestampUS })
	var projected []ZoomEvent
	for _, r := range observed {
		projected = append(projected, r.Event)
	}
	if !reflect.DeepEqual(projected, expected) {
		t.Fatal("source wrapper differs")
	}
	raw, err := json.Marshal(map[string]any{"rows": rows, "source": hex.EncodeToString(data), "output": observed})
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-zoom-padding.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
