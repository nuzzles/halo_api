package grammar

import (
	"encoding/json"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustSlotBand(t *testing.T) {
	rng := rand.New(rand.NewSource(418192))
	rows := []any{}
	for run := 0; run < 256; run++ {
		flags := map[uint32]bool{}
		for j := 0; j < run%129; j++ {
			flags[uint32(rng.Intn(32768))] = rng.Intn(3) != 0
		}
		flags[^uint32(0)] = false
		if run%9 == 0 {
			flags = map[uint32]bool{}
		}
		band := NewSlotBand(flags)
		queries := []uint32{0, 8191, 8192, 32767, 32768, ^uint32(0)}
		for j := 0; j < 1024; j++ {
			queries = append(queries, uint32(rng.Intn(65536)))
		}
		has := []bool{}
		for _, q := range queries {
			has = append(has, band.Has(q))
		}
		rows = append(rows, map[string]any{"flags": flags, "count": band.Count(), "slots": band.Slots(), "queries": queries, "has": has})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-slot-band.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
