package replay

import (
 "encoding/json"
 "math/rand"
 "os"
 "testing"
 "levelup/go-api/internal/games/halo_infinite/film/types"
)

func TestHaloRustFactsAbilities(t *testing.T) {
 rng:=rand.New(rand.NewSource(417251))
 rows:=[]any{}
 domain:=[]int{-9223372036854775808,-4294967296,-2,-1,0,1,2,16,23,27,28,2147483648,4294967296,9223372036854775807}
 paletteJSON:=func(p *AbilityPalette) any {if p==nil{return nil};return map[string]any{"id":p.ID,"markers":p.Markers,"ranks":p.Ranks,"families":p.Families}}
 for run:=0;run<1024;run++ {
  origin:=uint64(run%3)*1000000; step:=uint64(1+run%7)*100000
  ranks:=[]types.AbilityRank{};inv:=[]KeyframeInventory{}
  for j:=0;j<run%47;j++ {
   time:=uint64(rng.Intn(30))*100000;slot:=uint32(rng.Intn(5));rank:=domain[rng.Intn(len(domain))]
   if j%13==12 {time=^uint64(0)}
   if run%5==0 {rank=1}
   ranks=append(ranks,types.AbilityRank{TimestampUS:time,Slot:slot,Rank:rank})
   inv=append(inv,KeyframeInventory{TimestampUS:time,Slot:slot,AbilityRank:domain[(run+j)%len(domain)]})
  }
  tracks:=[]Track{};slots:=[]uint32{}
  for slot:=uint32(0);slot<5;slot++ {if (int(slot)+run)%3!=0 {slots=append(slots,slot);tracks=append(tracks,Track{Slot:slot})}}
  palettes:=[]AbilityPalette{};paletteRows:=[]any{}
  for j:=0;j<run%4;j++ {
   p:=AbilityPalette{ID:[]string{"a","b","c"}[j],Markers:[]int{j-1,j,j+1},Ranks:map[int]Label{},Families:map[int]string{}}
   for rank:=-2;rank<4;rank++ {p.Ranks[rank]=Label{En:"name",Fr:"nom",Img:"icon",Tinted:rank%2==0,Family:"stale"};if rank%2==0 {p.Families[rank]="grapple"}}
   palettes=append(palettes,p);paletteRows=append(paletteRows,paletteJSON(&p))
  }
  raw:=buildAbilityReads(ranks,inv,origin,step);before:=append([]AbilityRead{},raw...)
  clean,noise:=rejectAbilityScanNoise(raw);published:=keepAbilitiesOfPublishedTracks(clean,tracks)
  cov:=buildAbilityCoverage(raw,clean,published,noise);chosen:=classifyAbilityPalette(published,palettes);labels:=abilityLabelsUsed(published,chosen)
  if published==nil {published=[]AbilityRead{}};if labels==nil {labels=map[string]Label{}}
  rows=append(rows,map[string]any{"ranks":ranks,"inventory":inv,"origin":origin,"step":step,"slots":slots,"palettes":paletteRows,"raw":before,"output":map[string]any{"reads":published,"coverage":cov,"palette":paletteJSON(chosen),"labels":labels}})
 }
 raw,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-facts-abilities.json",raw,0600);err!=nil {t.Fatal(err)}
}
