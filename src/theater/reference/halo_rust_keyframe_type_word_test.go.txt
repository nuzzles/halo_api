func TestHaloRustKeyframeTypeWord(t *testing.T){
 rng:=rand.New(rand.NewSource(64141));rows:=[]any{}
 for c:=0;c<512;c++ {
  data:=make([]byte,256);rng.Read(data);start:=c%8
  word:=[]uint32{5,69,0xffffffff,50,63,0,0x12345685,0xffffffc0,23,41,44}[c%11]

  ctx:=ContexteParDefaut();ctx.Profil.Cadre.EnTeteBits=[]int{64,108,160}[(c/5)%3];ctx.Profil.Cadre.MotDeTailleBits=[]int{0,1,16,32}[(c/3)%4];ctx.Profil.Grammaire.ControleDeCorruption=c%2==0;ctx.Obs=NouvelleObservation();obs:=haloRustComponentHooks(ctx.Obs)
  put:=func(at,n int,v uint32){for j:=0;j<n;j++{m:=byte(1<<uint(7-(at+j)%8));data[(at+j)/8]&=^m;if v&(1<<uint(n-1-j))!=0{data[(at+j)/8]|=m}}}
  put(start,32,id);put(start+32,32,word);hdr,mot:=ctx.Profil.Cadre.EnTeteBits,ctx.Profil.Cadre.MotDeTailleBits;n1:=uint32(0);ti:=word&63;if ti==23||ti==41||ti==44{n1=1};put(start+hdr,mot,n1);n2at:=start+hdr+mot;if n1>0&&mot>0&&ctx.Profil.Grammaire.ControleDeCorruption{put(n2at,mot,0);n2at+=mot};put(n2at,mot,1)
  reg:=&Registry{Archetypes:make([]Archetype,50)};for j:=range reg.Archetypes{reg.Archetypes[j].Index=j};reg.Archetypes[5].Components=[]string{"biped-emp-timer-component"};reg.Archetypes[5].Levels=[]uint32{0}
  tr:=WalkKeyframeFullState(data,start,reg,ctx)
  table:=make([]byte,len(data)+1);for bit:=start;bit<len(data)*8;bit++{if data[bit/8]&(1<<uint(7-bit%8))!=0{at:=bit-start+1;table[at/8]|=1<<uint(7-at%8)}}
  ctx.Obs=NouvelleObservation();recs,stop:=WalkKeyframeRecords(table,reg,ctx);firstEnd,firstDesync,firstTI:=0,0,uint32(0);if len(recs)>0{firstEnd=recs[0].BitEnd;firstDesync=recs[0].DesyncAt;firstTI=recs[0].Archetype}
  rows=append(rows,map[string]any{"hex":hex.EncodeToString(data),"table_hex":hex.EncodeToString(table),"start":start,"word":word,"id":id,"layout":map[string]any{"header_bits":hdr,"size_word_bits":mot},"corruption":ctx.Profil.Grammaire.ControleDeCorruption,"ti":tr.TypeIndex,"end":tr.EndBit,"desync":tr.DesyncAt,"observations":*obs,"table_count":len(recs),"table_end":firstEnd,"table_desync":firstDesync,"table_ti":firstTI,"table_stop":stop.String()})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-keyframe-type-word.json",b,0600);e!=nil{t.Fatal(e)}
}
