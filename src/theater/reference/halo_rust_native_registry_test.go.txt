package grammar

import (
 "encoding/hex"
 "encoding/json"
 "os"
 "testing"
)

// Calls the pinned native slot readers and tail predicate. These tentative
// reads include the boundary block that parseRegistry refuses to publish.
func TestHaloRustNativeRegistry(t *testing.T) {
 input,err:=os.ReadFile("/private/tmp/halo-native-registry-input.json"); if err!=nil {t.Fatal(err)}
 var cases []struct{Hex string}; if err=json.Unmarshal(input,&cases);err!=nil{t.Fatal(err)}
 for mode:=0;mode<6;mode++ {
  d:=make([]byte,8+archetypeBlockSize*2); d[0]=41;d[4]=27
  for slot:=0;slot<64;slot++ {off:=8+slot*registrySlotSize;copy(d[off:],"component");d[off+256]=byte(slot)}
  switch mode {
  case 0: // All 64 slots accepted; no terminator in this block.
  case 1: d[8+9]=0;d[8+255]=0xfe // Ignored string tail must survive.
  case 2: d[8]=0xff // Invalid first name rejects the block.
  case 3: d[8+20*registrySlotSize]=0 // Empty name, nonzero tail.
  case 4: for i:=8;i<8+archetypeBlockSize;i++ {d[i]=0};d[8+256]=7
  case 5: for i:=8;i<8+archetypeBlockSize;i++ {d[i]=0};d[8+archetypeBlockSize-1]=1
  }
  cases=append(cases,struct{Hex string}{hex.EncodeToString(d)})
 }
 rows:=[]any{}
 for _,c:=range cases {
  d,e:=hex.DecodeString(c.Hex);if e!=nil{t.Fatal(e)}
  reg,e:=ParseRegistryChunk(d)
  row:=map[string]any{"hex":c.Hex,"error":e!=nil};blocks:=[]any{}
  if e==nil {
   whole,_:=registryWholeBlocks(len(d))
   for b:=0;b<whole;b++ {
    base:=registryEntryBase+b*archetypeBlockSize;slots:=[]any{};run:=0
    for s:=0;s<archetypeBlockSlots;s++ {
     off:=base+s*registrySlotSize;name:=entryName(d,off)
     slot:=map[string]any{"index":s,"start_byte":off,"name_hex":hex.EncodeToString(d[off:off+256]),"name":name}
     if name!="" {slot["level"]=entryLevel(d,off)}
     slots=append(slots,slot);if name==""{break};run++
    }
    accepted:=registryBlockTail(d,base,run);first:=-1
    for off:=base+run*registrySlotSize;off<base+archetypeBlockSize;off++ {if d[off]!=0{first=off;break}}
    end:=base+archetypeBlockSize;if first>=0{end=first+1}
    blocks=append(blocks,map[string]any{"index":b,"start_byte":base,"end_byte":base+archetypeBlockSize,"slots":slots,"tail_start_byte":base+run*registrySlotSize,"tail_checked_end_byte":end,"first_nonzero_byte":first,"accepted":accepted})
    if !accepted{break}
   }
   row["accepted_blocks"]=len(reg.Archetypes)
  }
  row["blocks"]=blocks;rows=append(rows,row)
 }
 output,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-native-registry.json",output,0600);err!=nil{t.Fatal(err)}
}
