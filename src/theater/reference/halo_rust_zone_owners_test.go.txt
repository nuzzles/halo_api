package replay
import("testing";"encoding/json";"os";"math/rand";"fmt")
func TestHaloRustZoneOwners(t *testing.T){
 rng:=rand.New(rand.NewSource(41600));rows:=[]any{}
 samples:=func(ss []zoneSample)[]any{out:=[]any{};for _,s:=range ss{out=append(out,map[string]any{"t":s.t,"v":s.v})};return out}
 for i:=0;i<1024;i++{
  owner:=[]zoneSample{};gauge:=[]zoneSample{};for j:=0;j<i%80;j++{v:=[]uint64{0,1,2,5,0xffffffff,0xffffffffffffffff}[rng.Intn(6)];owner=append(owner,zoneSample{t:j/2+3,v:v});gauge=append(gauge,zoneSample{t:j/3,v:uint64(8300000+rng.Intn(250000))})}
  teams:=map[string]int{};switch i%4{case 1:teams=map[string]int{"a":0,"b":1};case 2:teams=map[string]int{"a":2,"b":5,"c":-1};case 3:teams=map[string]int{"a":-1}}
  frames:=[]int{-1,0,1,5,20,100}[i%6];cov:=&ZonesCoverage{};spans:=ownerSpansOf(owner,gauge,zoneSpanCtx{frames:frames,teams:zoneTeamSet(teams)},cov)
  ser:=zoneSeries{owner:map[uint32][]zoneSample{0:owner,1:owner}};owners:=map[int]uint32{0:0,1:1};pairs:=[]zonePair{};pr:=[]any{};for j:=0;j<40;j++{p:=zonePair{t:rng.Intn(50),ref:j%3,xuid:[]string{"a","b","c","unknown"}[j%4]};pairs=append(pairs,p);pr=append(pr,map[string]any{"t":p.t,"zone_ref":p.ref,"xuid":p.xuid})};win:=[]int{-1,0,1,5,20}[i%5];checkOwnerAgreement(ser,owners,pairs,teams,win,cov)
  gauges:=map[int]uint32{};for j:=0;j<i%5;j++{gauges[j]=uint32(rng.Intn(5))};catalog:=i%6;hill:=i%7==0;letters:=zoneLetterRanks(gauges,catalog,hill);if letters==nil{letters=map[int]int{}}
  rows=append(rows,map[string]any{"owner":samples(owner),"gauge":samples(gauge),"frames":frames,"teams":teams,"spans":spans,"unknown":cov.UnknownOwner,"series":map[string]any{"owner":map[string]any{"0":samples(owner),"1":samples(owner)},"gauge":map[string]any{},"keys":map[string]any{},"desig":map[string]any{},"owner_chained":map[string]any{},"slots":0},"owners":owners,"pairs":pr,"win":win,"checked":cov.OwnerChecked,"agreed":cov.OwnerAgreed,"gauges":gauges,"catalog":catalog,"hill":hill,"letters":letters})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-zone-owners.json",b,0600);e!=nil{t.Fatal(fmt.Sprint(e))}
}
