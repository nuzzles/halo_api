func haloRustStateValue(v reflect.Value)any{
 switch v.Kind(){
 case reflect.String:out:=[]uint64{};for _,b:=range []byte(v.String()){out=append(out,uint64(b))};return out

 };panic(v.Kind())
}
func haloRustStateFill(v reflect.Value,rng *rand.Rand,n int){switch v.Kind(){
 case reflect.String:v.SetString([]string{"", "crouch", "sprint", "\x00\xff\xc0\x80", "跳跃"}[n%5])
 case reflect.Pointer:if n%3!=0{v.Set(reflect.New(v.Type().Elem()));haloRustStateFill(v.Elem(),rng,n)}

 case reflect.Array:for i:=0;i<v.Len();i++{haloRustStateFill(v.Index(i),rng,n+i)}
 case reflect.Struct:for i:=0;i<v.NumField();i++{haloRustStateFill(v.Field(i),rng,n+i)}
 case reflect.Int,reflect.Int64,reflect.Int32,reflect.Int8:v.SetInt(int64(rng.Uint64()))
 case reflect.Uint,reflect.Uint64,reflect.Uint32,reflect.Uint16,reflect.Uint8:v.SetUint(rng.Uint64())
 case reflect.Bool:v.SetBool(rng.Intn(2)==0)
 case reflect.Float32:v.Set(reflect.ValueOf(math.Float32frombits(rng.Uint32())))
 case reflect.Float64:v.SetFloat(math.Float64frombits(rng.Uint64()))
 default:panic(v.Kind())}}
