func TestHaloRustFramesParity(t *testing.T) {
 reg:=&Registry{Archetypes:make([]Archetype,4)}
 reg.Archetypes[3]=Archetype{Index:3,Components:[]string{"high-frequency"},Levels:[]uint32{1}}
 rows:=[]any{}
 for sample:=0;sample<32;sample++ {
  data:=make([]byte,2048);bit:=0
  put:=func(v uint64,n int){for j:=n-1;j>=0;j-- {if v&(uint64(1)<<uint(j))!=0 {data[bit/8]|=1<<uint(7-bit%8)};bit++}}
  cfg:=DefaultFrameConfig();cfg.IDLowBits=5;cfg.HasExtraFields=sample&1!=0
  cfg.Profil.Grammaire.ControleDeCorruption=sample&2!=0
  cfg.Profil.Grammaire.DeserEtatParArchetype=true
  cfg.Profil.Grammaire.GenerationStricte=true
  prefix:=func(){if cfg.HasExtraFields{put(0x12345678,32)}}
  extra:=func(){if cfg.HasExtraFields{put(1,1);put(0x39,8)}}
  check:=func(){if cfg.Profil.Grammaire.ControleDeCorruption{put(1,1);put(0xabcdef01,32)}}
  create:=func(gen uint64){prefix();put(1,3);put(1,5);put(gen,2);extra();put(3,6)
   put(0,1);put(0,1);put(0,1);put(1,1);put(42,8) // ti3 default
   put(1,1);put(0,1);put(1,3);put(0,6);put(uint64(sample),8);check()}
  delta:=func(gen uint64){prefix();put(1,1);put(1,5);put(gen,2);put(1,1);put(17,7);put(1,1);put(1,64);put(99,8);check()}
  put(1,1);put(0,1) // configuration, empty message view
  create(1);delta(1)
  prefix();put(2,3);put(1,5);put(1,2);extra();put(0x91827364,32)
  create(2)
  if sample&4!=0{delta(1)}else{delta(2)}
  prefix();put(0,3);put(0,1) // entity End, empty control view
  br:=LecteurSur(data[:(bit+7)/8]);w:=NewWorld(reg)
  records,err:=DecodeFrameRecords(br,w,cfg)
  relaxedCfg:=cfg;relaxedCfg.Profil.Grammaire.GenerationStricte=false
  relaxedReader:=LecteurSur(data[:(bit+7)/8]);relaxedWorld:=NewWorld(reg)
  relaxedRecords,relaxedErr:=DecodeFrameRecords(relaxedReader,relaxedWorld,relaxedCfg)
  pw:=NewWorld(reg); production,views,productionEnd:=DecodeFrameViewsCurseur(data[:(bit+7)/8],pw,cfg,3,2)
  tails:=[]any{}
  for _,cut:=range []int{0,1,2,5,(bit+7)/8-1,(bit+7)/8} {
   for _,strict:=range []bool{false,true} {
    tc:=cfg;tc.Profil.Grammaire.GenerationStricte=strict
    tr:=LecteurSur(data[:cut]);tr.SetBitPos(2);tw:=NewWorld(reg)
    trecs,terr:=DecodeFrameRecords(tr,tw,tc)
    if trecs==nil { trecs=[]FrameRecord{} }
    tails=append(tails,map[string]any{"cut":cut,"strict":strict,"records":trecs,"end":tr.BitPos(),"complete":terr==nil})
   }
  }
  wildcards:=[]any{}
  for mode:=0;mode<4;mode++ {for _,strict:=range []bool{false,true} {
   wc:=cfg;wc.Profil.Grammaire.GenerationStricte=strict
   wr:=LecteurSur(data[:(bit+7)/8]);wr.SetBitPos(records[1].HeaderBit);ww:=NewWorld(reg)
   switch mode {case 1:ww.BindFull(0x40000001,3);case 2:ww.BindFull(0x80000001,3);case 3:ww.BindWildcard(0xc0000001,3)}
   wrecs,werr:=DecodeFrameRecords(wr,ww,wc)
   if wrecs==nil {wrecs=[]FrameRecord{}}
   state,exists:=ww.slots[1]
   wildcards=append(wildcards,map[string]any{"mode":mode,"strict":strict,"start":records[1].HeaderBit,"records":wrecs,"complete":werr==nil,"end":wr.BitPos(),"exists":exists,"id":state.FullID,"generation_any":state.GenAny})
  }}
  readers:=[]any{}
  for _,start:=range []int{0,2,records[1].HeaderBit,((bit+7)/8)*8+3} {for _,preamble:=range []int{-1,0,2,7} {
   rc:=cfg;rc.PacketPreambleBits=preamble
   rr:=LecteurSur(data[:(bit+7)/8]);rr.SetBitPos(start);rw:=NewWorld(reg)
   rrecs,rerr:=DecodeFrameRecords(rr,rw,rc)
   if rrecs==nil {rrecs=[]FrameRecord{}}
   readers=append(readers,map[string]any{"start":start,"preamble":preamble,"records":rrecs,"complete":rerr==nil,"end":rr.BitPos()})
  }}
  simulations:=[]any{}
  for _,complete:=range []bool{false,true} {
   sc:=cfg;sc.Profil.Grammaire.SimStateComplet=complete
   sr:=&Registry{Archetypes:make([]Archetype,4)};copy(sr.Archetypes,reg.Archetypes)
   sr.Archetypes[3].Components=[]string{"simulation-state"}
   sb:=LecteurSur(data[:(bit+7)/8]);sw:=NewWorld(sr)
   srecs,serr:=DecodeFrameRecords(sb,sw,sc)
   if srecs==nil {srecs=[]FrameRecord{}}
   simulations=append(simulations,map[string]any{"policy":complete,"records":srecs,"complete":serr==nil,"end":sb.BitPos()})
  }
  rows=append(rows,map[string]any{"simulations":simulations,"readers":readers,"wildcards":wildcards,"tails":tails,"relaxed_records":relaxedRecords,"relaxed_end":relaxedReader.BitPos(),"relaxed_complete":relaxedErr==nil,"production":production,"production_views":views,"production_end":productionEnd,"extra":cfg.HasExtraFields,"check":cfg.Profil.Grammaire.ControleDeCorruption,"complete":err==nil,"end_bit":br.BitPos(),"records":records,"hex":hex.EncodeToString(data[:(bit+7)/8])})
 }
 data,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-levelup-frames-oracle.json",data,0600);err!=nil{t.Fatal(err)}
}

func TestHaloRustProductionAdmission(t *testing.T) {
 reg:=&Registry{Archetypes:make([]Archetype,4)}
 reg.Archetypes[3]=Archetype{Index:3,Components:[]string{"high-frequency"},Levels:[]uint32{1}}
 rows:=[]any{}
 for mode:=0;mode<8;mode++ {
  for _,start:=range []int{2,13} {
   data:=make([]byte,32);bit:=0
   put:=func(v uint64,n int){for j:=n-1;j>=0;j--{if v&(uint64(1)<<uint(j))!=0{data[bit/8]|=1<<uint(7-bit%8)};bit++}}
   put(0,start);put(1,1);put(1,5);put(1,2);put(0,1);put(0,1);if start==13 {put(2,3);put(0,6);put(1,6)}else{put(1,3);put(0,6)};put(77,8);put(0,3);put(0,1)
   cfg:=DefaultFrameConfig();cfg.IDLowBits=5;cfg.Obs=NouvelleObservation()
   w:=NewWorld(reg)
   switch mode {
   case 1:w.BindFull(0x40000001,3)
   case 2:w.BindFull(0x80000001,3)
   case 3:w.PoserVueCourante(1);w.BindFull(0x40000001,3)
   case 4:w.BindDatum(1,3)
   case 5,6,7:
    table:=NouvelleTableAnticipee();head:=uint8(1);chunk:=5;if mode==6{chunk=0};if mode==7{head=2}
    table.entrees[cleAnticipee{slot:1,tete:head}]=[]declarationAnticipee{{chunk:chunk,ti:3}};table.Clore();w.PoserTableAnticipee(table)
   }
   records,views,end:=DecodeFrameViewsCurseur(data[:(bit+7)/8],w,cfg,3,start)
   ant:=map[uint32]int{};for k,v:=range cfg.Obs.LiaisonsParAnticipation{ant[k]=v};diag:=map[string]any{"rejected_unbound":cfg.Obs.RejetsHorsDatum,"rejected_other_view":cfg.Obs.RejetsDeVue,"anticipated_bindings":ant}
   rows=append(rows,map[string]any{"diagnostics":diag,"mode":mode,"start":start,"hex":hex.EncodeToString(data[:(bit+7)/8]),"records":records,"views":views,"end":end,"anticipations":w.AnticipationsParArchetype()})
  }
 }
 data,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-levelup-production-admission.json",data,0600);err!=nil{t.Fatal(err)}
}

func TestHaloRustStrictLocator(t *testing.T) {
 reg:=&Registry{Archetypes:make([]Archetype,4)}
 reg.Archetypes[3]=Archetype{Index:3,Components:[]string{"high-frequency"},Levels:[]uint32{1}}
 rows:=[]any{}
 for sample:=0;sample<256;sample++ {
  data:=make([]byte,32);bit:=0
  put:=func(v uint64,n int){for j:=n-1;j>=0;j--{if v&(uint64(1)<<uint(j))!=0{data[bit/8]|=1<<uint(7-bit%8)};bit++}}
  for i:=0;i<2+sample%37;i++{put(uint64((i*13+sample)%2),1)}
  put(uint64(sample%2),1)
  cfg:=DefaultFrameConfig();cfg.IDLowBits=13
  w:=NewWorld(reg);w.BindFull(0x8000007b,3)
  if sample%4==0 {w.Unbind(123)}
  put(1,1);put(123,13);put(1,2);put(0,1);put(0,1);put(1,3);put(0,6);put(uint64(sample),8)
  if sample%8!=0 {put(0,8)}
  payload:=data[:(bit+7)/8]
  rows=append(rows,map[string]any{"hex":hex.EncodeToString(payload),"bound":sample%4!=0,"offset":marchLocateStrict(payload,w,cfg)})
 }
 data,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-levelup-strict-locator.json",data,0600);err!=nil{t.Fatal(err)}
}
