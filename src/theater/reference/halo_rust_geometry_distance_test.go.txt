func TestHaloRustGeometryDistance(t *testing.T) {
 rows:=[]any{}
 var state uint64=0x51adebeef
 for i:=0;i<4096;i++ {
  var a,b [3]float32
  var ab,bb [3]uint32
  for j:=0;j<3;j++ {
   state=state*6364136223846793005+1442695040888963407
   ab[j]=(uint32(state)&0x807fffff)|(uint32(100+state%50)<<23)
   state=state*6364136223846793005+1442695040888963407
   bb[j]=(uint32(state)&0x807fffff)|(uint32(100+state%50)<<23)
   a[j]=math.Float32frombits(ab[j]);b[j]=math.Float32frombits(bb[j])
  }
  rows=append(rows,map[string]any{"a":ab,"b":bb,"dist3":math.Float64bits(dist3(a,b)),"plan":math.Float64bits(planDist(a[0],a[1],b[0],b[1]))})
 }
 special:=[]uint32{0,0x80000000,1,0x80000001,0x7f7fffff,0xff7fffff,0x7f800000,0xff800000,0x7fc00001,0xffc00001}
 for _,x:=range special{for _,y:=range special{a:=[3]float32{math.Float32frombits(x),math.Float32frombits(y),0};b:=[3]float32{};rows=append(rows,map[string]any{"a":[3]uint32{x,y,0},"b":[3]uint32{},"dist3":math.Float64bits(dist3(a,b)),"plan":math.Float64bits(planDist(a[0],a[1],0,0))})}}
 for _,radius:=range []float32{0.5,4,10}{for exponent:=-150;exponent<=4;exponent++{x,y:=radius,float32(math.Ldexp(1,exponent));a:=[3]float32{x,y,0};rows=append(rows,map[string]any{"a":[3]uint32{math.Float32bits(x),math.Float32bits(y),0},"b":[3]uint32{},"dist3":math.Float64bits(dist3(a,[3]float32{})),"plan":math.Float64bits(planDist(x,y,0,0))})}}
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-geometry-distance.json",raw,0600);e!=nil{t.Fatal(e)}
}

func TestHaloRustGeometryDistanceGrenades(t *testing.T) {
 rows:=[]any{}
 for mode:=0;mode<4;mode++ {
  radius:=float32(1);if mode>0{radius=4}
  small:=float32(math.Ldexp(float64(radius),-26));large:=math.Float32frombits(math.Float32bits(small)+1)
  if mode==3{large=small*2}
  pos:=[]grammar.BipedPosition{{Slot:1,TimestampUS:1000000,HasWorld:true}}
  throws:=[]grammar.GrenadeThrow{{TimestampUS:1000000,FilmIndex:0,TypeID:0xb0171062}}
  owner:=map[uint32]int{1:0};links:=map[int]int{0:10,1:11}
  proj:=[]types.ProjectileTrack{{Slot:2,Pts:[]types.ProjectileSample{{TimestampUS:1000000,X:radius,Y:large}}}}
  if mode<2{proj=append(proj,types.ProjectileTrack{Slot:3,Pts:[]types.ProjectileSample{{TimestampUS:1000001,X:radius,Y:small}}})}
  grenades,cov:=buildGrenades(pos,throws,0,100000,owner,proj,links)
  if grenades==nil{grenades=[]Grenade{}}
  rows=append(rows,map[string]any{"positions":pos,"throws":throws,"owners":owner,"projectiles":proj,"links":links,"output":map[string]any{"grenades":grenades,"coverage":cov}})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-geometry-distance-grenades.json",raw,0600);e!=nil{t.Fatal(e)}
}
