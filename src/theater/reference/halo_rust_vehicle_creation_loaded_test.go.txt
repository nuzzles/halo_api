package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"testing"
)

func TestHaloRustVehicleCreationLoaded(t *testing.T) {
	var samples []struct {
		Map    profile.MapQuantEntry `json:"map"`
		Widths [2]int                `json:"widths"`
		Hex    string                `json:"hex"`
		Layout struct {
			GateBits int     `json:"gate_bits"`
			AxisW    [3]uint `json:"axis_widths"`
			Region   uint32  `json:"region"`
		} `json:"layout"`
	}
	raw, e := os.ReadFile("/private/tmp/halo-vehicle-creation-layout.json")
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(raw, &samples); e != nil {
		t.Fatal(e)
	}
	rows := []any{}
	packet := func(kind uint16, ts uint64, pay []byte) []byte {
		h := make([]byte, 16)
		binary.LittleEndian.PutUint16(h, kind)
		binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
		binary.LittleEndian.PutUint64(h[8:], ts)
		return append(h, pay...)
	}
	for run := 0; run < 256; run++ {
		sample := samples[(run*4)%len(samples)]
		pay, _ := hex.DecodeString(sample.Hex)
		lay := profile.I0Layout{GateBits: sample.Layout.GateBits, AxisW: sample.Layout.AxisW, Region: sample.Layout.Region}
		wr := sample.Map.Range()
		header := make([]byte, 8)
		binary.LittleEndian.PutUint32(header, 41)
		binary.LittleEndian.PutUint32(header[4:], 27)
		chunks := source.MemoryChunks{header}
		meta := []types.ChunkMeta{{Index: 0, ChunkType: 1}}
		inputs := []any{}
		for c := 0; c < 3; c++ {
			if run%13 == 0 {
				break
			}
			data := []byte{}
			w := &translocWriter{}
			w.put(1, 0)
			for _, slot := range []uint32{512, 515} {
				w.put(32, uint64(1<<30|slot))
				ti := uint64(40)
				if run%11 == 0 {
					ti = 35
				}
				w.put(32, ti)
				w.put(44, 0)
				w.put(32, 0)
				w.put(32, 0)
			}
			w.put(64, 0)
			data = append(data, packet(2, 9000, w.pay)...)
			for j := 0; j < 4; j++ {
				kind := uint16(0)
				if j == 1 {
					kind = 3
				}
				data = append(data, packet(kind, uint64(j%2*1000), pay)...)
			}
			if run%17 == 0 {
				data = data[:len(data)-9]
			}
			index := c + 1
			if run%7 == 0 && c == 1 {
				index = 1
			}
			if run%19 == 0 && c == 1 {
				index = 3
			}
			chunks = append(chunks, data)
			meta = append(meta, types.ChunkMeta{Index: index})
			inputs = append(inputs, map[string]any{"index": index, "hex": hex.EncodeToString(data)})
		}
		film, e := source.Load(chunks, meta)
		if e != nil {
			t.Fatal(e)
		}
		fc := NewFilmContext(film)
		fc.layLu = true
		fc.lay = lay
		layoutOK := run%5 != 0
		if !layoutOK {
			fc.layErr = fmt.Errorf("layout oracle refusal")
		}
		reg := &Registry{Archetypes: make([]Archetype, 41)}
		for i := range reg.Archetypes {
			reg.Archetypes[i].Index = i
		}
		reg.Archetypes[40].Components = make([]string, 33)
		reg.Archetypes[40].Levels = make([]uint32, 33)
		archOK := run%9 != 0
		if !archOK {
			reg.Archetypes = reg.Archetypes[:40]
		}
		registryOK := run%23 != 0
		fc.regLu = true
		fc.reg = reg
		if !registryOK {
			fc.regErr = ErrNoRegistryChunk
		}
		p := ProfilDeBalayageParDefaut()
		p.PoserLargeursObjetDuMonde(profile.PrecisionDescriptor{IndexW: uint(sample.Map.EffectiveRegionIndexBits()), AxisW: sample.Map.Layout().AxisW, Region: sample.Map.Layout().Region})
		p.MPP = profile.MPPWidths{Lead: sample.Widths[0], Index: sample.Widths[1]}
		fc.bal = p
		boundsOK := run%6 != 0
		var bounds *profile.Vec3Range
		if boundsOK {
			bounds = &wr
		}
		band := map[uint32]bool{512: true, 514: true, 515: true}
		if run%8 == 0 {
			band = map[uint32]bool{}
		}
		top := (run/16)%2 == 0
		var records []types.EquipmentCreation
		var st types.EquipmentCreationStats
		if top {
			records, st, e = ScanVehicleCreations(fc, bounds)
		} else {
			records, st, e = ScanVehicleCreationsForBand(fc, bounds, band)
		}
		if records == nil {
			records = []types.EquipmentCreation{}
		}
		errKind := ""
		if e != nil {
			switch {
			case len(fc.ChunkNumbers()) == 0 && (top || boundsOK):
				errKind = "chunks"
			case top && len(worldObjectSlotBand(film, 40)) == 0:
				errKind = "band"
			case !boundsOK:
				errKind = "bounds"
			case !layoutOK:
				errKind = "layout"
			case !registryOK:
				errKind = "registry"
			case !archOK:
				errKind = "archetype"
			default:
				t.Fatal(e)
			}
		}
		slots := []uint32{}
		for _, s := range []uint32{512, 514, 515} {
			if band[s] {
				slots = append(slots, s)
			}
		}
		rows = append(rows, map[string]any{"chunks": inputs, "map": sample.Map, "layout": sample.Layout, "widths": sample.Widths, "top": top, "band": slots, "bounds_ok": boundsOK, "layout_ok": layoutOK, "registry_ok": registryOK, "arch_ok": archOK, "records": records, "stats": st, "error": errKind})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-vehicle-creation-loaded.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
