package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"os"
	"path/filepath"
	"testing"
)

func TestHaloRustResearchDirectory(t *testing.T) {
	var prior []struct {
		Files   map[string]string
		Missing bool
	}
	raw, e := os.ReadFile("/private/tmp/halo-ground-directory.json")
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(raw, &prior); e != nil {
		t.Fatal(e)
	}
	rows := []any{}
	for run := 0; run < 34; run++ {
		dir := t.TempDir()
		files := map[string]string{}
		folder := ""
		missing := false
		if run < 32 {
			c := prior[run]
			missing = c.Missing
			for name, h := range c.Files {
				files[name] = h
			}
			if h, ok := files["chunk_01.bin"]; ok && len(h) > 2 {
				w := &translocWriter{}
				for j := 0; j < 24; j++ {
					w.put(1, 1)
					w.put(13, 512+uint64(j%8))
					w.put(2, uint64(j%4))
					w.put(2, 0)
					w.put(3, 1)
					w.put(6, 0)
					w.put(2, 0)
					w.put(1, 0)
					w.put(13, 1000+uint64(j))
					w.put(13, 2000+uint64(j))
					w.put(14, 3000+uint64(j))
					w.put(2, 0)
				}
				head := make([]byte, 16)
				binary.LittleEndian.PutUint32(head[4:], uint32(len(w.pay)))
				binary.LittleEndian.PutUint64(head[8:], 7654)
				files["chunk_01.bin"] = h + hex.EncodeToString(append(head, w.pay...))
			}
			if missing {
				dir = filepath.Join(dir, "missing")
			}
			for name, h := range files {
				d, _ := hex.DecodeString(h)
				if e = os.WriteFile(filepath.Join(dir, name), d, 0600); e != nil {
					t.Fatal(e)
				}
			}
		} else {
			folder = []string{"maps/01-bazaar-idle", "maps/02-aquarius"}[run-32]
			root := filepath.Join("/Users/simbleau/git/halo_api/experiments/films", folder)
			var manifest struct {
				Chunks []struct {
					Index int    `json:"index"`
					File  string `json:"file"`
				}
			}
			b, err := os.ReadFile(filepath.Join(root, "film.json"))
			if err != nil {
				t.Fatal(err)
			}
			if err = json.Unmarshal(b, &manifest); err != nil {
				t.Fatal(err)
			}
			for _, c := range manifest.Chunks {
				if c.Index > 7 {
					continue
				}
				b, err := os.ReadFile(filepath.Join(root, c.File))
				if err != nil {
					t.Fatal(err)
				}
				if err = os.WriteFile(filepath.Join(dir, fmt.Sprintf("chunk_%02d.bin", c.Index)), b, 0600); err != nil {
					t.Fatal(err)
				}
			}
		}
		wr := &profile.Vec3Range{{Min: -100, Max: 300}, {Min: -200, Max: 400}, {Min: -300, Max: 500}}
		if run < 32 && run%5 == 0 {
			wr = nil
		}
		band := map[uint32]bool{512: true, 513: true, 515: true}
		if run%7 == 0 {
			band = map[uint32]bool{}
		}
		slots := GroundWeaponSlotBand(dir)
		if slots == nil {
			slots = map[uint32]bool{}
		}
		precision := ProfilDeBalayageParDefaut().LargeursObjetDuMonde()
		if f, err := source.LoadDir(dir, nil); err == nil {
			precision = contexteDeBobine(f).LargeursObjetDuMonde()
		}
		rows = append(rows, map[string]any{"folder": folder, "files": files, "missing": missing, "bounds": wr != nil, "band": band, "slots": slots, "precision": precision, "samples": WorldObjectPositionsForBand(dir, wr, band), "ground": GroundWeaponPositions(dir, wr)})
	}
	raw, e = json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-research-directory.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
