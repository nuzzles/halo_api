package replay
import("testing";"os";"encoding/json";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
type rustClosureLife struct{Slot uint32 `json:"slot"`;From int64 `json:"from"`;To int64 `json:"to"`;XUID uint64 `json:"xuid"`;Bid string `json:"bid"`;Cause string `json:"cause"`;Named string `json:"named_by"`}
func TestHaloRustIdentityClosures(t *testing.T){
 rng:=rand.New(rand.NewSource(7621));rows:=[]any{}
 for run:=0;run<1024;run++{
  lives:=[]lifeSpan{};positions:=[]grammar.BipedPosition{};owners:=map[uint32]int{};indices:=map[uint64]int{};deaths:=[]Death{};fire:=[]FireEventRef{};offset:=int64(run%3-1)*1000
  for i:=0;i<run%25;i++{slot:=uint32(rng.Intn(8));from:=int64(rng.Intn(40))*1000000;to:=from+int64(1+rng.Intn(8))*1000000;x:=uint64(0);if i%3==0{x=uint64(100+rng.Intn(4));owners[slot]=int(x-100);indices[x]=int(x-100)};lives=append(lives,lifeSpan{slot:slot,from:from,to:to,xuid:x});if i%7!=0{positions=append(positions,grammar.BipedPosition{Slot:slot,TimestampUS:uint64(from),HasWorld:i%2==0},grammar.BipedPosition{Slot:slot,TimestampUS:uint64(to),HasWorld:true})};deaths=append(deaths,Death{XUID:uint64(100+rng.Intn(4)),TimeMS:int64(rng.Intn(40))*1000});fire=append(fire,FireEventRef{FilmIndex:rng.Intn(4),TimestampUS:uint64(rng.Intn(40))*1000000})}
  if run%4==0{offset=0;lives=[]lifeSpan{{slot:1,from:0,to:1000000,xuid:100},{slot:2,from:2000000,to:3000000}};owners=map[uint32]int{1:0};indices=map[uint64]int{100:0};positions=[]grammar.BipedPosition{{Slot:1,TimestampUS:0},{Slot:1,TimestampUS:1000000},{Slot:2,TimestampUS:2000000},{Slot:2,TimestampUS:3000000}};fire=[]FireEventRef{{FilmIndex:0,TimestampUS:2500000}};deaths=[]Death{}}
  if run%4==1{offset=0;lives=[]lifeSpan{{slot:1,from:10000000,to:11000000,xuid:100},{slot:2,from:20000000,to:21000000}};owners=map[uint32]int{1:0};indices=map[uint64]int{100:0,101:1};positions=[]grammar.BipedPosition{{Slot:1,TimestampUS:10000000},{Slot:1,TimestampUS:11000000},{Slot:2,TimestampUS:20000000},{Slot:2,TimestampUS:21000000}};deaths=[]Death{{XUID:100,TimeMS:2000},{XUID:101,TimeMS:12000}};fire=[]FireEventRef{}}
  if run%9==0&&len(lives)>1{l:=lives[len(lives)-1];l.slot+=20;lives=append(lives,l)}
  if run%13==0{indices=map[uint64]int{}}
  input:=[]rustClosureLife{};for _,l:=range lives{input=append(input,rustClosureLife{l.slot,l.from,l.to,l.xuid,l.bid,l.cause,l.nomPar})};after,rep:=closeBridge(indexBySlot(positions),owners,lives,deaths,offset,indices,fire);if rep.closedLife==nil{rep.closedLife=map[uint32]int{}};lo,hi:=respawnWindow(lives,deaths,offset)
  byXUID:=map[uint32]uint64{};for _,l:=range lives{if l.xuid!=0{byXUID[l.slot]=l.xuid}};extended:=extendSlotXUID(byXUID,after,indices);nameClosedLives(lives,after,rep.closedLife,indices);named:=[]rustClosureLife{};for _,l:=range lives{named=append(named,rustClosureLife{l.slot,l.from,l.to,l.xuid,l.bid,l.cause,l.nomPar})}
  rows=append(rows,map[string]any{"named":named,"slot_xuids":byXUID,"extended":extended,"lives":input,"positions":positions,"owners":owners,"indices":indices,"deaths":deaths,"fire":fire,"offset":offset,"window":[]int64{lo,hi},"output":map[string]any{"owners":after,"report":map[string]any{"by_shot":rep.byShot,"by_respawn":rep.byRespawn,"contested":rep.contested,"refused":rep.refused,"closed_life":rep.closedLife}}})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-identity-closures.json",raw,0600);e!=nil{t.Fatal(e)}
}
