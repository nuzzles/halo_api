package objectives

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"testing"
)

func TestHaloRustStatComponents(t *testing.T) {
	catalog := map[string]StatComponent{"mode": ModeScoreComponent, "personal": PersonalScoreComponent, "kills": KillsComponent, "deaths": DeathsComponent, "assists": AssistsComponent, "skull_ticks": SkullTicksComponent, "skull_grabs": SkullGrabsComponent}
	rows := []any{}
	for run := 0; run < 128; run++ {
		recs := []types.StatRecord{}
		for r := 0; r < 3; r++ {
			for _, slot := range []int{6, 8, 10, 12} {
				for j := 0; j < 12; j++ {
					v := int64(j)
					if j == 5 && run%3 == 0 {
						v = 1000
					}
					if j == 4 && run%7 == 0 {
						v = -1
					}
					stamp := r*30000 + j*1000
					if run%4 == 0 {
						stamp = r*30000 + (11-j)*1000
					}
					recs = append(recs, types.StatRecord{TimeMS: stamp, Slot: slot, Round: r, Comps: map[int]types.StatValue{0: {A: v, B: int64(run % 252)}, 1: {B: v * 100}, 2: {A: v, B: int64(j / 2)}, 3: {A: int64(j / 3)}, 21: {B: int64(j / 2)}}})
				}
			}
		}
		queries := []any{}
		for _, name := range []string{"mode", "personal", "kills", "deaths", "assists", "skull_ticks", "skull_grabs"} {
			c := catalog[name]
			for _, teams := range []bool{false, true} {
				queries = append(queries, map[string]any{"name": name, "component": c, "teams": teams, "rounds": SeriesByRound(recs, c, teams), "total": SeriesTotal(recs, c, teams)})
			}
		}
		// Independently supplied descriptor, including unusual indices and both policies.
		c := StatComponent{Comp: run%25 - 2, SideB: run%2 == 0, Strict: run%3 == 0, Unitary: run%4 == 0}
		queries = append(queries, map[string]any{"name": "custom", "component": c, "teams": false, "rounds": SeriesByRound(recs, c, false), "total": SeriesTotal(recs, c, false)})
		rows = append(rows, map[string]any{"records": recs, "queries": queries})
	}
	domain := []any{}
	for _, a := range []int64{-9223372036854775808, -1, 0, 1, 249, 250, 251, 9223372036854775807} {
		for _, b := range []int64{-9223372036854775808, -1, 0, 1, 249, 250, 251, 9223372036854775807} {
			v := types.StatValue{A: a, B: b}
			domain = append(domain, map[string]any{"value": v, "ok": modeScoreInDomain(v)})
		}
	}
	roster := []any{}
    for _, n := range []int{-9223372036854775808, -1, 0, 1, 7, 8, 9, 16, 24, 32, 9223372036854775807} {
        roster = append(roster, map[string]any{"seats": n, "fits": RosterFitsStatborg(n)})
    }
    raw, e := json.Marshal(map[string]any{"catalog": catalog, "cases": rows, "domain": domain, "player_slots": StatPlayerSlots, "roster": roster})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-stat-components.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
