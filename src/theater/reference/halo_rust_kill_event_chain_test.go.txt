package killsource

import("testing";"encoding/json";"encoding/hex";"os";"math/rand")
func TestHaloRustKillEventChain(t *testing.T) {
 type Fields struct { Killer int `json:"killer"`; Victim int `json:"victim"`; Assist int `json:"assist"`; KillerPct uint32 `json:"killer_pct"`; AssistPct uint32 `json:"assist_pct"`; Flag int `json:"flag"`; End int `json:"end"` }
 fields:=func(k killEventFields)Fields{return Fields{k.killer,k.victim,k.assist,k.killerPct,k.assistPct,k.flag,k.end}}
 type Event struct {Bit int `json:"bit"`; Fields Fields `json:"fields"`; Chain int `json:"chain"`}
 type Row struct { Data string `json:"data"`; Position int `json:"position"`; Gate bool `json:"gate"`; Fields Fields `json:"fields"`; Length int `json:"length"`; Events []Event `json:"events"` }
 rows:=[]Row{}
 add:=func(data []byte,p int,g bool){events:=[]Event{};for _,r:=range killEventsIn(data,g){events=append(events,Event{r.bit,fields(r.fields),r.chain})};rows=append(rows,Row{hex.EncodeToString(data),p,g,fields(readKillEvent(data,p)),evChainLen(data,p,g,12),events})}
 rng:=rand.New(rand.NewSource(314159))
 for i:=0;i<2048;i++ {data:=make([]byte,rng.Intn(180));rng.Read(data);p:=rng.Intn(len(data)*8+1);add(data,p,i%2==0)}
 // Every event code, with random presence/body and both runtime configurations.
 for code:=0;code<128;code++ {for j:=0;j<8;j++ {data:=make([]byte,192);rng.Read(data);data[0]=byte(128+code);add(data,0,j%2==0)}}
 // Real grammar: victim, killer, integer shares (including >100), assist,
 // followed by a controlled number of complete stub events and a terminator.
 for i:=0;i<128;i++ {
  bits:=[]byte{};put:=func(v uint64,n int){for k:=n-1;k>=0;k-- {bits=append(bits,byte((v>>uint(k))&1))}}
  for k:=0;k<i%8;k++ {put(0,1)}
  put(1,1);put(85,7);put(0,3);put(0,1);put(uint64(i%32),5);put(0,1);put(uint64((i+1)%32),5);put(uint64(i*17),32);put(uint64(i%2),1)
  if i%3==0 {put(1,1)} else {put(0,1);put(uint64((i+2)%32),5)}
  put(uint64(i*23),32)
  for j:=0;j<i%15;j++ {put(1,1);put(3,7);put(0,3)};put(0,1)
  data:=make([]byte,(len(bits)+7)/8);for j,b:=range bits {data[j/8]|=b<<uint(7-j%8)}
  add(data,i%8,i%2==0)
  for n:=0;n<len(data);n++ {add(data[:n],0,i%2==0)}
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-kill-event-chain.json",b,0600);e!=nil{t.Fatal(e)}
 t.Logf("%d oracle cases",len(rows))
}
