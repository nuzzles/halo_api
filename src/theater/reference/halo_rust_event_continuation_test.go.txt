package killsource

import (
 "encoding/hex"
 "encoding/json"
 "os"
 "testing"
 "levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
)

// Compose the pinned native event-list walker with its production-view reader.
// The input contains only the encoded frames and config flags from the existing
// independently generated frames-levelup-v41 fixture, never Rust expectations.
func TestHaloRustEventContinuation(t *testing.T) {
 var inputs []struct { Hex string; Extra bool; Check bool }
 input, err := os.ReadFile("/private/tmp/halo-native-continuation-input.json"); if err != nil { t.Fatal(err) }
 if err = json.Unmarshal(input, &inputs); err != nil { t.Fatal(err) }
 rows := []any{}
 for sample, input := range inputs {
  frame, err := hex.DecodeString(input.Hex); if err != nil { t.Fatal(err) }
  for mode:=0; mode<8; mode++ { for _, gate := range []bool{false,true} {
   bits := []byte{1}
   put := func(v uint64,n int) { for b:=n-1;b>=0;b-- { bits=append(bits,byte(v>>uint(b)&1)) } }
   head := func(code uint64) { put(1,1);put(code,7);put(0,3) }
   switch mode {
   case 0: // empty list: generic frame path owns the following views
   case 1: for n:=0;n<65;n++ { head(3) }
   case 2: head(21);put(uint64(sample%4),2)
   case 3: head(15);if gate { put(0x4321,15) };put(123,13);put(70,10);put(0,70)
   case 4: head(5);put(0,111)
   case 5: head(36) // unsupported: never search for a later terminator
   case 6: put(1,1);put(14,7);put(1,1) // unsupported presence provider
   case 7: head(85);put(0,1);put(1,5);put(0,1);put(2,5);put(50,32);put(0,1);put(1,1);put(0,32)
   }
   put(0,1)
   prefixBits:=len(bits)
   for b:=2;b<len(frame)*8;b++ { put(uint64(frame[b/8]>>uint(7-b%8)&1),1) }
   data:=make([]byte,(len(bits)+7)/8);for b,v:=range bits { data[b/8]|=v<<uint(7-b%8) }
   for _,cut:=range []int{0,1,prefixBits/8,len(data)-1,len(data)} {
    payload:=data[:cut];r:=nouveauCurseurEv(payload,1);count:=0;terminated:=false
    for count<len(payload) { fin,ok:=evStep(r,gate);if !ok { break };if fin { terminated=true;break };count++ }
    row:=map[string]any{"sample":sample,"mode":mode,"gate":gate,"extra":input.Extra,"check":input.Check,"hex":hex.EncodeToString(payload),"terminated":terminated,"event_end":r.pos(),"count":count,"continued":terminated&&count>0}
    if terminated&&count>0 {
     reg:=&grammar.Registry{Archetypes:make([]grammar.Archetype,4)}
     reg.Archetypes[3]=grammar.Archetype{Index:3,Components:[]string{"high-frequency"},Levels:[]uint32{1}}
     cfg:=grammar.DefaultFrameConfig();cfg.IDLowBits=5;cfg.HasExtraFields=input.Extra
     cfg.Profil.Grammaire.ControleDeCorruption=input.Check;cfg.Profil.Grammaire.DeserEtatParArchetype=true;cfg.Profil.Grammaire.GenerationStricte=true
     world:=grammar.NewWorld(reg)
     records,views,end:=grammar.DecodeFrameViewsCurseur(payload,world,cfg,3,r.pos())
     if records==nil { records=[]grammar.FrameRecord{} }
     row["records"]=records;row["views"]=views;row["end"]=end
     // A second pass sees the same native grammar bindings, not a fresh world.
     records,views,end=grammar.DecodeFrameViewsCurseur(payload,world,cfg,3,r.pos())
     if records==nil { records=[]grammar.FrameRecord{} }
     row["second"]=map[string]any{"records":records,"views":views,"end":end}
    }
    rows=append(rows,row)
   }
  } }
 }
 encoded,err:=json.Marshal(rows);if err!=nil { t.Fatal(err) }
 if err=os.WriteFile("/private/tmp/halo-native-continuation.json",encoded,0600);err!=nil { t.Fatal(err) }
 t.Logf("%d continuation cases",len(rows))
}
