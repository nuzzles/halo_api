package replay
import("testing";"os";"encoding/json";"math/rand";"sort";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustReplayVehicleShots(t *testing.T){
 rng:=rand.New(rand.NewSource(42008)); rows:=[]any{}
 for run:=0;run<1024;run++{
  vehicles:=[]VehicleTrack{}; owner:=map[uint32]int{}; published:=[]Track{}; slots:=[]uint32{}
  for slot:=0;slot<8;slot++ {if run%13!=0{owner[uint32(slot)]=slot%4};if rng.Intn(3)!=0{published=append(published,Track{Slot:uint32(slot)});slots=append(slots,uint32(slot))}}
  for i:=0;i<run%7;i++ {
   v:=VehicleTrack{Slot:uint32(i%3),Gen:uint32(i%4),End:VehicleEndUnknown}
   if rng.Intn(3)!=0 {v.Spawn=&VehicleSpawn{X:float32(i)/7,Y:float32(i)/-9}}
   for j:=0;j<rng.Intn(12);j++ {v.Samples=append(v.Samples,VehicleSample{T:rng.Intn(15)-2,X:float32(rng.Intn(10000))/317,Y:float32(rng.Intn(10000))/-313})}
   if run%9!=0 {sort.SliceStable(v.Samples,func(a,b int)bool{return v.Samples[a].T<v.Samples[b].T})}
   for j:=0;j<rng.Intn(7);j++ {r:=VehicleRide{T0:rng.Intn(8)-2,T1:rng.Intn(16),Slot:uint32(rng.Intn(8)),Src:"event"};if rng.Intn(3)!=0{s:=rng.Intn(4)-1;r.Seat=&s};v.Rides=append(v.Rides,r)}
   vehicles=append(vehicles,v)
  }
  orphans:=[]orphanShot{}; exported:=[]any{}; cov:=LayerCoverage{Available:1,Attached:1}; initial:=[]Shot{{T:3,Slot:0,X:12,Y:14}}
  for i:=0;i<run%60;i++ {
   e:=grammar.FireEvent{TimestampUS:uint64(rng.Intn(18))*100000,FilmIndex:rng.Intn(6),WeaponID:rng.Uint64(),HasAim:rng.Intn(3)!=0,Aim:[3]float32{float32(rng.Intn(200)-100)/100,float32(rng.Intn(200)-100)/100,0}}
   if i%7==0{e.WeaponID=0};if i%11==0{e.WeaponID=0x1234567842C9679F}
   reason:=reasonNoSlot;rs:="NoSlot";if i%3==0{reason=reasonOutOfWindow;rs="OutOfWindow";cov.OutOfWindow++}else{cov.NoSlot++};cov.Available++
   orphans=append(orphans,orphanShot{ev:e,reason:reason});exported=append(exported,map[string]any{"event":e,"reason":rs})
  }
  origin:=uint64(run%3)*200000;step:=uint64(100000);frames:=run%17
  vc:=&VehicleCoverage{};if run%5==0{vc=nil}
  doc:=ReplayDocument{Vehicles:vehicles,Tracks:published,Shots:append([]Shot{},initial...),Coverage:&Coverage{Shots:cov,Vehicles:vc,Verdict:map[string]string{}}}
  attachVehicleShots(&doc,orphans,IdentityRegistry{own:OwnerReport{Owner:owner}},replayClock{origin:origin,step:step,frames:frames})
  var verdict any;if s,ok:=doc.Coverage.Verdict["shots"];ok{verdict=s}
  probes:=[]any{};for i,v:=range vehicles{for f:=-3;f<20;f++{x,y,ok:=vehiclePosAt(v,f);var p any;if ok{p=[2]float32{x,y}};probes=append(probes,map[string]any{"track":i,"frame":f,"position":p})}}
  rows=append(rows,map[string]any{"vehicles":vehicles,"owners":owner,"published":slots,"origin":origin,"step":step,"frames":frames,"input":map[string]any{"shots":initial,"orphans":exported,"coverage":cov},"output":map[string]any{"shots":doc.Shots,"orphans":exported,"coverage":doc.Coverage.Shots},"vehicleCoverage":vc,"verdict":verdict,"probes":probes})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-vehicle-shots.json",raw,0600);e!=nil{t.Fatal(e)}
}
