package grammar

import (
 "encoding/json"
 "os"
 "testing"
)

func TestHaloRustUnknownBuild(t *testing.T) {
 builds := []string{"", "HI_1_13_0", "../HI Build\n", "İÉΣß中🙂", "\x00\x7f", "ＡＢＣ", "e\u0301"}
 // Exercise each ASCII code and mixed Unicode scalar strings independently.
 for r := rune(0); r < 128; r++ { builds = append(builds, "A"+string(r)+"Z") }
 state := uint32(41)
 for i:=0; i<1024; i++ {
  rs:=[]rune{'H','I','_'}
  for j:=0; j<i%19+1; j++ {
   state=state*1664525+1013904223
   r:=rune(state%0x110000)
   if r>=0xd800 && r<=0xdfff {r='X'}
   rs=append(rs,r)
  }
  builds=append(builds,string(rs))
 }
 rows:=[]any{}
 for _,build:=range builds {
  pairs:=[]any{}
  for _,p:=range UnknownBuildExpvarPairs(build) { pairs=append(pairs,[]any{p.Name,p.Value}) }
  rows=append(rows,map[string]any{"build":build,"pairs":pairs})
 }
 for _,format:=range []int{0,27,28,-1,4294967295,9223372036854775807,-9223372036854775808} {
  pairs:=[]any{}
  for _,p:=range UnknownFormatExpvarPairs(format) { pairs=append(pairs,[]any{p.Name,p.Value}) }
  rows=append(rows,map[string]any{"format":format,"pairs":pairs})
 }
 data,err:=json.Marshal(rows); if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-unknown-build.json",data,0644); err!=nil {t.Fatal(err)}
}
