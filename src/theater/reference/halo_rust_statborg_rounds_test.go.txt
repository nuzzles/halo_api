func TestHaloRustStatborgRounds(t *testing.T){
 rng:=rand.New(rand.NewSource(7641));rows:=[]any{}

  recs:=[]types.StatRecord{}
  for round:=0;round<run%10;round++{n:=rng.Intn(65);if run%7==0&&round==1{n=0};for i:=0;i<n;i++{slot:=6+2*rng.Intn(10);ts:=round*20000+rng.Intn(18000);if run%9==0{ts=rng.Intn(80000)};if run%13==0&&slot==10{ts-=30000};v:=types.StatValue{A:int64(rng.Intn(255)-2),B:int64(rng.Intn(255)-2)};if run%5==0{v.A=int64(i);v.B=0};comps:=map[int]types.StatValue{};if i%3!=0{comps[0]=v};recs=append(recs,types.StatRecord{Slot:slot,Round:round,TimeMS:ts,Comps:comps})}}

  rows=append(rows,map[string]any{"records":recs,"decision":d,"bounds":map[string]any{"by_round":spans,"kept":b.KeptSegments()},"starts":b.Starts(),"round_starts":RoundStartsMS(recs),"posed":b.Posed(),"excluded":excluded,"outliers":b.Outliers(recs),"real_set":d.RealSet(),"outliers_nominal_max":OutliersNominalMax})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-statborg-rounds.json",raw,0600);e!=nil{t.Fatal(e)}
}
