package replay
import("testing";"os";"encoding/json";"math/rand";"sort";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustFactsScope(t *testing.T){
 rng:=rand.New(rand.NewSource(41937));rows:=[]any{}
 for run:=0;run<1024;run++{
  ev:=[]grammar.ZoomEvent{};events:=[]any{};ls:=[]lifeSpan{};lives:=[]any{};queries:=[]any{};hold:=[]uint64{0,1,3500000,10000000,^uint64(0)}[run%5]
  for i:=0;i<run%50;i++{ev=append(ev,grammar.ZoomEvent{Slot:uint32(rng.Intn(4)),TimestampUS:uint64(rng.Intn(40))*500000,Level:[]int{-9223372036854775808,-1,0,1,2,255,256,4294967296,9223372036854775807}[rng.Intn(9)]})};if run%3!=0{sort.SliceStable(ev,func(i,j int)bool{return ev[i].TimestampUS<ev[j].TimestampUS})}
  for _,e:=range ev{events=append(events,map[string]any{"slot":e.Slot,"time":e.TimestampUS,"level":e.Level})}

  rows=append(rows,map[string]any{"events":events,"lives":lives,"hold":hold,"queries":queries})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-facts-scope.json",raw,0600);e!=nil{t.Fatal(e)}
}
