package objectives
import("testing";"encoding/json";"os";"fmt")
func TestHaloRustRoundIdentityQueries(t *testing.T){
 rows:=[]any{}
 for run:=0;run<256;run++ {
  by:=map[int]map[int]string{};origins:=map[int]map[int]string{};flat:=map[int]string{}
  for r:=-2;r<run%6-2;r++ {by[r]=map[int]string{};origins[r]=map[int]string{};for s:=0;s<run%5;s++ {x:=fmt.Sprint(s+1);if (run+s)%3==0{x=""};by[r][s]=x;flat[s]=x;if (r+s)%2==0{origins[r][s]="instants_de_mort"}}}
  origins[20]=map[int]string{9:"independent_origin"}
  identity:=RoundIdentity{byRound:by,origins:origins};f:=FlatRoundIdentity(flat)
  queries:=[]any{};for r:=-3;r<=20;r++ {for s:=-1;s<=9;s++ {queries=append(queries,map[string]any{"round":r,"slot":s,"value":identity.AtRound(r,s),"origin":identity.Origin(r,s),"flat_value":f.AtRound(r,s),"flat_origin":f.Origin(r,s)})}}
  rows=append(rows,map[string]any{"by_round":by,"origins":origins,"flat":flat,"queries":queries,"resolved":identity.Resolved(),"rounds":identity.Rounds(),"named":identity.NamedCount(),"flat_resolved":f.Resolved(),"flat_rounds":f.Rounds(),"flat_named":f.NamedCount()})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-round-identity-queries.json",raw,0600);e!=nil{t.Fatal(e)}
}
