package replay
import("testing";"os";"encoding/json";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustReplayClock(t *testing.T){
 rng:=rand.New(rand.NewSource(41931));rows:=[]any{}
 sample:=func(p grammar.BipedPosition)any{return map[string]any{"position":map[string]any{"Slot":p.Slot,"TimestampUS":p.TimestampUS,"X":p.X,"Y":p.Y,"Z":p.Z,"HasWorld":p.HasWorld},"aim":[]uint32{p.YawRaw,p.PitchRaw},"health":p.Body.Health,"shield":p.Shield.Shield}}
 for run:=0;run<1024;run++{
  pos:=[]grammar.BipedPosition{};samples:=[]any{}
  for i:=0;i<run%37;i++{stamp:=uint64(rng.Intn(30))*1000000;if run%7==0{stamp+=1<<63};p:=grammar.BipedPosition{Slot:uint32(i),TimestampUS:stamp,HasWorld:i%3!=0,X:float32(i)};p.YawRaw=uint32(i);p.PitchRaw=uint32(run);p.Body.Health=float32(i)/10;p.Shield.Shield=float32(run)/1000;pos=append(pos,p);samples=append(samples,sample(p))}
  interval:=[]int{-100,0,1,33,100,1000,9223372036854775807}[run%7];a:=assemblage{pos:pos,opt:Options{FrameIntervalMS:interval}};var timeline any
  if a.ouvrir("halo-infinite"){sorted:=[]any{};for _,p:=range a.sorted{sorted=append(sorted,sample(p))};timeline=map[string]any{"samples":sorted,"clock":map[string]any{"origin_us":a.origin,"step_us":a.step,"frame_count":a.doc.FrameCount},"frame_interval_ms":a.interval,"duration_ms":a.doc.DurationMS}}
  first:=uint64(rng.Intn(10000000));clock:=uint64(rng.Intn(10000000));if run%3==0{first+=clock};if run%11==0{clock=0};if run%17==0{first+=1<<63};offset:=int64(clock/1000)+[]int64{-1001,-1000,-999,0,999,1000,1001}[run%7];matched:=run%8
  origin:=resolveOriginMs(first,clock,offset,matched)
  rows=append(rows,map[string]any{"samples":samples,"interval":interval,"timeline":timeline,"first":first,"film_clock":clock,"offset":offset,"matched":matched,"origin":origin})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-clock.json",raw,0600);e!=nil{t.Fatal(e)}
}
