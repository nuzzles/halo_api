package replay
import("encoding/json";"math";"os";"testing")
func TestHaloRustMapBackgroundCalibration(t *testing.T){
 values:=[]float64{math.Inf(-1),-1e30,-2,-0.25,math.Copysign(0,-1),0,math.SmallestNonzeroFloat64,0.25,1,1.5,2,100,1e30,math.Inf(1),math.NaN()}
 dims:=[]int{-1<<63,-1,0,1,2,99,1<<32,1<<63-1}
 rows:=[]any{}
 put:=func(c MapBackgroundCalibration,x,y float64){
  px,py,ok:=c.MondeVersPixel(x,y);raw,e:=json.Marshal(c);var wire any;if e==nil{wire=json.RawMessage(raw)}
  rows=append(rows,map[string]any{"bits":[]uint64{math.Float64bits(c.MetersPerPixel),math.Float64bits(c.OriginX),math.Float64bits(c.OriginY),math.Float64bits(x),math.Float64bits(y)},"width":c.WidthPx,"height":c.HeightPx,"convention":c.Convention,"px":px,"py":py,"ok":ok,"marshal_error":e!=nil,"calibration":wire})
 }
 for i,mpp:=range values{for j,x:=range values{for k,y:=range values{
  c:=MapBackgroundCalibration{MetersPerPixel:mpp,OriginX:values[(j+3)%len(values)],OriginY:values[(k+5)%len(values)],WidthPx:dims[(i+j)%len(dims)],HeightPx:dims[(i+k)%len(dims)],Convention:"centers"}
  put(c,x,y)
 }}}
 for _,x:=range []float64{-1.01,-1,-0.99,-0.5,-0.01,0,0.5,1,1.99,2}{for _,y:=range []float64{-1.01,-1,-0.99,-0.5,0,0.5,1,1.99,2}{put(MapBackgroundCalibration{MetersPerPixel:1,WidthPx:2,HeightPx:2},x,y)}}
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-map-background-calibration.json",raw,0600);e!=nil{t.Fatal(e)}
}
