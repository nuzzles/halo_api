func haloRustBodyValue(v reflect.Value)any{
 switch v.Kind(){
 case reflect.Pointer:if v.IsNil(){return nil};return haloRustBodyValue(v.Elem())
 case reflect.Slice:if v.IsNil(){return nil};fallthrough
 case reflect.Array:out:=[]any{};for i:=0;i<v.Len();i++{out=append(out,haloRustBodyValue(v.Index(i)))};return out
 case reflect.Struct:out:=map[string]any{};for i:=0;i<v.NumField();i++{out[v.Type().Field(i).Name]=haloRustBodyValue(v.Field(i))};return out
 case reflect.Int,reflect.Int64,reflect.Int32,reflect.Int8:return v.Int()
 case reflect.Uint,reflect.Uint64,reflect.Uint32,reflect.Uint16,reflect.Uint8:return v.Uint()
 case reflect.Bool:return v.Bool()
 case reflect.Float32:return math.Float32bits(v.Interface().(float32))
 case reflect.Float64:return math.Float64bits(v.Float())
 };panic(v.Kind())
}
func haloRustBodyFill(v reflect.Value,rng *rand.Rand,n int){switch v.Kind(){
 case reflect.Pointer:if n%3!=0{v.Set(reflect.New(v.Type().Elem()));haloRustBodyFill(v.Elem(),rng,n)}
 case reflect.Slice:if n%4!=0{v.Set(reflect.MakeSlice(v.Type(),n%3,n%3));for i:=0;i<v.Len();i++{haloRustBodyFill(v.Index(i),rng,n+i+1)}}
 case reflect.Array:for i:=0;i<v.Len();i++{haloRustBodyFill(v.Index(i),rng,n+i)}
 case reflect.Struct:for i:=0;i<v.NumField();i++{haloRustBodyFill(v.Field(i),rng,n+i)}
 case reflect.Int,reflect.Int64,reflect.Int32,reflect.Int8:v.SetInt(int64(rng.Uint64()))
 case reflect.Uint,reflect.Uint64,reflect.Uint32,reflect.Uint16,reflect.Uint8:v.SetUint(rng.Uint64())
 case reflect.Bool:v.SetBool(rng.Intn(2)==0)
 case reflect.Float32:v.Set(reflect.ValueOf(math.Float32frombits(rng.Uint32())))
 case reflect.Float64:v.SetFloat(math.Float64frombits(rng.Uint64()))
 default:panic(v.Kind())}}
