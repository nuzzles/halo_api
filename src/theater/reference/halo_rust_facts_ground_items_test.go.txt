  scan:=WorldObjectScan{Scanned:true,Keyframes:grammar.WorldObjectKeyframes{TimesUS:[]uint64{0,400000,800000,1200000,1600000,2000000},SeenUS:map[types.EquipmentLifeKey][]uint64{}}};positions:=[]grammar.BipedPosition{};seen:=[]any{}
  for i:=0;i<64;i++{c:=types.EquipmentCreation{Slot:uint32(rng.Intn(8)),Gen:generations[rng.Intn(2)],TimestampUS:uint64(rng.Intn(20))*100000,X:float32(rng.Intn(5))/4,Y:float32(rng.Intn(5))/4,Z:float32(rng.Intn(5))/4,HasAmmo:rng.Intn(2)==0,Ammo:types.GroundWeaponAmmo{Mag:uint32(i),Res:uint32(i*2)}};c.MPPPresent[1]=i%7!=0;c.MPPVal[1]=uint64(rng.Intn(5));if i%11==10 {c.MPPVal[1]|=uint64(1)<<63};if i%17==16 {c.TimestampUS=^uint64(0)-100000};scan.Creations=append(scan.Creations,c);if i%3==0{scan.Tracks=append(scan.Tracks,types.ProjectileTrack{Slot:c.Slot,Gen:c.Gen,Pts:[]types.ProjectileSample{{TimestampUS:c.TimestampUS,X:c.X,Y:c.Y,Z:c.Z},{TimestampUS:c.TimestampUS+100000,X:c.X+0.5,Y:c.Y,Z:c.Z}}})}}
  for slot:=uint32(0);slot<8;slot++{for gen:=uint32(0);gen<2;gen++{ts:=[]uint64{};for _,t:=range scan.Keyframes.TimesUS{if rng.Intn(2)==0{ts=append(ts,t)}};scan.Keyframes.SeenUS[types.EquipmentLifeKey{Slot:slot,Gen:generations[gen]}]=ts;seen=append(seen,map[string]any{"slot":slot,"generation":generations[gen],"times_us":ts})}}
  for i:=0;i<80;i++{positions=append(positions,grammar.BipedPosition{Slot:uint32(rng.Intn(4)),TimestampUS:uint64(rng.Intn(22))*100000,HasWorld:i%5!=0,X:float32(rng.Intn(7))/4,Y:float32(rng.Intn(7))/4,Z:float32(rng.Intn(7))/4})};sort.Slice(positions,func(i,j int)bool{return positions[i].TimestampUS<positions[j].TimestampUS})
  rule:=padRule{Kind:"weapon",Objectives:map[uint32]Label{2:{En:"objective"}},Family:func(id uint32)(string,bool){if id==1||id==2||id==3{return "family",true};return "",false}}

 }
 d,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-facts-ground-items.json",d,0600);e!=nil{t.Fatal(e)}
}
