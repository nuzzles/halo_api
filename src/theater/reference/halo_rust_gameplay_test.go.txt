// Preserve struct-keyed diagnostic maps as sorted key/value arrays. JSON cannot
// otherwise represent these reference fields; no field is discarded for export.
func haloJSON(v reflect.Value) any {
 if !v.IsValid(){return nil}
 switch v.Kind(){
 case reflect.Interface,reflect.Pointer:if v.IsNil(){return nil};return haloJSON(v.Elem())
 case reflect.Bool:return v.Bool()
 case reflect.Int,reflect.Int8,reflect.Int16,reflect.Int32,reflect.Int64:return v.Int()
 case reflect.Uint,reflect.Uint8,reflect.Uint16,reflect.Uint32,reflect.Uint64:return v.Uint()
 case reflect.Float32,reflect.Float64:return v.Float()
 case reflect.String:return v.String()
 case reflect.Slice,reflect.Array:
  if v.Kind()==reflect.Slice&&v.IsNil(){return nil};out:=make([]any,v.Len());for i:=range out{out[i]=haloJSON(v.Index(i))};return out
 case reflect.Struct:
  out:=map[string]any{};typ:=v.Type();for i:=0;i<v.NumField();i++ {f:=typ.Field(i);if f.PkgPath!=""&&!f.Anonymous{continue};name:=strings.Split(f.Tag.Get("json"),",")[0];if name=="-"{continue};value:=haloJSON(v.Field(i));if f.Anonymous&&name=="" {if embedded,ok:=value.(map[string]any);ok{for k,x:=range embedded{out[k]=x};continue}};if name==""{name=f.Name};out[name]=value};return out
 case reflect.Map:
  if v.IsNil(){return nil};keys:=v.MapKeys();sort.Slice(keys,func(i,j int)bool{a,_:=json.Marshal(haloJSON(keys[i]));b,_:=json.Marshal(haloJSON(keys[j]));return string(a)<string(b)})
  if v.Type().Key().Kind()==reflect.Struct {out:=[]any{};for _,k:=range keys{out=append(out,map[string]any{"key":haloJSON(k),"value":haloJSON(v.MapIndex(k))})};return out}
  out:=map[string]any{};for _,k:=range keys{out[fmt.Sprint(haloJSON(k))]=haloJSON(v.MapIndex(k))};return out
 default:panic("unsupported oracle field "+v.Type().String())
 }
}
