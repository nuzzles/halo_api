func haloFactsVehiclePosition(p grammar.BipedPosition) any {
	var v, d, r any
	if p.HasVel {
		v = [2]uint32{p.VelRaw, p.VelScale}
	}
	if p.HasAim {
		d = p.AimRaw
	}
	if p.HasRoll {
		r = p.RollRaw
	}
	return map[string]any{"position": p, "companions": map[string]any{"velocity": v, "forward": d, "chassis": map[string]any{"direction": d, "roll": r, "mode": p.FwdMode, "delta": false, "direction_default": p.AimDefault}, "aim": nil, "aim_flag_0": false, "aim_flag_1": false, "aim_b": nil, "aim_flag_2": false, "body": nil, "shield": nil}}
}
func TestHaloRustFactsVehicleSamples(t *testing.T) {
	previous := slog.Default(); defer slog.SetDefault(previous)
 rng := rand.New(rand.NewSource(42014))
	rows := []any{}
	keys := []uint32{0, 0xFFFFFFFF}
	for k := range vehicleFamilyByChassis {
		keys = append(keys, k)
	}
	sort.Slice(keys, func(i, j int) bool { return keys[i] < keys[j] })
	for run := 0; run < 1024; run++ {
		life := vehicleLife{key: types.EquipmentLifeKey{Slot: 2, Gen: uint32(run % 4)}, firstUS: 400000, lastUS: 1200000, loUS: uint64(run%3) * 100000, hiUS: 1800000, goneByUS: 2000000}
		if run%5 == 0 {
			life.goneByUS = 0
			life.hiUS = ^uint64(0)
		}
		if run%7 == 0 {
			life.deathUS = 800000
		}
		spawn := types.EquipmentCreation{TimestampUS: uint64(run%4) * 100000, X: float32(run) / 317, Y: float32(run) / -313, Mask: []int{}}
		spawn.MPPPresent[grammar.MPPWord32] = run%6 != 0
		spawn.MPPVal[grammar.MPPWord32] = uint64(keys[run%len(keys)])
		all := []grammar.BipedPosition{}
		inputs := []any{}
		for i := 0; i < run%50; i++ {
			p := grammar.BipedPosition{Slot: uint32(rng.Intn(3)), TimestampUS: uint64(rng.Intn(20)) * 100000, X: float32(i) / 317, Y: float32(i) / -313, Z: float32(i) / 73, HasWorld: rng.Intn(4) != 0}
			p.HasVel = rng.Intn(3) != 0
			p.VelRaw = rng.Uint32()
			p.VelScale = rng.Uint32()
			p.HasAim = rng.Intn(3) != 0
			p.FwdMode = []uint8{0,1,2,3,255}[rng.Intn(5)]
			p.AimRaw = rng.Uint32()
			p.HasRoll = rng.Intn(4) != 0
			p.RollRaw = rng.Uint32()
			p.AimDefault = rng.Intn(4) == 0
			all = append(all, p)
			inputs = append(inputs, haloFactsVehiclePosition(p))
		}
		positions := vehiclePositionsBySlot(all)[2]
		grouped := []any{}
		headings := []any{}
		for _, p := range positions {
			grouped = append(grouped, haloFactsVehiclePosition(p))
			h, ok := vehicleHeadingOf(p)
			var v any
			if ok {
				v = h
			}
			headings = append(headings, v)
		}
		rides := []VehicleRide{}
		for i := 0; i < run%8; i++ {
			r := VehicleRide{T0: rng.Intn(20) - 3, T1: rng.Intn(30), Slot: uint32(i), Src: "event"}
			seat := i % 3
			r.Seat = &seat
			for j := 0; j < 4; j++ {
				r.Aim = append(r.Aim, VehicleAim{T: j * 5, H: float32(j)})
			}
			rides = append(rides, r)
		}
		origin := uint64(run%3) * 100000
		step := uint64([]int{1, 100000, 1000000}[run%3])
		frames := []int{-9223372036854775808,-1,0,1,30,4294967296,9223372036854775807}[run%7]
		clock := replayClock{origin: origin, step: step, frames: frames}
		samples, last := vehicleSamplesOf(positions, life, clock)
		if samples == nil {
			samples = []VehicleSample{}
		}
		track, ok := vehicleTrackOf(life, spawn, positions, rides, clock)
		var output any
		if ok {
			output = track
		}
		creations := []types.EquipmentCreation{}
		for j := 0; j < 12; j++ {
			s := spawn
			s.Slot = uint32(j % 3)
			s.Gen = uint32(j % 2)
			s.TimestampUS = uint64(rng.Intn(4)) * 100000
			s.X = float32(j)
			creations = append(creations, s)
		}
		byLife := vehicleSpawnsByLife(creations)
		chosen := []types.EquipmentCreation{}
		for _, s := range byLife {
			chosen = append(chosen, s)
		}
		sort.Slice(chosen, func(a, b int) bool {
			if chosen[a].Slot != chosen[b].Slot {
				return chosen[a].Slot < chosen[b].Slot
			}
			return chosen[a].Gen < chosen[b].Gen
		})
		coverage := VehicleCoverage{Scanned: true, UnknownChassis: map[string]int{}}
		coverageTracks := []VehicleTrack{}
		if ok {
			coverageTracks = append(coverageTracks, track)
		}
		tallyVehicleCoverage(coverageTracks, &coverage, nil)
		tallyVehicleEnds(coverageTracks, &coverage)
		cycleRng := rand.New(rand.NewSource(int64(run) + 717778))
		cycleTracks := []VehicleTrack{}
		for j := 0; j < run%40; j++ {
			born := j * 100
			end := born + 50 + cycleRng.Intn(10)
			tr := VehicleTrack{T0: born, End: "destroyed", TEnd: &end, Family: []string{"", "warthog", "ghost"}[j%3]}
			if run%3 == 0 {
				tr.T0 = cycleRng.Intn(4000)
				if j%3 == 0 {
					tr.End = "unknown"
				}
			}
			if j%13 != 0 {
				tr.Spawn = &VehicleSpawn{X: float32(j%3)*6 + float32(cycleRng.Intn(4))/4, Y: float32(j%2) / 2}
			}
			if run%5 == 0 && j%4 == 0 {
				tr.TEnd = nil
			}
			cycleTracks = append(cycleTracks, tr)
		}
		cycleStep := []uint64{0, 1, 100000, 1000000}[run%4]
		cycleCoverage := VehicleCoverage{}
		cycles := buildVehicleCycles(cycleTracks, cycleStep, &cycleCoverage)
		if cycles == nil {
			cycles = []VehicleCycle{}
		}
		var logBuf bytes.Buffer
        slog.SetDefault(slog.New(slog.NewJSONHandler(&logBuf, nil)))
        logVehicleHeadingSource(all)
        headingLog := map[string]any{}
        if err := json.Unmarshal(logBuf.Bytes(), &headingLog); err != nil { t.Fatal(err) }
        delete(headingLog, "time")
        rows = append(rows, map[string]any{"heading_log": headingLog, "drawable": vehicleDrawableLives([]vehicleLife{life}, byLife, vehiclePositionsBySlot(all))[life.key], "cycle_tracks": cycleTracks, "cycle_step": cycleStep, "cycles": cycles, "cycle_coverage": cycleCoverage, "coverage": coverage, "spawnInputs": creations, "spawns": chosen, "life": map[string]any{"slot": life.key.Slot, "generation": life.key.Gen, "first_us": life.firstUS, "last_us": life.lastUS, "gone_by_us": life.goneByUS, "lo_us": life.loUS, "hi_us": life.hiUS, "census": 0, "death_us": life.deathUS, "death_tail_desync": false}, "spawn": spawn, "inputs": inputs, "grouped": grouped, "headings": headings, "rides": rides, "origin": origin, "step": step, "frames": frames, "samples": samples, "last": last, "track": output})
	}
	families := map[uint32]any{}
	for _, k := range keys {
		f := vehicleFamilyOf(k)
		families[k] = map[string]any{"family": f, "rideable": vehicleFamilyIsRideable(f), "formatted": formatChassisID(k)}
	}
	data, e := json.Marshal(map[string]any{"cases": rows, "families": families})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-facts-vehicle-samples.json", data, 0600); e != nil {
		t.Fatal(e)
	}
}
