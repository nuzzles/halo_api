var haloClosers=[]string{"None","Bound","Handoff","Return","Home","CarrierKill","Object"}
func haloFlagRaw(rs []flagCarryRaw) []any {out:=[]any{};for _,r:=range rs{out=append(out,map[string]any{"xuid":r.xuid,"t0":r.t0,"t1":r.t1,"steal":r.steal,"x0":r.x0,"y0":r.y0,"x1":r.x1,"y1":r.y1,"captured":r.captured,"closed":r.closed,"homed":r.homed,"closed_by":haloClosers[r.closedBy],"flag_index":r.flagIndex,"confirmed":r.confirmed,"observable":r.observable})};return out}
func TestHaloRustReplayFlagCarries(t *testing.T){
 rng:=rand.New(rand.NewSource(41250));rows:=[]any{}
 for run:=0;run<2048;run++{
 identity:=map[int]string{};for slot:=0;slot<8;slot++{if rng.Intn(5)!=0{identity[slot]=[]string{"100","101","102","103"}[rng.Intn(4)]}}

 for j:=0;j<run%17;j++{bursts=append(bursts,j*1000)}
 for j:=0;j<80;j++{events=append(events,objectives.NamedEvent{TimeMS:rng.Intn(80)*250-1000,Slot:rng.Intn(8),Stat:[]string{"flag_grabs","flag_steals","flag_captures","flag_carriers_killed","flag_returns","other"}[rng.Intn(6)],Comp:0,Side:"A"})}
 for j:=0;j<run%20;j++{deaths=append(deaths,Death{XUID:uint64(100+rng.Intn(4)),TimeMS:int64(rng.Intn(80)*250-1000)})}
 spawns:=[]FlagSpawn{};teams:=map[string]int{};for j:=0;j<run%5;j++{spawns=append(spawns,FlagSpawn{Team:j%3-1,Neutral:j%2==0,X:float32(j),Y:0})};for _,x:=range []string{"100","101","102","103"}{if rng.Intn(4)!=0{teams[x]=rng.Intn(4)-1}}
 c:=matchClock{origin:uint64(run%7)*1000000,step:[]uint64{0,16667,100000,1000000}[run%4],frames:[]int{-1,0,10,1000}[run%4],deathOffsetMS:int64(run%5-2)*1000};ctx:=flagCarryCtx{matchClock:c,deaths:deaths}

 lives:=[]flagFreeLife{};for j:=0;j<20;j++{x:=float32(rng.Intn(5));y:=float32(0);if j%3==0{x+=[]float32{0.03,0.1,0.11}[j%3]};pts:=[]flagFreeSample{{TUS:uint64(j)*1000000,X:x,Y:y}};if j%7==0{pts=[]flagFreeSample{}};lives=append(lives,flagFreeLife{T0US:uint64(j)*1000000,Pts:pts})}
 homeCtx:=ctx;if homeCtx.step==0{homeCtx.step=100000};scan:=FlagCarryScan{Events:events,Identity:ri,Spawns:spawns,TeamOf:teams,Free:lives};choice:=flagChooseSpawns(scan);if choice.Spawns==nil{choice.Spawns=[]FlagSpawn{}};homes:=flagObjectHomecomings(scan,homeCtx);returns:=flagReturns(scan);homeRows:=func(hs []flagHomecoming)[]any{out:=[]any{};for _,h:=range hs{out=append(out,map[string]any{"flag":h.flag,"at":h.at,"x":h.x,"y":h.y})};return out}
 raw:=boundFlagCarries(ops,events,ctx);bound:=haloFlagRaw(raw);unnamed:=closeByHandoff(raw,ops,FlagCarryScan{Spawns:spawns,TeamOf:teams});handoff:=haloFlagRaw(raw);closeByHomecoming(raw,scan,homeCtx);homed:=haloFlagRaw(raw);raw,ambiguous:=closeByCarrierKills(raw,events,ri);killed:=haloFlagRaw(raw)
 closes:=[]any{};for j:=0;j<15;j++{r:=flagCarryRaw{xuid:"100",t0:100,t1:1000,captured:j%2==0,homed:j%3==0,closed:j%4==0,flagIndex:-1};before:=haloFlagRaw([]flagCarryRaw{r})[0];at:=[]int64{0,100,101,500,999,1000,1001}[rng.Intn(7)];by:=flagCloser(rng.Intn(7));flagCloseAt(&r,at,by);closes=append(closes,map[string]any{"before":before,"at":at,"by":haloClosers[by],"after":haloFlagRaw([]flagCarryRaw{r})[0]})}
 rows=append(rows,map[string]any{"lives":lives,"choice":choice,"homes":homeRows(homes),"returns":homeRows(returns),"homed":homed,"identity":identity,"events":events,"deaths":deaths,"bursts":bursts,"signals":signals,"recognized":signals.IsFlagFilm(),"spawns":spawns,"teams":teams,"clock":map[string]any{"origin_us":c.origin,"step_us":c.step,"frames":c.frames,"death_offset_ms":c.deathOffsetMS},"openings":openings,"bound":bound,"handoff":handoff,"unnamed":unnamed,"killed":killed,"ambiguous":ambiguous,"closes":closes})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-flag-carries.json",raw,0600);e!=nil{t.Fatal(e)}
}
