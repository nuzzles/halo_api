func TestHaloRustM4bTails(t *testing.T) {
 rows := []any{}
 seed := uint64(0xa94710e33)
 for _, name := range []string{"object-dead-state-component", "biped-mobility-action-component"} {
  for variant:=0; variant<3; variant++ {
   modes:=1; if name=="biped-mobility-action-component" {modes=3}
   for mode:=0; mode<modes; mode++ { for sample:=0; sample<128; sample++ {
    data:=make([]byte,128)
    for i:=range data {seed ^= seed<<13; seed ^= seed>>7; seed ^= seed<<17; data[i]=byte(seed); if sample<8 {data[i]=0}; if sample>=8 && sample<16 {data[i]=255}}
    start:=sample%8
    // Keep the zero-filled mobility controls active, so map widths discriminate.
    if name=="biped-mobility-action-component" && sample<8 {data[start/8] |= 1<<uint(7-start%8)}
    br:=LecteurSur(data); br.SetBitPos(start)
    ctx:=ContexteParDefaut(); ctx.Obs=NouvelleObservation()
    actions:=[][2]bool{}
    ctx.Obs.MobilityActionHook=func(a,b bool){actions=append(actions,[2]bool{a,b})}
    p:=ctx.Profil
    p.Mouvement.FullPrecision=mode==1; p.Grammaire.PorteeBaseline=mode==2
    p.Mouvement.WorldObject.IndexW=uint(variant+1)
    p.Mouvement.WorldObject.AxisW=[][3]uint{{13,13,14},{17,17,15},{1,2,3}}[variant]
    ctx.Profil=p; br.PoserContexte(ctx)
    ti:=uint32(35); if name=="object-dead-state-component" {ti=[]uint32{35,40,22}[variant]}
    encoding:=map[string]any{"index_bits":p.Mouvement.WorldObject.IndexW,"world_axis_bits":p.Mouvement.WorldObject.AxisW,"handle_bits":p.Mouvement.Traversal.IndexW,"traversal_axis_bits":p.Mouvement.Traversal.AxisW,"default_axis_bits":profile.LargeursAxeParDefautDuBuild(profile.NiveauPositionDObjet),"region_axis_bits":map[string]any{},"delta_axis_bits":[3]uint{p.Mouvement.DeltaAxisWidth,p.Mouvement.DeltaAxisWidth,p.Mouvement.DeltaAxisWidth},"full_precision":p.Mouvement.FullPrecision,"baseline_scope":p.Grammaire.PorteeBaseline,"writer_absolute":p.Grammaire.GrammaireEcrivainI0,"delta_handle_tail":p.Mouvement.DeltaHasHandleTail}
    _,dead,ported:=consumeByName(br,name,ti,1)
    end:=br.BitPos(); next:=br.ReadBits(8)
    rows=append(rows,map[string]any{"name":name,"archetype":ti,"input":hex.EncodeToString(data),"start":start,"end":end,"next":next,"ported":ported,"encoding":encoding,"dead":dead,"actions":actions,"indices":ctx.Obs.IndexAbsolus})
   }}
  }
 }
 f,err:=os.Create(os.Getenv("HALO_RUST_M4B_TAILS_OUT"));if err!=nil{t.Fatal(err)}
 z:=zlib.NewWriter(f);if err=json.NewEncoder(z).Encode(rows);err!=nil{t.Fatal(err)};if err=z.Close();err!=nil{t.Fatal(err)};if err=f.Close();err!=nil{t.Fatal(err)}
 t.Logf("wrote %d native tail cases",len(rows))
}
