package killsource

import (
 "encoding/json"
 "os"
 "testing"
)

// Exercise the actual native method, including opaque origin text and counts
// that must not affect presentation. Only the two published Path values are used.
func TestHaloRustKillProvenance(t *testing.T) {
 rows:=[]any{}
 for _,path:=range []Path{PathWalk,PathScan} {
  for _,origin:=range []Origin{OriginCredit,OriginSelfSource,OriginBot,OriginBotKiller,OriginUnclaimed,"","custom / origin","\tligne\n雪"} {
   for _,count:=range []int{0,1,2147483647} {
    p:=Provenance{Path:path,Origin:origin,Multiplicity:count}
    rows=append(rows,map[string]any{"path":p.Path,"origin":p.Origin,"multiplicity":p.Multiplicity,"display":p.String()})
   }
  }
 }
 raw,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-kill-provenance.json",raw,0600);err!=nil{t.Fatal(err)}
}
