package grammar

import (
 "encoding/binary"
 "encoding/hex"
 "encoding/json"
 "levelup/go-api/internal/games/halo_infinite/film/internal/source"
 "levelup/go-api/internal/games/halo_infinite/film/types"
 "os"
 "reflect"
 "testing"
)

func TestHaloRustEquipmentSpawnSource(t *testing.T) {
 rows:=[]any{}
 orders:=[][]int{{1,2,3},{0,1,2},{1,3,2},{2,1},{1,1,2},{-1,1,2},{1},{0}}
 packet:=func(kind uint16,ts uint64,pay []byte) []byte {p:=make([]byte,16);binary.LittleEndian.PutUint16(p,kind);binary.LittleEndian.PutUint32(p[4:],uint32(len(pay)));binary.LittleEndian.PutUint64(p[8:],ts);return append(p,pay...)}
 for run:=0;run<128;run++ {
  raw:=source.MemoryChunks{};meta:=[]types.ChunkMeta{};inputs:=[]any{}
  for ci,index:=range orders[run%len(orders)] {
   data:=[]byte{}
   for j:=0;j<24;j++ {
    var w bitWriter
    w.bits(uint64(j%2),1);w.bits(1,1);w.bits(103,7)
    for k:=0;k<3;k++ {
     present:=(j>>(k))&1;w.bits(uint64(present),1)
     if present!=0 {w.bits(uint64(ci*100+j*3+k),13);w.bits(uint64((j+k)%4),2)}
    }
    pay:=w.buf
    switch j%8 {case 0:pay=[]byte{0x40};case 1:pay[0]&=0xbf;case 2:pay[0]=0xc4}
    kind:=uint16(0);if j%11==0 {kind=6}
    if j==12 {
     switch (run/8)%4 {
     case 1:kind=7
     case 2:pay=nil
     case 3:p:=packet(kind,3,pay);binary.LittleEndian.PutUint32(p[4:],99999);data=append(data,p...);continue
     }
    }
    data=append(data,packet(kind,uint64(j%3),pay)...)
   }
   data=append(data,1,2,3)
   raw=append(raw,data);meta=append(meta,types.ChunkMeta{Index:index});inputs=append(inputs,map[string]any{"index":index,"hex":hex.EncodeToString(data)})
  }
  film,err:=source.Load(raw,meta);if err!=nil {t.Fatal(err)}
  fc:=NewFilmContext(film)
  expected,stats,scanErr:=ScanEquipmentSpawnEvents(fc)
  type item struct {Event types.EquipmentSpawnEvent `json:"event"`;Start int `json:"start"`;Size int `json:"size"`}
  observed:=[]item{};var projected []types.EquipmentSpawnEvent
  for _,c:=range fc.ChunkNumbers() {
   data,pks,_:=fc.ChunkAt(c)
   for _,pk:=range pks {
    if pk.Type!=0 || pk.Size<1 {continue}
    ev,_,ok:=decodeEquipmentSpawnEvent(pk.Payload(data));if !ok {continue}
    ev.Chunk=c;ev.PacketIndex=pk.Index;ev.TimestampUS=pk.TimestampUS
    observed=append(observed,item{ev,pk.Start,pk.Size});projected=append(projected,ev)
   }
  }
  if !reflect.DeepEqual(expected,projected) {t.Fatal("wrapper differs")}
  rows=append(rows,map[string]any{"inputs":inputs,"stats":stats,"events":observed,"no_chunks":scanErr==ErrNoFilmChunk})
 }
 // Truncation is a native raw-reference panic, not the Lecteur zero-tail policy.
 short:=[]any{}
 for i:=0;i<256;i++ {
  for n:=0;n<=8;n++ {
   pay:=[]byte{0xf3,byte(i),byte(i*7),byte(i*13),byte(i*23),byte(i*37),byte(i*53),byte(i*71)}[:n]
   var ev types.EquipmentSpawnEvent;var present,ok,panicked bool
   func(){defer func(){if recover()!=nil {panicked=true}}();ev,present,ok=decodeEquipmentSpawnEvent(pay)}()
   short=append(short,map[string]any{"hex":hex.EncodeToString(pay),"event":ev,"present":present,"ok":ok,"panic":panicked})
  }
 }
 data,err:=json.Marshal(map[string]any{"rows":rows,"short":short});if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-equipment-spawn-source.json",data,0600);err!=nil {t.Fatal(err)}
}
