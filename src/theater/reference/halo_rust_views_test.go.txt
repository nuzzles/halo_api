package grammar

import (
 "encoding/hex"
 "encoding/json"
 "math/rand"
 "os"
 "testing"
)

func TestHaloRustViewsParity(t *testing.T) {
 rng := rand.New(rand.NewSource(4103))
 rows := []any{}
 for _, control := range []bool{false, true} {
  for sample := 0; sample < 1024; sample++ {
   data := make([]byte, 512)
   if sample > 0 { if _, err := rng.Read(data); err != nil { t.Fatal(err) } }
   start := sample % 8
   br := LecteurSur(data); br.SetBitPos(start)
   var kinds []int
   var complete bool
   if control { out := consumeVueC(br, len(data)*8); kinds, complete = out.Kinds, out.Porte
   } else { out := consumeVueA(br, len(data)*8); kinds, complete = out.Genres, out.Porte }
   if kinds == nil { kinds = []int{} }
   end := br.BitPos()
   prefixes := []any{}
   for cut:=0;cut<=(end+7)/8;cut++ {
    short:=data[:cut];r:=LecteurSur(short);r.SetBitPos(start)
    var ks []int;var ok bool
    if control {out:=consumeVueC(r,len(short)*8);ks,ok=out.Kinds,out.Porte} else {out:=consumeVueA(r,len(short)*8);ks,ok=out.Genres,out.Porte}
    if ks==nil {ks=[]int{}}
    entry:=map[string]any{"bytes":cut,"end_bit":r.BitPos(),"complete":ok,"kinds":ks}
    if control {
     // Preserve bit alignment while inserting empty message/entity views.
     framed:=append([]byte{0},short...)
     if len(framed)>1 {framed[1] &= byte(0xff >> uint(start))}
     cfg:=DefaultFrameConfig();cfg.PacketPreambleBits=start+5
     reg:=&Registry{Archetypes:[]Archetype{{Index:0}}}
     w:=NewWorld(reg);w.PoserVueCourante(2);w.BindFull(50,0);w.SetPos(50,[3]float32{1,2,3})
     records,views,endpoint:=DecodeFrameViewsCurseur(framed,w,cfg,3,cfg.PacketPreambleBits)
     if len(records)!=0 {t.Fatal("empty entity prefix emitted records")}
     entry["production"]=map[string]any{"hex":hex.EncodeToString(framed),"preamble":cfg.PacketPreambleBits,"end":endpoint,"views":views,"world_view":w.vueCourante,"slots":w.slots}
    }
    if control && cut>0 {
     var bw bitWriter;bw.bits(0,5) // configuration, message End, entity End
     k:=0;for (5+3*k)%8!=start {k++}
     for j:=0;j<k;j++ {bw.bits(7,3)} // control no-op records retain alignment
     for p:=start;p<cut*8;p++ {bw.bits(uint64((short[p/8]>>uint(7-p%8))&1),1)}
     reg:=&Registry{Archetypes:[]Archetype{{Index:0}}};cfg:=DefaultFrameConfig()
     records,views,endpoint:=DecodeFrameViewsCurseur(bw.buf,NewWorld(reg),cfg,3,2)
     if len(records)!=0 {t.Fatal("film control frame emitted entity records")}
     cr:=LecteurSur(bw.buf);cr.SetBitPos(5);cv:=consumeVueC(cr,len(bw.buf)*8)
     kinds:=cv.Kinds;if kinds==nil{kinds=[]int{}}
     entry["film"]=map[string]any{"hex":hex.EncodeToString(bw.buf),"end":endpoint,"views":views,"kinds":kinds,"complete":cv.Porte}
    }
    prefixes=append(prefixes,entry)
   }
   rows = append(rows, map[string]any{"control":control,"start_bit":start,"end_bit":end,"complete":complete,"kinds":kinds,"hex":hex.EncodeToString(data[:(end+7)/8]),"prefixes":prefixes})
  }
 }
 data, err := json.Marshal(rows); if err != nil { t.Fatal(err) }
 if err = os.WriteFile("/private/tmp/halo-levelup-views-oracle.json", data, 0600); err != nil { t.Fatal(err) }
}
