package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"testing"
)

func TestHaloRustBipedAimLoaded(t *testing.T) {
	rows := []any{}
	packet := func(kind uint16, ts uint64, pay []byte) []byte {
		h := make([]byte, 16)
		binary.LittleEndian.PutUint16(h, kind)
		binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
		binary.LittleEndian.PutUint64(h[8:], ts)
		return append(h, pay...)
	}
	for run := 0; run < 256; run++ {
		raw := make([]byte, 8)
		binary.LittleEndian.PutUint32(raw, 41)
		binary.LittleEndian.PutUint32(raw[4:], 27)
		chunks := source.MemoryChunks{raw}
		meta := []types.ChunkMeta{{Index: 0, ChunkType: 1}}
		inputs := []any{}
		count := 4
		if run%13 == 0 {
			count = 0
		}
		for c := 0; c < count; c++ {
			data := []byte{}
			for k := 0; k < 2; k++ {
				w := &translocWriter{}
				w.put(1, 0)
				ti := uint64(35)
				if run%7 == 0 {
					ti = 40
				}
				slot := uint64(512 + c*3 + k*30)
				w.put(32, (1<<30)|slot)
				w.put(32, ti)
				w.put(44, 0)
				w.put(32, 0)
				w.put(32, 0)
				w.put(64, 0)
				if run%11 != 0 {
					data = append(data, packet(2, uint64(9000-c*1000), w.pay)...)
				}
			}
			for j := 0; j < 16; j++ {
				w := &translocWriter{}
				w.put(1, 1)
				w.put(13, uint64(512+j))
				w.put(2, 1)
				w.put(2, 0)
				w.put(3, 1)
				w.put(6, 21)
				w.put(1, 1)
				w.put(12, uint64(run+j))
				w.put(11, uint64(1000+j))
				w.put(1, 0)
				kind := uint16(0)
				if j%5 == 0 {
					kind = 3
				}
				data = append(data, packet(kind, uint64(j%4*1000), w.pay)...)
			}
			if run%17 == 0 {
				data = data[:len(data)-7]
			}
			index := c + 1
			switch run % 8 {
			case 1:
				if c == 1 {
					index = 1
				}
				if c == 2 && run%16 < 8 {
					index = 2
				}
			case 2:
				if c == 1 {
					index = 3
				}
			case 3:
				if c == 2 {
					index = 5
				}
			case 4:
				index = 4 - c
			case 5:
				if c == 0 {
					index = -1
				}
			case 6:
				if c == 2 {
					index = 2
				}
				if c == 3 && run%16 < 8 {
					index = 3
				}
			}
			chunks = append(chunks, data)
			meta = append(meta, types.ChunkMeta{Index: index})
			inputs = append(inputs, map[string]any{"index": index, "hex": hex.EncodeToString(data)})
		}
		film, e := source.Load(chunks, meta)
		if e != nil {
			t.Fatal(e)
		}
		fc := NewFilmContext(film)
		aims, e := ScanBipedAimOnly(fc)
		if aims == nil {
			aims = []BipedAim{}
		}
		errKind := ""
		if e == ErrNoFilmChunk {
			errKind = "chunks"
		} else if e == ErrNoReadableFilmChunk {
			errKind = "readable"
		} else if e != nil {
			errKind = "band"
		}
		slots := fc.BipedSlots().Slots()
		if slots == nil {
			slots = []uint32{}
		}
		rows = append(rows, map[string]any{"chunks": inputs, "aims": aims, "slots": slots, "error": errKind})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-biped-aim-loaded.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
