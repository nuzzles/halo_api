package replay
import("testing";"os";"encoding/json";"encoding/hex";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/types")
func TestHaloRustFactsWeaponChanges(t *testing.T){
 rng:=rand.New(rand.NewSource(934457));rows:=[]any{}
 for run:=0;run<1024;run++{
  raw:=[]types.HeldWeaponChange{};origin:=uint64(run%5)*1000000;step:=uint64(run%5)*100000
  for j:=0;j<run%53;j++{r:=types.HeldWeaponChange{TimestampUS:uint64(rng.Intn(50))*100000,Slot:uint32(rng.Intn(8)),Family:rng.Uint32(),Previous:rng.Uint32(),Kind:[]types.HeldWeaponChangeKind{types.HeldWeaponTaken,types.HeldWeaponDropped,types.HeldWeaponSwapped,types.HeldWeaponRestated}[rng.Intn(4)]};if j%5==0{r.Family=^uint32(0)};if j%5==1{r.Previous=^uint32(0)};if j%7==0{r.Family=0};if j%19==0{r.TimestampUS=^uint64(0)};raw=append(raw,r)}
  kinds:=[]string{"taken","dropped","swapped","restated","","unknown","restated\x00","\xff\xe2\x82"};domain:=[]int{-9223372036854775808,-1,0,1,4294967296,9223372036854775807};input:=[]any{}
  for j:=range raw {r:=&raw[j];r.Kind=types.HeldWeaponChangeKind(kinds[(run+j)%len(kinds)]);r.SlotIndex=domain[(run+j)%len(domain)];input=append(input,map[string]any{"TimestampUS":r.TimestampUS,"Slot":r.Slot,"SlotIndex":r.SlotIndex,"Family":r.Family,"Previous":r.Previous,"KindHex":hex.EncodeToString([]byte(r.Kind))})}
  changes,cov:=buildWeaponChanges(raw,origin,step);if changes==nil{changes=[]WeaponChange{}}
  rows=append(rows,map[string]any{"raw":input,"origin":origin,"step":step,"output":map[string]any{"changes":changes,"coverage":cov}})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-facts-weapon-changes.json",b,0600);e!=nil{t.Fatal(e)}
}
