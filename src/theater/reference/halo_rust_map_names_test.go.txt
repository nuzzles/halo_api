package grammar

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"os"
	"testing"
	"unicode"
)

func TestHaloRustMapNames(t *testing.T) {
	inputs := []string{"", "  Live   FIRE - Ranked ", "Fragmentation Heavies", "Breaker Heavies - Ranked", "Breaker - Ranked Heavies", "İSTANBUL", "ΟΣ", "ΟΣ - RANKED", "\u0085İ\u2003Σ\u2029HEAVIES", "X - ranked - ranked"}
	for r := rune(0); r <= unicode.MaxRune; r++ {
		if unicode.IsUpper(r) {
			inputs = append(inputs, string(r), "A"+string(r)+" - Ranked")
		}
	}
	rows := []any{}
	for _, input := range inputs {
		rows = append(rows, map[string]string{"input": input, "output": profile.NormalizeMapName(input)})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-map-names.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
