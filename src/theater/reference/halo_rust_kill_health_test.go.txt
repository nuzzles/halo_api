 rows:=[]any{}
 for _,c:=range []int{0,1,100,1000,10000} {for _,u:=range []int{0,1,17,18,19,35,36,37,179,180,181,359,360,361} {for flags:=0;flags<8;flags++ {for coverage:=0;coverage<5;coverage++ {

 }}}}
 raw,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)};if err=os.WriteFile("/private/tmp/halo-kill-health.json",raw,0600);err!=nil{t.Fatal(err)}
}
