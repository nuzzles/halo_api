func haloRustTransportRow(full,input []byte,writes []any,ops []map[string]any)any{
 r:=greader{b:input};results:=[]any{}
 for _,op:=range ops{var value any;panicked:=false;func(){defer func(){if recover()!=nil{panicked=true}}();switch op["op"]{case "u":value=r.u();case "i":value=r.i();case "byte":value=r.byte8();case "f32":value=math.Float32bits(r.f32());case "str":value=hex.EncodeToString([]byte(r.str()));case "bool":value=r.bool8();case "count":value=r.compte(int(op["arg"].(int64)));case "section":v:=r.tranche(int(op["arg"].(int64)));if v!=nil{value=hex.EncodeToString(v)}}}();err:="";if r.err!=nil{err=r.err.Error()};results=append(results,map[string]any{"value":value,"offset":r.off,"error":err,"panic":panicked});if panicked{break}}
 return map[string]any{"written_hex":hex.EncodeToString(full),"input_hex":hex.EncodeToString(input),"writes":writes,"operations":ops,"results":results}
}
