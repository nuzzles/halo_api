package grammar

import (
 "bytes"
 "encoding/json"
 "log/slog"
 "os"
 "sync"
 "testing"
)

func TestHaloRustRegistryWarning(t *testing.T) {
 previous:=slog.Default()
 defer slog.SetDefault(previous)
 registryWarned=sync.Map{}
 defer func(){registryWarned=sync.Map{}}()
 var output bytes.Buffer
 slog.SetDefault(slog.New(slog.NewJSONHandler(&output,nil)))
 rows:=[]any{}
 for i:=0;i<1024;i++ {
  fp:=uint64(i%127)
  if i%5==0 { fp=KnownRegistryFingerprint }
  if i%11==0 { fp=^uint64(i%127) }
  blocks,slots:=i%51,i%1068
  output.Reset()
  warnUnknownRegistry(fp,blocks,slots)
  var event map[string]any
  if output.Len()>0 {
   if err:=json.Unmarshal(output.Bytes(),&event);err!=nil {t.Fatal(err)}
   delete(event,"time");delete(event,"msg")
  }
  rows=append(rows,map[string]any{"fingerprint":fp,"blocks":blocks,"slots":slots,"warning":event!=nil})
 }
 data,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-registry-warning.json",data,0644);err!=nil {t.Fatal(err)}
}
