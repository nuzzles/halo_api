package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustKFQ(t *testing.T) {
	rng := rand.New(rand.NewSource(414617))
	anchors := []any{}
	walks := []any{}
	for run := 0; run < 256; run++ {
		w := &translocWriter{}
		w.put(1, 0)
		for i := 0; i < run%17; i++ {
			w.put(32, uint64(uint32(1+run%3)<<30|uint32(512+i)))
			w.put(32, uint64(i%9))
			w.put(44, 0)
			for j := 0; j < run%11; j++ {
				w.put(8, uint64(rng.Intn(256)))
			}
			w.put(run%8, 0)
		}
		pay := w.pay
		if run%3 == 0 && len(pay) > 0 {
			pay = pay[:rng.Intn(len(pay))]
		}
		if run%7 == 0 {
			pay = make([]byte, run%128)
			rng.Read(pay)
		}
		anchors = append(anchors, map[string]any{"hex": hex.EncodeToString(pay), "shape": MeasureKeyframeAnchors(pay)})
		from, to := run%14, run%14+run%3
		if run%13 == 0 {
			to = from - 1
		}
		variants := KFQFrameVariants(from, to)
		if variants == nil {
			variants = []KFQVariant{}
		}
		raw := make([]byte, run%40)
		rng.Read(raw)
		reg := &Registry{}
		for ti := 0; ti < 4; ti++ {
			reg.Archetypes = append(reg.Archetypes, Archetype{Index: ti})
		}
		best, all := BestVariant(reg, raw, variants, nil)
		measurements := []any{}
		for _, walk := range all {
			measurements = append(measurements, map[string]any{"walk": walk, "coverage": walk.Coverage(), "tis": walk.SortedTIs()})
		}
		walks = append(walks, map[string]any{"hex": hex.EncodeToString(raw), "from": from, "to": to, "variants": variants, "best": best, "measurements": measurements})
	}
	seeded := []any{}
	for run := 0; run < 128; run++ {
		reg := &Registry{Archetypes: make([]Archetype, 4)}
		for i := range reg.Archetypes {
			reg.Archetypes[i].Index = i
		}
		reg.Archetypes[3].Components = []string{"high-frequency"}
		reg.Archetypes[3].Levels = []uint32{1}
		v := KFQVariant{IDLowBits: 5, Preamble: run % 3, ExtraFields: run%2 == 0}
		bw := &translocWriter{}
		bw.put(v.Preamble, 0)
		prefix := func() {
			if v.ExtraFields {
				bw.put(32, 0)
			}
		}
		extra := func() {
			if v.ExtraFields {
				bw.put(1, 1)
				bw.put(8, 57)
			}
		}
		delta := func(gen uint64) {
			prefix()
			bw.put(1, 1)
			bw.put(5, 1)
			bw.put(2, gen)
			bw.put(1, 1)
			bw.put(7, 17)
			bw.put(1, 0)
			bw.put(3, 1)
			bw.put(6, 0)
			bw.put(8, 99)
		}
		delta(1)
		prefix()
		bw.put(3, 2)
		bw.put(5, 1)
		bw.put(2, 1)
		extra()
		bw.put(32, 42)
		prefix()
		bw.put(3, 1)
		bw.put(5, 1)
		bw.put(2, 2)
		extra()
		bw.put(6, 3)
		bw.put(1, 0)
		bw.put(1, 0)
		bw.put(1, 0)
		bw.put(1, 1)
		bw.put(8, 42)
		bw.put(1, 1)
		bw.put(1, 0)
		bw.put(3, 1)
		bw.put(6, 0)
		bw.put(8, uint64(run))
		delta(2)
		prefix()
		bw.put(3, 0)
		pay := bw.pay
		if run%5 == 0 {
			pay = pay[:len(pay)/2]
		}
		seed := func() *World {
			w := NewWorld(reg)
			switch run % 4 {
			case 0:
				w.BindFull(0x40000001, 3)
			case 1:
				w.BindFull(0x80000001, 3)
			case 2:
				w.BindSoft(1, 3)
			}
			return w
		}
		world := seed()
		walk := WalkPacketRecordsWithWorld(world, pay, v)
		slots := map[uint32]any{}
		for slot, state := range world.slots {
			slots[slot] = map[string]any{"id": state.FullID, "ti": state.TypeIndex, "soft": state.Soft}
		}
		vs := KFQFrameVariants(4, 6)
		best, all := BestVariant(reg, pay, vs, seed)
		seeded = append(seeded, map[string]any{"hex": hex.EncodeToString(pay), "mode": run % 4, "variant": v, "walk": walk, "slots": slots, "variants": vs, "best": best, "all": all})
	}
	p := ProfilDeBalayageParDefaut()
	position := map[string]any{"index_bits": p.Mouvement.WorldObject.IndexW, "world_axis_bits": p.Mouvement.WorldObject.AxisW, "handle_bits": p.Mouvement.Traversal.IndexW, "traversal_axis_bits": p.Mouvement.Traversal.AxisW, "default_axis_bits": profile.LargeursAxeParDefautDuBuild(16), "region_axis_bits": map[string]any{}, "delta_axis_bits": [3]uint{p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth}, "full_precision": p.Mouvement.FullPrecision, "writer_absolute": p.Grammaire.GrammaireEcrivainI0, "delta_handle_tail": p.Mouvement.DeltaHasHandleTail}
	raw, e := json.Marshal(map[string]any{"anchors": anchors, "walks": walks, "seeded": seeded, "position": position, "strict": p.Grammaire.GenerationStricte, "simulation": p.Grammaire.SimStateComplet, "corruption": p.Grammaire.ControleDeCorruption, "mpp": [2]int{p.MPP.Lead, p.MPP.Index}})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-kfq.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
