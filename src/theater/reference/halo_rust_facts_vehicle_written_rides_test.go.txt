package replay

import (
 "log/slog"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"sort"
	"testing"
)

func TestHaloRustFactsVehicleWrittenRides(t *testing.T) {
 previous := slog.Default(); defer slog.SetDefault(previous)
	rng := rand.New(rand.NewSource(872219))
	rows := []any{}
	for run := 0; run < 1024; run++ {
		clock := replayClock{origin: uint64(run%7) * 100000, step: []uint64{0, 1, 100000, 200000}[run%4], frames: []int{-9223372036854775808,-4294967296,-1,0,1,21,4294967296,9223372036854775807}[run%8]}
		in := vehicleRideInputs{clock: clock, drawable: map[types.EquipmentLifeKey]bool{}}
		il := []lifeSpan{{slot: 0, from: 0, to: 900000, xuid: 1}, {slot: 1, from: 0, to: 300000, xuid: 2}, {slot: 1, from: 400000, to: 900000, xuid: 3}}
		idx := map[uint64]int{1: 0, 2: 1, 3: 2}
		owners, xuids, amb := ownersFromLives(il, idx)
		in.reg = IdentityRegistry{own: OwnerReport{lives: il, Owner: owners, SlotXUID: xuids, SlotAmbiguous: amb}}
		lifes := []any{}
		draw := []types.EquipmentLifeKey{}
		for i := 0; i < run%13; i++ {
			key := types.EquipmentLifeKey{Slot: uint32(i/3 + 20), Gen: uint32(i % 3) + 4294967290}
			lo := uint64(rng.Intn(8)) * 100000
			hi := lo + uint64(rng.Intn(5))*100000
			l := vehicleLife{key: key, loUS: lo, hiUS: hi}
			in.lives = append(in.lives, l)
			lifes = append(lifes, map[string]any{"key": key, "lo": lo, "hi": hi})
			if rng.Intn(4) != 0 {
				in.drawable[key] = true
				draw = append(draw, key)
			}
		}
		occ := []types.VehicleOccupancy{}
		for i := 0; i < run%49; i++ {
			o := types.VehicleOccupancy{TimestampUS: uint64(rng.Intn(13)) * 100000, Slot: uint32(rng.Intn(5)), Gen: uint32(rng.Intn(4)), Attached: rng.Intn(4) != 0, ParentSlot: uint32(20 + rng.Intn(6)), ParentGen: uint32(rng.Intn(4)), HasSeat: rng.Intn(3) != 0, Seat: rng.Uint32()}
			occ = append(occ, o)
		}
		in.occupancy = occ
		points := []any{}
		for i := 0; i < run%25; i++ {
			p := grammar.BipedPosition{Slot: uint32(rng.Intn(5)), TimestampUS: uint64(rng.Intn(13)) * 100000, HasWorld: rng.Intn(3) != 0}
			in.bipeds = append(in.bipeds, p)
			points = append(points, map[string]any{"slot": p.Slot, "time": p.TimestampUS, "world": p.HasWorld})
		}
		aims := []grammar.BipedAim{}
		for i := 0; i < run%151; i++ {
			aims = append(aims, grammar.BipedAim{Slot: uint32(rng.Intn(5)), TimestampUS: uint64(rng.Intn(130)) * 10000, YawRaw: rng.Uint32(), PitchRaw: rng.Uint32()})
		}
		in.aimBySlot = vehicleAimBySlot(aims)
		out, st := buildVehicleFilmRides(in)
		keys := []types.EquipmentLifeKey{}
		for k := range out.rides {
			keys = append(keys, k)
		}
		sort.Slice(keys, func(i, j int) bool {
			if keys[i].Slot != keys[j].Slot {
				return keys[i].Slot < keys[j].Slot
			}
			return keys[i].Gen < keys[j].Gen
		})
		rides := []any{}
		for _, k := range keys {
			rides = append(rides, map[string]any{"key": k, "rides": out.rides[k]})
		}
		probes := []any{}
		for i := 0; i < 20; i++ {
			k := types.EquipmentLifeKey{Slot: uint32(20 + rng.Intn(6)), Gen: uint32(rng.Intn(4)) + 4294967290}
			r := VehicleRide{Slot: uint32(rng.Intn(5)), T0: rng.Intn(30) - 2, T1: rng.Intn(30) - 2}
			probes = append(probes, map[string]any{"key": k, "ride": r, "contradicts": out.contredit(k, r)})
		}
		aimprobes := []any{}
		for i := 0; i < 8; i++ {
			start := uint64(rng.Intn(13)) * 100000
			end := uint64(rng.Intn(13)) * 100000
			v := aims
			if i%2 == 0 {
				v = in.aimBySlot[uint32(i%5)]
			}
			a := vehicleRideAimOf(v, start, end, clock)
			if a == nil {
				a = []VehicleAim{}
			}
			if v == nil {
				v = []grammar.BipedAim{}
			}
			aimprobes = append(aimprobes, map[string]any{"input": v, "start": start, "end": end, "aim": a})
		}
        rows = append(rows, map[string]any{"origin": clock.origin, "step": clock.step, "frames": clock.frames, "lives": lifes, "drawable": draw, "occupancy": occ, "positions": points, "aims": aims, "rides": rides, "probes": probes, "aim_probes": aimprobes, "tally": map[string]any{"reads": st.lectures, "named": st.nommees, "outside_life": st.horsVie, "not_drawable": st.nonDessinable, "published": st.publies, "closed_by_read": st.fermetureParLecture, "closed_by_return": st.fermetureParRetour, "closed_by_life": st.fermetureParVie}})
    }
    b,e:=json.Marshal(rows);if e!=nil {t.Fatal(e)}
    if e=os.WriteFile("/private/tmp/halo-facts-vehicle-written-rides.json",b,0644);e!=nil {t.Fatal(e)}
}
