type haloDirectoryTrace struct { Source; Calls []int }
func(p *haloDirectoryTrace) NumChunks() int { p.Calls=append(p.Calls,-1); return p.Source.NumChunks() }
func(p *haloDirectoryTrace) Chunk(i int)([]byte,error) { p.Calls=append(p.Calls,i);return p.Source.Chunk(i) }
func haloDirectoryKind(e error) string {
 if e==nil{return ""};if errors.Is(e,os.ErrNotExist){return "missing"};if errors.Is(e,syscall.EISDIR){return "directory"};return "other"
}
func TestHaloRustSourceDirectoryLifecycle(t *testing.T) {
 rows:=[]any{}
 for c:=0;c<64;c++ {
  dir:=t.TempDir(); n:=c%8+1; inputs:=[]any{}
  for j:=0;j<n;j++ {

   name:=fmt.Sprintf("chunk_%02d.bin",j*2+1);if e:=os.WriteFile(filepath.Join(dir,name),data,0600);e!=nil{t.Fatal(e)}
   inputs=append(inputs,map[string]any{"name":name,"hex":hex.EncodeToString(data)})
  }
  src,e:=newDirSource(dir);if e!=nil{t.Fatal(e)}
  first,e:=src.Chunk(0);if e!=nil{t.Fatal(e)}

  second,e:=src.Chunk(0);if e!=nil{t.Fatal(e)}
  if e=os.WriteFile(filepath.Join(dir,"chunk_999.bin"),[]byte{0xff},0600);e!=nil{t.Fatal(e)}

  reads:=[]any{}
  for j:=0;j<n;j++ {b,err:=src.Chunk(j);reads=append(reads,map[string]any{"hex":hex.EncodeToString(b),"kind":haloDirectoryKind(err)})}
