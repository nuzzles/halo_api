package replay
import (
 "encoding/json"
 "levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
 "math/rand"
 "os"
 "testing"
)
func TestHaloRustFactsVehicleAim(t *testing.T) {
 rng:=rand.New(rand.NewSource(849213))
 rows:=[]any{}
 times:=[]uint64{0,1,100000,4294967296,9223372036854775807,9223372036854775808,18446744073709551615}
 frames:=[]int{-9223372036854775808,-4294967296,-1,0,1,20,4294967296,9223372036854775807}
 for i:=0;i<1024;i++ {
  aims:=[]grammar.BipedAim{}
  for j:=0;j<i%73;j++ {
   time:=uint64(rng.Intn(100))*10000
   if j%7==0 {time=times[rng.Intn(len(times))]}
   aims=append(aims,grammar.BipedAim{Slot:uint32(j%3),TimestampUS:time,YawRaw:rng.Uint32(),PitchRaw:rng.Uint32()})
  }
  grouped:=vehicleAimBySlot(aims)
  if grouped==nil {grouped=map[uint32][]grammar.BipedAim{}}
  clock:=replayClock{origin:times[i%len(times)],step:[]uint64{0,1,100000,4294967296,18446744073709551615}[i%5],frames:frames[i%len(frames)]}
  probes:=[]any{}
  for j:=0;j<12;j++ {
   input:=aims
   if j%2==0 {input=grouped[uint32(j%3)]}
   if input==nil {input=[]grammar.BipedAim{}}
   start,end:=times[(i+j)%len(times)],times[(i+j+2)%len(times)]
   if j%3==0 {start,end=0,18446744073709551615}
   out:=vehicleRideAimOf(input,start,end,clock)
   if out==nil {out=[]VehicleAim{}}
   probes=append(probes,map[string]any{"input":input,"start":start,"end":end,"aim":out})
  }
  rows=append(rows,map[string]any{"input":aims,"grouped":grouped,"origin":clock.origin,"step":clock.step,"frames":clock.frames,"probes":probes})
 }
 b,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-facts-vehicle-aim.json",b,0644);err!=nil {t.Fatal(err)}
}
