package replay
import("testing";"os";"encoding/json";"math/rand";"sort";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar";"levelup/go-api/internal/games/halo_infinite/film/internal/facts/fallback")
func TestHaloRustReplayTracks(t *testing.T){
 rng:=rand.New(rand.NewSource(41201));rows:=[]any{}
 for run:=0;run<1024;run++{
  ps:=[]grammar.BipedPosition{};samples:=[]any{};ls:=[]lifeSpan{};lives:=[]any{}
  for i:=0;i<run%160;i++{p:=grammar.BipedPosition{Slot:uint32(rng.Intn(6)),TimestampUS:uint64(rng.Intn(2000))*100000,HasWorld:rng.Intn(6)!=0,X:float32(rng.Intn(100000)-50000)/317,Y:float32(rng.Intn(100000)-50000)/307,Z:float32(rng.Intn(10000)-5000)/313};p.HasYaw=rng.Intn(3)!=0;p.YawRaw=uint32(rng.Intn(4096));p.PitchRaw=uint32(rng.Intn(2048));p.HasBody=rng.Intn(3)!=0;p.Body.Health=float32(rng.Intn(600)-300)/200;p.HasShield=rng.Intn(3)!=0;p.Shield.Shield=float32(rng.Intn(600)-100)/200;ps=append(ps,p)}
  sort.SliceStable(ps,func(i,j int)bool{return ps[i].TimestampUS<ps[j].TimestampUS})
  for _,p:=range ps{var aim,health,shield any;if p.HasYaw{aim=[]uint32{p.YawRaw,p.PitchRaw}};if p.HasBody{health=p.Body.Health};if p.HasShield{shield=p.Shield.Shield};samples=append(samples,map[string]any{"position":map[string]any{"Slot":p.Slot,"TimestampUS":p.TimestampUS,"X":p.X,"Y":p.Y,"Z":p.Z,"HasWorld":p.HasWorld},"aim":aim,"health":health,"shield":shield})}
  if run%3!=0 {for slot:=uint32(0);slot<6;slot++{for r:=0;r<3;r++{if run%7==0&&r==1{continue};l:=lifeSpan{slot:slot,from:int64(r)*70000000,to:int64(r+1)*70000000-1};ls=append(ls,l);lives=append(lives,map[string]any{"slot":l.slot,"from":l.from,"to":l.to,"xuid":0,"bid":"","cause":"","named_by":""})}}}
  origin,step:=uint64(run%3)*100000,uint64([]int{100000,1000000,10000000}[run%3]);minPoints:=1+run%4;scoped:=func(slot uint32,ts uint64)int{return int((ts/100000+uint64(slot))%4)}
  fb:=fallback.NouveauCompteur();tracks,cov:=decimateTracks(ps,decoupeDesTraces{origin:origin,step:step,minPoints:minPoints,vies:ls,scoped:scoped,fb:fb})
  rows=append(rows,map[string]any{"samples":samples,"lives":lives,"origin":origin,"step":step,"min_points":minPoints,"output":map[string]any{"tracks":tracks,"coverage":cov,"replication_gap_fallbacks":fb.Compte(fallback.NomVieCoupeeAuTrouDeReplication)}})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-tracks.json",raw,0600);e!=nil{t.Fatal(e)}
}
