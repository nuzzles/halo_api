package objectives
import("testing";"encoding/json";"encoding/hex";"encoding/binary";"path/filepath";"os";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/internal/source";"levelup/go-api/internal/games/halo_infinite/film/types";"levelup/go-api/internal/domain")
func TestHaloRustObjectiveExtract(t *testing.T){
 rng:=rand.New(rand.NewSource(922));rows:=[]any{};modes:=[]string{"CTF:Arena","Flag Bomb","Strongholds","Land Grab","Total Control","King of the Hill","KOTH","Oddball","Assault:One Bomb","VIP","Slayer"}
 for i:=0;i<1024;i++{
  bitdata:=[]byte{};put:=func(n int,v uint64){for b:=n-1;b>=0;b--{bitdata=append(bitdata,byte(v>>uint(b)&1))}};put(i%8,0)
  for j:=0;j<i%9;j++{
   x:=uint64(2500000000000000+j);if i%13==0{x=2000000000000000};if i%17==0{x=3000000000000000}
   for b:=0;b<8;b++{put(8,x>>uint(b*8))};put(8,uint64([]int{0x2d,0x25,0x26}[i%3]));put(8,0xc0)
   block:=make([]byte,60);block[36]=byte(j);block[37]=byte((i+j)%7);block[47]=10;if i%11==0{block[47]=9};tm:=uint32(rng.Intn(20)*1000);block[48]=byte(tm>>24);block[49]=byte(tm>>16);block[50]=byte(tm>>8);block[51]=byte(tm)
   if i%19==0{put(32,0x2ee0)}
   for _,b:=range block{put(8,uint64(b))};put(32,0x2ee0)
  }
  data:=make([]byte,(len(bitdata)+7)/8);for j,b:=range bitdata{data[j/8]|=b<<uint(7-j%8)};if i%7==0 && len(data)>0{data=data[:len(data)-1]}
  footer:=scanTh10Events(data);if footer==nil{footer=[]FooterEvent{}}
  roster:=MapRoster{};for j:=0;j<8;j++{if j%3!=i%3{roster[formatXUID(uint64(2500000000000000+j))]=(i+j)%4}}
  packet:=func(kind uint16,ts uint64,p []byte)[]byte{b:=make([]byte,16+len(p));binary.LittleEndian.PutUint16(b,kind);binary.LittleEndian.PutUint32(b[4:],uint32(len(p)));binary.LittleEndian.PutUint64(b[8:],ts);copy(b[16:],p);return b}
  tiers,_:=hex.DecodeString("a4000000034800000106900000020d200000051a4000000a3480000015")
  gameplay:=packet(0,10000000,tiers)
  for j:=0;j<7;j++{payload:=tiers;if j%3==i%3{payload=tiers[:20]};gameplay=append(gameplay,packet(uint16(j%4/3),uint64(j*3000000),payload)...)}
  chunks:=source.MemoryChunks{data,gameplay,{},data,gameplay}
  metas:=[]types.ChunkMeta{{Index:8,ChunkType:3},{Index:12,ChunkType:2,StartMS:11000},{Index:8,ChunkType:3},{Index:2,ChunkType:3},{Index:1,ChunkType:2,StartMS:7000}}
  if i%5==0{metas[0].Index=-1;metas[3].Index=-2};if i%7==0{metas[2].Index=9};if i%11==0{metas[1].ChunkType=1}
  film,e:=source.Load(chunks,metas);if e!=nil{t.Fatal(e)}
  chunkRows:=[]any{};for j,c:=range chunks{chunkRows=append(chunkRows,map[string]any{"data":hex.EncodeToString(c),"index":metas[j].Index,"kind":metas[j].ChunkType,"start":metas[j].StartMS})}
  selected:=FooterEvents(film);if selected==nil{selected=[]FooterEvent{}}
  captureTimes:=[]int{};for _,b:=range collectCaptureBursts(film){captureTimes=append(captureTimes,b.matchMS)}
  events,ctl:=Extract("fixture",modes[i%len(modes)],film,roster);if events==nil{events=[]domain.ObjectiveEvent{}};for j:=range events{if events[j].Players==nil{events[j].Players=[]domain.ObjectiveEventPlayer{}}}
  // Direct CTF scorer selection covers overlapping windows and timestamp ties.
  bursts:=[]int{};for j:=0;j<i%7;j++{bursts=append(bursts,rng.Intn(20)*1000+[]int{-2001,-2000,0,2000,2001}[j%5])};scorers:=[]any{};for _,b:=range bursts{e,ok:=captureScorer(footer,b);var v any;if ok{v=e};scorers=append(scorers,v)}
  rows=append(rows,map[string]any{"chunks":chunkRows,"selected":selected,"capture_times":captureTimes,"data":hex.EncodeToString(data),"footer":footer,"variant":modes[i%len(modes)],"roster":roster,"events":events,"teams":ctl,"bursts":bursts,"scorers":scorers})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-objective-extract.json",b,0600);e!=nil{t.Fatal(e)}
}

func TestHaloRustObjectiveCorpus(t *testing.T) {
 rows:=[]any{}
 root:="/Users/simbleau/git/halo_api/experiments/films"
 err:=filepath.WalkDir(root,func(path string,d os.DirEntry,err error)error{
  if err!=nil{return err};if d.Name()!="film.json"{return nil}
  raw,err:=os.ReadFile(path);if err!=nil{return err}
  var manifest struct{Chunks []struct{Index int `json:"index"`;Kind int `json:"chunk_type"`;Start int `json:"start_time_offset_ms"`;File string `json:"file"`}}
  if err=json.Unmarshal(raw,&manifest);err!=nil{return err}
  chunks:=source.MemoryChunks{};meta:=[]types.ChunkMeta{}
  for _,c:=range manifest.Chunks{b,e:=os.ReadFile(filepath.Join(filepath.Dir(path),c.File));if e!=nil{return e};chunks=append(chunks,b);meta=append(meta,types.ChunkMeta{Index:c.Index,ChunkType:c.Kind,StartMS:c.Start})}
  film,err:=source.Load(chunks,meta);if err!=nil{return err}
  footer:=FooterEvents(film);if footer==nil{footer=[]FooterEvent{}}
  bursts:=[]int{};for _,b:=range collectCaptureBursts(film){bursts=append(bursts,b.matchMS)}
  events,ctl:=Extract("fixture","CTF",film,MapRoster{});if events==nil{events=[]domain.ObjectiveEvent{}};for i:=range events{if events[i].Players==nil{events[i].Players=[]domain.ObjectiveEventPlayer{}}}
  relative,_:=filepath.Rel(root,filepath.Dir(path));rows=append(rows,map[string]any{"folder":relative,"footer":footer,"bursts":bursts,"events":events,"teams":ctl});return nil
 });if err!=nil{t.Fatal(err)}
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-objective-corpus.json",b,0600);e!=nil{t.Fatal(e)}
}
