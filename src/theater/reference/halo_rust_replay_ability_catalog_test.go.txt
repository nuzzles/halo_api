func TestHaloRustReplayAbilityCatalog(t *testing.T){
 c,e:=replaylabels.Load("/private/tmp/halo-levelup-v75","halo_infinite");if e!=nil{t.Fatal(e)}
 palettes:=[]any{};for _,p:=range c.Abilities{palettes=append(palettes,map[string]any{"id":p.ID,"markers":p.Markers,"ranks":p.Ranks,"families":p.Families})}
 raw,e:=json.MarshalIndent(map[string]any{"palettes":palettes,"impulse_families":c.AbilityImpulseFamilies,"charge_families":c.AbilityChargeFamilies},"","  ");if e!=nil{t.Fatal(e)}
 if e=os.WriteFile("/private/tmp/halo-replay-ability-catalog.json",raw,0600);e!=nil{t.Fatal(e)}
}
