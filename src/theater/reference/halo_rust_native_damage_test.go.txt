package grammar

import (
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
 "levelup/go-api/internal/games/halo_infinite/film/internal/source"
)

// Direct wire readers only. No reconstructed World, slot-base selection or pairing.
func TestHaloRustNativeDamage(t *testing.T) {
 root:="/Users/simbleau/git/halo_api/experiments/films"
 paths,err:=filepath.Glob(filepath.Join(root,"*/*/chunk-*-type-2.bin"));if err!=nil { t.Fatal(err) }
 rows:=[]any{}
 for _,path:=range paths {
  data,err:=os.ReadFile(path);if err!=nil { t.Fatal(err) }
  relative,err:=filepath.Rel(root,path);if err!=nil { t.Fatal(err) }
  offset:=0
  for ordinal,pk:=range source.Paquets(data,0) {
   start:=offset+16;offset=start+len(pk.Payload)
   if pk.Type!=0 || len(pk.Payload)<2 || pk.Payload[0]!=0xc0 { continue }
   br:=LecteurSur(pk.Payload)
   ok:=readPacketHead(br).Type==0
   row:=map[string]any{"file":relative,"packet":ordinal,"offset":start,"size":len(pk.Payload),"time":pk.TS,"ok":ok}
   if ok {
    d:=WeaponDamage{TimestampUS:pk.TS,VictimIdx:-1,ResponsibleIdx:-1}
    if index,present:=lot1RefDom1(br);present { d.VictimIdx=int(index) }
    if index,present:=lot1RefDom1(br);present { d.ResponsibleIdx=int(index) }
    lot1RefDom(br,7)
    body:=lot1DecodeDamageAftermath(br)
    d.Source=body.sourceID;d.HasSource=body.hasSource;d.Negative=body.negatif;d.MagClear=body.dmgClear;d.MagRaw=body.dmgRaw
    row["damage"]=d;row["secondary"]=body.dmg2Raw
    var victim any;if body.hasVictim { victim=body.victimIdx };row["body_victim"]=victim
   }
   row["end"]=br.BitPos();rows=append(rows,row)
  }
 }
 raw,err:=json.Marshal(rows);if err!=nil { t.Fatal(err) }
 if err=os.WriteFile("/private/tmp/halo-native-damage.json",raw,0600);err!=nil { t.Fatal(err) }
 t.Logf("%d selected native damage attempts",len(rows))
}
