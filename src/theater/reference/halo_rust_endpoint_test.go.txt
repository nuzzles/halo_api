package grammar

import (
	"encoding/json"
	"math"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustEndpoint(t *testing.T) {
	rng := rand.New(rand.NewSource(41484))
	rows := []any{}
	widths := []uint{0, 1, 2, 8, 16, 17, 31, 32, 63, 64, 65}
	for run := 0; run < 4096; run++ {
		w := widths[run%len(widths)]
		excl := (run/4)%2 == 0
		exact := (run/8)%2 == 0
		levels := uint64(1) << w
		if excl {
			levels--
		}
		q := rng.Uint64()
		switch run % 4 {
		case 0:
			q = 0
		case 1:
			q = levels - 1
		case 2:
			q = (levels - 1) / 2
		}
		min, max := float32(-1), float32(1)
		switch run % 7 {
		case 1:
			min, max = 0, 36000
		case 2:
			min, max = -0.025, 6553.5752
		case 3:
			min, max = 17, -31
		case 4:
			min, max = 0, 0
		case 5:
			min, max = -1e-25, 1e-25
		case 6:
			min, max = float32(rng.Intn(10000)-5000)*0.37, float32(rng.Intn(10000)-5000)*1.71
		}
		rows = append(rows, map[string]any{"q": q, "width": w, "min": math.Float32bits(min), "max": math.Float32bits(max), "exclude": excl, "exact": exact, "value": math.Float32bits(DequantEndpoint(q, min, max, w, excl, exact))})
	}
	timers := make([]uint32, 1<<17)
	for q := range timers {
		timers[q] = math.Float32bits(NavpointManualTimerValue(uint64(q)))
	}
	raw, e := json.Marshal(map[string]any{"cases": rows, "navpoint": timers})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-endpoint.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
