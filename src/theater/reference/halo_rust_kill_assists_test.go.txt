package killsource
import("testing";"encoding/json";"os";"math/rand")
func TestHaloRustKillAssists(t *testing.T){
 rows:=[]any{};rng:=rand.New(rand.NewSource(726));names:=[]string{"A","B","C","D","?"}
 packet:=func(p paquetID)any{if !p.ok{return nil};return map[string]any{"chunk":p.chunk,"packet":p.pidx}}
 targets:=func(ks []Kill)[]any {out:=[]any{};for _,k:=range ks {var killer any;if k.Feed.Present{killer=k.Feed.Killer};out=append(out,map[string]any{"time_ms":k.TimeMS,"victim":k.Victim,"killer":killer,"packet":packet(k.paquet),"assist":map[string]any{"name":k.Assist.Name,"index":k.Assist.Index,"known":k.Assist.Known,"rejected":k.Assist.Rejected,"extra":k.Assist.Extra},"killer_damage":map[string]any{"pct":k.KillerDamage.Pct,"known":k.KillerDamage.Known},"assist_damage":map[string]any{"pct":k.AssistDamage.Pct,"known":k.AssistDamage.Known}})};return out}
 for i:=0;i<1024;i++ {
  c:=decodeCtx{roster:&roster{names:names,perm:[]int{0,1,2,3,4}}};s:=assistScan{gate15:i%2==0};ks:=[]Kill{}
  for j:=0;j<1+i%9;j++ {ks=append(ks,Kill{TimeMS:j*2500,Victim:names[j%4],Feed:FeedTruth{Killer:names[(j+1)%4],Present:(i+j)%3!=0},paquet:paquetID{chunk:0,pidx:j,ok:(i+j)%4!=0}})}
  for j:=0;j<2+i%21;j++ {victim:=j%4;killer:=(j+1)%4;assist:=rng.Intn(8)-1;ms:=rng.Intn(12)*500;chunk:=rng.Intn(2);pidx:=rng.Intn(9)
   if i%2==0 {k:=ks[j%len(ks)];victim=j%len(ks)%4;killer=(victim+1)%4;ms=k.TimeMS+[]int{-2501,-2500,0,2500,2501}[j%5];chunk=0;pidx=j%len(ks)}
   s.recs=append(s.recs,killEventRec{ms:ms,chunk:chunk,pidx:pidx,fields:killEventFields{killer:killer,victim:victim,assist:assist,killerPct:uint32(rng.Intn(301)),assistPct:uint32(rng.Intn(301)),flag:j%2,end:100}})
  }
  before:=targets(ks);events:=[]any{};for _,r:=range s.recs {f:=r.fields;events=append(events,map[string]any{"time_ms":r.ms,"packet":packet(paquetID{r.chunk,r.pidx,true}),"fields":map[string]any{"killer":f.killer,"victim":f.victim,"assist":f.assist,"killer_pct":f.killerPct,"assist_pct":f.assistPct,"flag":f.flag,"end":f.end}})}
  st:=c.attachAssists(ks,&s)
  stats:=map[string]any{"kill_events":st.KillEvents,"attached":st.Attached,"multi":st.Multi,"named":st.Named,"no_assist":st.NoAssist,"rejected_self":st.RejectedSelf,"rejected_victim":st.RejectedVictim,"rejected_roster":st.RejectedRoster,"flag_set":st.FlagSet,"assist_multi":st.AssistMulti,"assist_extra_total":st.AssistExtraTotal,"assist_field_disagree":st.AssistFieldDisagree,"killer_pct_over100":st.KillerPctOver100,"assist_pct_over100":st.AssistPctOver100,"window_fallback":st.ParLaFenetre,"gate15":st.Gate15}
  rows=append(rows,map[string]any{"names":names,"events":events,"before":before,"after":targets(ks),"stats":stats,"gate":s.gate15})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-kill-assists.json",b,0600);e!=nil{t.Fatal(e)}
}
