package replay

import (
	"encoding/json"
	"fmt"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/fallback"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives"

func TestHaloRustReplayScoreDocument(t *testing.T) {
	old := slog.Default()
	defer slog.SetDefault(old)
	h := &rustScoreSeriesHandler{}
	slog.SetDefault(slog.New(h))
	rows := []any{}
	for run := 0; run < 1536; run++ {
		in := ScoreInput{TeamByXUID: map[string]int{}, TargetScore: []int{0, 10, 20, 50, 100}[run%5]}
		records := []types.StatRecord{}
		lines := []types.PlayerLine{}
		sums := [2]int{}
		for p := 0; p < 2+run%5; p++ {
			team := p % 2
			k := p + 3
			x := fmt.Sprint(100 + p)
			sums[team] += k
			in.TeamByXUID[x] = team
			lines = append(lines, types.PlayerLine{XUID: x, Kills: k, Deaths: p + 1, Assists: p})
			for j := 0; j <= k; j++ {
				d := j
				if d > p+1 {
					d = p + 1
				}
				a := j
				if a > p {
					a = p
				}
				records = append(records, types.StatRecord{Slot: 10 + p*2, Round: 0, TimeMS: j * 1000, Comps: map[int]types.StatValue{2: {A: int64(j), B: int64(d)}, 3: {A: int64(a)}, 23: {A: int64(j * (p%3 + 1))}}})
			}
		}
		for t := 0; t < run%4; t++ {
			slot := 6 + t
			goal := 10 + 5*t
			frags := sums[t%2]
			if run%11 == 0 {
				frags++
			}
			for j := 0; j <= goal; j++ {
				k := j
				if k > frags {
					k = frags
				}
				records = append(records, types.StatRecord{Slot: slot, Round: 0, TimeMS: j * 1000, Comps: map[int]types.StatValue{0: {A: int64(j), B: 0}, 2: {A: int64(k), B: 0}}})
			}
		}
		if run%7 == 0 {
			lines = []types.PlayerLine{}
		}
		if run%13 == 0 {
			in.TeamByXUID = map[string]int{}
		}
		if run%3 == 0 {
			in.TeamScores = &[2]int{10, 15}
		}
		if run%3 == 1 {
			in.TeamScores = &[2]int{15, 10}
		}
		if run%17 == 0 {
			in.TeamScores = &[2]int{10, 10}
		}
		in.Records = records
		in.Lines = lines
		clock := scoreClock{originMS: (run%4 - 1) * 1000, intervalMS: []int{-1, 0, 100, 1000}[run%4], frames: []int{0, -1, 400, 100}[run%4]}
		score := loadScoreSeries(records, objectives.ModeScoreComponent, true)
		tf := loadScoreSeries(records, objectives.KillsComponent, true)
		pf := loadScoreSeries(records, objectives.KillsComponent, false)
		slots := teamSlotsOf(score, tf)
		ids, method := resolveTeamIdentity(&in, slots, score, tf, pf, objectives.SlotIdentityFrom(records, lines))
		teams := buildTeamScores(slots, score, ids, clock)
		if teams == nil {
			teams = []TeamScore{}
		}
		fullRecords := append([]types.StatRecord(nil), records...)
		if run%9 == 0 {
			for i := range fullRecords {
				fullRecords[i].Round = 2
			}
		}
		if run%19 == 0 {
			fullRecords = []types.StatRecord{}
		}
		fullLines := lines
		if run >= 1024 {
			fullRecords = []types.StatRecord{}
			nr := 2 + run%3
			for r := 0; r < nr; r++ {
				for j := 0; j <= 30; j++ {
					at := (nr-r)*60000 + j*1000
					fullRecords = append(fullRecords, types.StatRecord{Slot: 6, Round: r, TimeMS: at, Comps: map[int]types.StatValue{0: {A: int64(j)}, 2: {A: int64(j)}}})
					if run%2 == 0 {
						at = r*60000 + j*1000
					}
					fullRecords = append(fullRecords, types.StatRecord{Slot: 10, Round: r, TimeMS: at, Comps: map[int]types.StatValue{0: {A: int64(j)}, 1: {B: int64(j)}, 2: {A: int64(j), B: int64(j)}, 3: {A: int64(j)}, 23: {A: int64(j * 2)}}})
				}
			}
			total := 30
			if run%2 == 0 {
				total *= nr
			}
			fullLines = []types.PlayerLine{{XUID: "100", Kills: total, Deaths: total, Assists: total}}
		}
		fullInput := in
		fullInput.Lines = fullLines
		fullInput.Records = fullRecords
		fullInput.HoldTicksPerPoint = []int{-1, 0, 35}[run%3]
		fullInput.Truncated = run%2 == 0
		fb := fallback.NouveauCompteur()
		ip := &fullInput
		if run%17 == 0 {
			ip = nil
		}
		events := []objectives.IdentifiedEvent{}
        for j := 0; j < run%19; j++ {
            events = append(events, objectives.IdentifiedEvent{NamedEvent: objectives.NamedEvent{TimeMS:(j%7-1)*1000, Slot:10, Stat:[]string{"flag_captures","kills","bomb_detonations","assists","zone_secures"}[j%5],Comp:0,Side:"A"},XUID:[]string{"100","101","","102"}[j%4]})
        }
        origin := int64(clock.originMS)
        seededTarget:=999
        doc := ReplayDocument{MatchID:"fixture",FrameCount:clock.frames,FrameIntervalMS:999,OriginMs:&origin,Tracks:[]Track{{Slot:1,XUID:"100",Points:[]Point{}},{Slot:2,Points:[]Point{}}},Objectives:[]ObjectiveAction{{Stat:"seed"}},ScoreTimeline:&ScoreTimeline{TargetScore:&seededTarget}}
        if run%3==0 {doc.OriginMs=nil}; if run%5==0 {doc.Tracks=nil}
        if run%2==0 {doc.Coverage=&Coverage{Shots:LayerCoverage{Available:999}}}
        bridge:=map[uint32]uint64{2:101}
        unnamed, refused:=run%4,0; if run%11==0 {refused=run%7+1}
        teamReport:=grammar.TeamScanReport{Component:"managed-player-team-designator-component",Records:30,Read:run%7,Unreached:run%4,OutOfDomainIndex:run%3,OutOfDomainValue:run%5,EntityDivergences:run%2,IndexDivergences:run%3}
        switch run%9{case 0:teamReport.ArchetypeAbsent=true;case 1:teamReport.ComponentMismatch=true;case 2:teamReport.Component=""}
        byIndex:=map[int]int{0:run%3-1,1:1};control:=map[string]int{"100":run%2,"101":1}
        doc.Roster=[]RosterEntry{{XUID:"100",FilmIndex:0},{XUID:"101",FilmIndex:1},{XUID:"102",FilmIndex:2},{FilmIndex:0}}
        before,_:=json.Marshal(doc)
        a:=assemblage{matchID:"fixture",doc:doc,interval:clock.intervalMS,equipes:teamPublication{byIndex:byIndex,rep:teamReport,controle:control},viesTotal:7,viesNommees:run%4,viesSlotAmbigu:run%3,reg:IdentityRegistry{own:OwnerReport{SlotXUID:bridge}},opt:Options{Score:ip,Objectives:events,ObjectivesUnnamed:unnamed,ObjectivesRefused:refused,Fallbacks:fb}}
        h.records = []any{}
        a.poserScoreEtObjectifs()
        stageLogs:=h.records
        stage:=map[string]any{"team_input":map[string]any{"by_index":byIndex,"report":teamReport},"control":control,"before":json.RawMessage(before),"after":a.doc,"teams":a.teamCov,"objectives":a.objCov,"score":a.scoreCov,"events":events,"unnamed":unnamed,"refused":refused,"bridge":bridge,"counts":[]int{7,run%4,run%3},"diagnostics":stageLogs,"round_zero_fallbacks":fb.Compte(fallback.NomReplayMancheZeroDecretee),"without_track":countActionsWithoutTrack(a.doc.Objectives,a.doc.Tracks,bridge)}
        fb=fallback.NouveauCompteur()
        h.records = []any{}
        tl, cov := buildScoreTimeline(ip, nil, clock, fb)
		warnings := h.records
		full := map[string]any{"timeline": tl, "coverage": cov, "round_zero_fallbacks": fb.Compte(fallback.NomReplayMancheZeroDecretee)}
		rows = append(rows, map[string]any{"stage":stage,"full": full, "full_records": fullRecords, "full_lines": fullLines, "warnings": warnings, "hold_ticks_per_point": fullInput.HoldTicksPerPoint, "truncated": fullInput.Truncated, "absent": ip == nil, "records": records, "lines": lines, "team_by_xuid": in.TeamByXUID, "team_scores": in.TeamScores, "target_score": in.TargetScore, "clock": map[string]int{"origin_ms": clock.originMS, "interval_ms": clock.intervalMS, "frames": clock.frames}, "expected": map[string]any{"teams": teams, "identity": method, "target_score": publishableTarget(in.TargetScore, slots, score, len(teams)), "mode_supported": len(score.total) > 0}})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-replay-score-document.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
