package killsource
import (
 "encoding/json"
 "os"
 "testing"
)
func TestHaloRustKillOptions(t *testing.T) {
 encode := func(o Options) any {return map[string]any{"bots":o.Bots,"self_source":o.SelfSource,"strong_tag_required":o.StrongTagRequired,"multiplicity_max":o.MultiplicityMax,"bijection_restarts":o.BijectionRestarts,"views":o.Views}}
 values := []int{-1<<63,-2147483649,-2147483648,-40,-1,0,1,2,8,40,2147483647,2147483648,4294967295,4294967296,1<<63-1}
 rows := []any{}
 for i := 0; i < 512; i++ {
  o := Options{Bots:i&1!=0,SelfSource:i&2!=0,StrongTagRequired:i&4!=0,MultiplicityMax:values[i%len(values)],BijectionRestarts:values[(i/len(values))%len(values)],Views:values[(i*7+3)%len(values)]}
  before := encode(o); o.normalize()
  rows = append(rows,map[string]any{"input":before,"normalized":encode(o)})
 }
 raw,err := json.Marshal(map[string]any{"default":encode(DefaultOptions()),"cases":rows});if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-kill-options.json",raw,0600);err!=nil{t.Fatal(err)}
}
