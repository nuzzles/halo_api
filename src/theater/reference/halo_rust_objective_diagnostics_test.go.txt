package replay

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"log/slog"
	"os"
	"testing"
)

func TestHaloRustObjectiveDiagnostics(t *testing.T) {
	old := slog.Default()
	defer slog.SetDefault(old)
	rows := []any{}
	for i := 0; i < 128; i++ {
		recs := []types.StatRecord{}
		deaths := []Death{}
		for r := 0; r < 2; r++ {
			for j := 0; j <= 30; j++ {
				at := r*60000 + j*1000
				if i%3 == 0 {
					at = (2-r)*60000 + j*1000
				}
				comps := map[int]types.StatValue{0: {A: int64(j)}}
				for _, c := range []int{1, 2, 3, 19, 20, 21, 22, 23, 24, 25} {
					v := int64(j)
					if i%3 == 1 {
						v *= 17
					}
					comps[c] = types.StatValue{A: v, B: v, C: v, D: v, HasC: true, HasD: true}
				}
				recs = append(recs, types.StatRecord{Slot: 10, Round: r, TimeMS: at, Comps: comps})
				if j > 0 {
					deaths = append(deaths, Death{XUID: 100, TimeMS: int64(at)})
				}
			}
		}
		bursts := []int{}
		for b := 0; b < i%65; b++ {
			bursts = append(bursts, b*1000)
		}
		h := &rustScoreSeriesHandler{records: []any{}}
		slog.SetDefault(slog.New(h))
		signals := flagFilmSignalsOf(FlagInput{Records: recs, Bursts: bursts})
		events := objectives.NamedEventsFrom(recs, objectives.ObjectiveTypeFlag)
		if events == nil {
			events = []objectives.NamedEvent{}
		}
		flagLogs := h.records
		h.records = []any{}
		clock := replayClock{origin: uint64(i%3) * 1000000, step: []uint64{0, 100000}[i%2], frames: []int{0, 2000}[i%2]}
		reg := IdentityRegistry{own: OwnerReport{DeathOffsetMS: int64(i%5-2) * 1000}}
		doc := ReplayDocument{Coverage: &Coverage{}}
		recognized := i%4 != 0
		attachVipCrown(&doc, Options{Vip: VipInput{Scanned: recognized, Records: recs}, Deaths: deaths}, reg, clock)
		if doc.VipCrown == nil {
			doc.VipCrown = []VipPeriod{}
		}
		rows = append(rows, map[string]any{"records": recs, "deaths": deaths, "bursts": bursts, "signals": signals, "events": events, "flag_logs": flagLogs, "recognized": recognized, "clock": map[string]any{"origin_us": clock.origin, "step_us": clock.step, "frames": clock.frames, "death_offset_ms": reg.DeathOffsetMS()}, "vip": map[string]any{"periods": doc.VipCrown, "coverage": doc.Coverage.VipCrown}, "vip_logs": h.records})
	}
	data, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-objective-diagnostics.json", data, 0600); e != nil {
		t.Fatal(e)
	}
}
