package objectives

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"testing"
)

func TestHaloRustRoundHelpers(t *testing.T) {
	rows := []any{}
	for run := 0; run < 128; run++ {
		recs := []types.StatRecord{}
		for round := -1; round <= 9; round++ {
			count := []int{0, 1, 2, 3, 24, 25, 26, 249, 250, 251}[(run+round+1)%10]
			for j := 0; j < count; j++ {
				slot := 10 + j%3*2
				if run%4 == 0 {
					slot = 6 + j%2*2
				}
				value := int64(j % 12)
				b := int64(0)
				if j%7 == 0 {
					b = 251
				}
				if j%11 == 0 {
					value = -1
				}
				comps := map[int]types.StatValue{}
				if run%5 != 0 {
					comps[0] = types.StatValue{A: value, B: b}
				}
				ts := round*10000 + j
				if run%3 == 0 {
					ts = round*10000 - j
				}
				recs = append(recs, types.StatRecord{TimeMS: ts, Slot: slot, Round: round, Comps: comps})
			}
		}
		runs := modeScoreRunsByRound(recs)
		material := materialRounds(recs)
		present := presentRounds(recs)
		real, decreed := contiguousRounds(runs, material, present)
		probes := []any{}
		for r := -2; r <= 9; r++ {
			probes = append(probes, map[string]any{"round": r, "after": hasRoundAfter(runs, material, r)})
		}
		rows = append(rows, map[string]any{"records": recs, "runs": runs, "material": material, "present": present, "real": real, "decreed": decreed, "probes": probes})
	}
	// Independently supplied helper inputs exercise false map entries and gaps
	// without relying on the record classifiers to generate every combination.
	decisions := []any{}
	for mask := 0; mask < 256; mask++ {
		runs := map[int]int{}
		material := map[int]bool{}
		present := map[int]bool{}
		for r := 0; r < 8; r++ {
			runs[r] = (mask >> uint(r)) % 4
			material[r] = (mask+r)%5 == 0
			present[r] = mask&(1<<uint(r)) != 0
		}
		real, decreed := contiguousRounds(runs, material, present)
		decisions = append(decisions, map[string]any{"runs": runs, "material": material, "present": present, "real": real, "decreed": decreed})
	}
	raw, e := json.Marshal(map[string]any{"cases": rows, "decisions": decisions, "constants": []int{statMinRoundRun, statMinRoundRecords, statMinRoundRecordShare, statMaxRound, statMaxEmptyRoundRun}})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-round-helpers.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
