package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"reflect"
	"sort"
	"testing"
)

func TestHaloRustTranslocatorSource(t *testing.T) {
	rows := []any{}
	orders := [][]int{{1, 2, 3}, {0, 1, 2}, {1, 3, 2}, {2, 1}, {1, 1, 2}, {-1, 1, 2}, {1}, {0}}
	for run := 0; run < 128; run++ {
		raw := source.MemoryChunks{}
		meta := []types.ChunkMeta{}
		inputs := []any{}
		for chunk, index := range orders[run%len(orders)] {
			data := []byte{}
			packet := func(kind uint16, ts uint64, pay []byte) []byte {
				p := make([]byte, 16)
				binary.LittleEndian.PutUint16(p, kind)
				binary.LittleEndian.PutUint32(p[4:], uint32(len(pay)))
				binary.LittleEndian.PutUint64(p[8:], ts)
				return append(p, pay...)
			}
			for j := 0; j < 24; j++ {
				w := translocWriter{}
				w.put(1, 1)
				w.put(1, 1)
				w.put(7, 117)
				w.put(1, 1)
				w.put(8, uint64(chunk*24+j))
				w.put(2, uint64(j%4))
				w.put(2, 0)
				w.put(1, 0)
				for k := 0; k < 2; k++ {
					w.put(1, 1)
					for axis := 0; axis < 3; axis++ {
						w.put(22, uint64(200+j*31+axis+k))
					}
				}
				pay := w.pay
				switch j % 6 {
				case 0:
					pay = pay[:1]
				case 1:
					pay = pay[:2]
				case 2:
					pay[0] = 0xba
				case 3:
					pay[1] &= 0xbf
				}
				kind := uint16(0)
				if j%11 == 0 {
					kind = 6
				}
				if j == 12 {
					switch (run / 8) % 4 {
					case 1:
						kind = 7
					case 2:
						pay = nil
					case 3:
						p := packet(0, 3, pay)
						binary.LittleEndian.PutUint32(p[4:], 99999)
						data = append(data, p...)
						continue
					}
				}
				data = append(data, packet(kind, uint64((j%3)*1000), pay)...)
			}
			data = append(data, 1, 2, 3)
			raw = append(raw, data)
			meta = append(meta, types.ChunkMeta{Index: index})
			inputs = append(inputs, map[string]any{"index": index, "hex": hex.EncodeToString(data)})
		}
		film, err := source.Load(raw, meta)
		if err != nil {
			t.Fatal(err)
		}
		expected := ScanTranslocatorTeleports(film, nil)
		type item struct {
			Event  types.TranslocatorTeleport `json:"event"`
			Chunk  int                        `json:"chunk"`
			Packet int                        `json:"packet"`
			Start  int                        `json:"start"`
			Size   int                        `json:"size"`
		}
		observed := []item{}
		for _, c := range FilmChunkNumbers(film) {
			d, pks, ok := FilmChunkAt(film, c)
			if !ok {
				continue
			}
			for pi, pk := range pks {
				if pk.Type != PacketTypeDelta || pk.Size < 2 {
					continue
				}
				pay := pk.Payload(d)
				if pay[0] != translocFamilyByte {
					continue
				}
				if ev, ok := decodeTranslocHead(pay, pk.TimestampUS, nil); ok {
					observed = append(observed, item{ev, c, pi, pk.Start, pk.Size})
				}
			}
		}
		sort.Slice(observed, func(i, j int) bool { return observed[i].Event.TimestampUS < observed[j].Event.TimestampUS })
		var projected []types.TranslocatorTeleport
		for _, r := range observed {
			projected = append(projected, r.Event)
		}
		if !reflect.DeepEqual(projected, expected) {
			t.Fatal("provenance wrapper differs")
		}
		rows = append(rows, map[string]any{"inputs": inputs, "output": observed})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-translocator-source.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
