package replay

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/fallback"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives"
	"levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
	"math/rand"
	"os"
	"sort"
	"testing"
)

func TestHaloRustReplayFlagGeometry(t *testing.T) {
	rng := rand.New(rand.NewSource(41150))
	rows := []any{}
	for run := 0; run < 1024; run++ {
		tracks := []Track{}
		bridge := map[uint32]uint64{}
		ambiguous := map[uint32]bool{}
		for slot := 0; slot < 8; slot++ {
			if rng.Intn(4) != 0 {
				bridge[uint32(slot)] = uint64(100 + rng.Intn(4))
			}
			if rng.Intn(7) == 0 {
				ambiguous[uint32(slot)] = true
			}
		}
		for j := 0; j < 16; j++ {
			slot := uint32(j % 8)
			xuid := []string{"", "100", "101", "102", "103"}[rng.Intn(5)]
			pts := []Point{}
			for k := 0; k < 30; k++ {
				pts = append(pts, Point{T: k*3 + rng.Intn(3), X: float32(k % 5), Y: float32(j % 3), Z: float32(j)})
			}
			if j%7 == 0 {
				pts = []Point{}
			}
			bot := ""
			if j%6 == 0 {
				bot = "bot"
			}
			tracks = append(tracks, Track{Slot: slot, XUID: xuid, Bot: bot, Points: pts, StartFrame: j, EndFrame: 100 - j})
		}
		fb := fallback.NouveauCompteur()
		idx, amb := tracksByXUID(tracks, bridge, ambiguous, fb)
		if amb == nil {
			amb = []uint32{}
		}
		noBridge := fb.Compte(fallback.NomPisteDrapeauSansPontEcartee)
		probes := []any{}
		for j := 0; j < 20; j++ {
			x := []string{"100", "101", "102", "103", "missing"}[j%5]
			frame := rng.Intn(120) - 10
			p, ok := pointOfXUIDAt(idx[x], frame)
			var point any
			if ok {
				point = p
			}
			probes = append(probes, map[string]any{"xuid": x, "frame": frame, "point": point})
		}
		c := matchClock{origin: uint64(run%3) * 1000000, step: []uint64{100000, 16667, 1000000}[run%3], frames: []int{0, 30, 100, 1000}[run%4], deathOffsetMS: int64(run%5-2) * 500}
		ctx := flagCarryCtx{matchClock: c, tracks: tracks, slotXUID: bridge, slotAmbiguous: ambiguous, fb: fallback.NouveauCompteur()}
		raws := []flagCarryRaw{}
		for j := 0; j < 30; j++ {
			a := int64(rng.Intn(8000) - 1000)
			raws = append(raws, flagCarryRaw{xuid: []string{"100", "101", "102", "103", "missing"}[j%5], t0: a, t1: a + int64(100+rng.Intn(6000)), closed: j%2 == 0, captured: j%5 == 0, homed: j%7 == 0, flagIndex: -1, x1: float32(j % 5), y1: float32(j % 3)})
		}
		input := haloFlagRaw(raws)
		spawns := []FlagSpawn{}
		if run%3 != 0 {
			spawns = append(spawns, FlagSpawn{Team: 0, X: 0, Y: 0}, FlagSpawn{Team: 1, X: 4, Y: 2})
		}
		lives := []flagFreeLife{}
		for j := 0; j < 60; j++ {
			ts := uint64(rng.Intn(100)) * 100000
			pts := []flagFreeSample{{TUS: ts, X: float32(rng.Intn(5)), Y: float32(rng.Intn(3))}, {TUS: ts + 100000, X: float32(rng.Intn(5)) + 0.25, Y: float32(rng.Intn(3))}}
			if j%13 == 0 {
				pts = []flagFreeSample{}
			}
			lives = append(lives, flagFreeLife{ID: uint32(j), T0US: ts, T1US: ts + 100000, Pts: pts})
		}
		sort.SliceStable(lives, func(i, j int) bool { return lives[i].T0US < lives[j].T0US })
		scan := FlagCarryScan{Spawns: spawns, Free: lives}
		drops := []any{}
		matches := []any{}
		for _, r := range raws {
			at, ok := flagFreeDropInside(r, ctx, idx[r.xuid], scan)
			var a any
			if ok {
				a = at
			}
			drops = append(drops, a)
			l, ok := flagFreeAtDrop(r, ctx, scan)
			var id any
			if ok {
				id = l.ID
			}
			matches = append(matches, id)
		}
		raws = closeByFreeLives(raws, ctx, scan)
		closed := haloFlagRaw(raws)
		cov := &FlagCarriesCoverage{}
		ctx.fb = fallback.NouveauCompteur()
		raws = attachFlagCarryPositions(raws, ctx, cov)
		attached := haloFlagRaw(raws)
		positionCov := map[string]any{"ambiguous_slot": cov.AmbiguousSlot, "out_of_window": cov.OutOfWindow, "no_track": cov.NoTrack, "drop_uses_pickup": ctx.fb.Compte(fallback.NomPositionLacherPrendLaPrise), "without_bridge": ctx.fb.Compte(fallback.NomPisteDrapeauSansPontEcartee)}
		moved := repositionFlagDrops(raws, ctx, scan)
		assigned := []flagCarryRaw{}
		for j := 0; j < 40; j++ {
			a := int64(rng.Intn(40)) * 250
			assigned = append(assigned, flagCarryRaw{xuid: []string{"100", "101", "102", "103"}[j%4], t0: a, t1: a + int64(rng.Intn(20))*250, closed: j%3 != 0, captured: j%7 == 0, homed: j%11 == 0, steal: j%4 == 0, flagIndex: -1, x0: float32(rng.Intn(20) - 10), y0: float32(rng.Intn(20) - 10), x1: float32(rng.Intn(20) - 10), y1: float32(rng.Intn(20) - 10)})
		}
		assignmentInput := haloFlagRaw(assigned)
		assignScan := scan
		assignScan.Spawns = []FlagSpawn{}
		for j := 0; j < run%7; j++ {
			assignScan.Spawns = append(assignScan.Spawns, FlagSpawn{Team: j%3 - 1, X: float32(j%3) * 4, Y: float32(j/3) * 4})
		}
		assignScan.TeamOf = map[string]int{}
		for _, x := range []string{"100", "101", "102", "103"} {
			if rng.Intn(4) != 0 {
				assignScan.TeamOf[x] = rng.Intn(4) - 1
			}
		}
		assignScan.Events = []objectives.NamedEvent{}
		for j := 0; j < 15; j++ {
			assignScan.Events = append(assignScan.Events, objectives.NamedEvent{TimeMS: rng.Intn(40) * 250, Stat: objectives.StatFlagReturns})
		}
		acov := &FlagCarriesCoverage{}
		assignFlags(assigned, assignScan, ctx, acov)
		beforeMarks := haloFlagRaw(assigned)
		marks := grammar.CarrierMarkScan{Marks: []grammar.CarrierMark{}, KeyframeUS: []uint64{}}
		for j := 0; j < 30; j++ {
			ts := uint64(rng.Intn(50)) * 250000
			marks.KeyframeUS = append(marks.KeyframeUS, ts)
			marks.Marks = append(marks.Marks, grammar.CarrierMark{Slot: uint32(rng.Intn(10)), TimestampUS: ts})
		}
		markFlagCarries(assigned, marks, ctx)
		tc := &FlagCarriesCoverage{}
		tallyFlagCarries(assigned, tc)
		tc.Overlaps, tc.ClosedOverlaps = countFlagOverlaps(assigned)
		lifeCov := &FlagCarriesCoverage{}
		flagLives := assembleFlagLives(assigned, assignScan, ctx, lifeCov)
		if flagLives == nil {
			flagLives = []FlagCarry{}
		}
		returnRows := []any{}
		for _, h := range flagReturns(assignScan) {
			returnRows = append(returnRows, map[string]any{"flag": h.flag, "at": h.at, "x": h.x, "y": h.y})
		}
		fullScan := assignScan
		fullScan.Scanned = run%17 != 0
		fullScan.Marks = marks
		fullScan.Events = []objectives.NamedEvent{}
		fullIdentity := map[int]string{0: "100", 1: "101", 2: "102", 3: "103"}
		if run%5 == 0 {
			delete(fullIdentity, 2)
		}
		fullScan.Identity = objectives.FlatRoundIdentity(fullIdentity)
		for j := 0; j < 60; j++ {
			fullScan.Events = append(fullScan.Events, objectives.NamedEvent{Slot: rng.Intn(5), TimeMS: rng.Intn(40) * 250, Stat: []string{"flag_grabs", "flag_steals", "flag_captures", "flag_returns", "flag_carriers_killed"}[rng.Intn(5)], Side: "A"})
		}
		bursts := make([]int, run%30)
		fullScan.Signals = objectives.FlagFilmSignalsFrom(bursts, fullScan.Events)
		fullCtx := ctx
		fullCtx.fb = fallback.NouveauCompteur()
		fullCarries, fullCov := buildFlagCarries(fullScan, fullCtx)
		if fullCarries == nil {
			fullCarries = []FlagCarry{}
		}
		emptyScan := fullScan
		emptyScan.Free = nil
		emptyCtx := fullCtx
		emptyCtx.fb = fallback.NouveauCompteur()
		emptyCarries, emptyCov := buildFlagCarries(emptyScan, emptyCtx)
		if emptyCarries == nil {
			emptyCarries = []FlagCarry{}
		}
		rows = append(rows, map[string]any{"empty_free_carries": emptyCarries, "empty_free_cov": emptyCov, "empty_free_no_bridge": emptyCtx.fb.Compte(fallback.NomPisteDrapeauSansPontEcartee), "empty_free_drop_pickup": emptyCtx.fb.Compte(fallback.NomPositionLacherPrendLaPrise), "full_events": fullScan.Events, "full_identity": fullIdentity, "full_scanned": fullScan.Scanned, "full_signals": fullScan.Signals, "full_carries": fullCarries, "full_cov": fullCov, "full_no_bridge": fullCtx.fb.Compte(fallback.NomPisteDrapeauSansPontEcartee), "full_drop_pickup": fullCtx.fb.Compte(fallback.NomPositionLacherPrendLaPrise), "return_rows": returnRows, "flag_lives": map[string]any{"carries": flagLives, "coverage": lifeCov}, "assignment_input": assignmentInput, "assignment_spawns": assignScan.Spawns, "assignment_teams": assignScan.TeamOf, "return_times": flagReturnTimes(assignScan), "assigned": beforeMarks, "marks": marks, "marked": haloFlagRaw(assigned), "tally": tc, "assignment_cov": map[string]any{"own_flag_refused": acov.OwnFlagRefused, "unresolved": acov.Unresolved, "assigned_by_play": acov.AssignedByPlay}, "tracks": tracks, "bridge": bridge, "ambiguous": ambiguous, "index": map[string]any{"tracks": idx, "ambiguous_slots": amb, "without_bridge": noBridge}, "probes": probes, "clock": map[string]any{"origin_us": c.origin, "step_us": c.step, "frames": c.frames, "death_offset_ms": c.deathOffsetMS}, "raw": input, "spawns": spawns, "lives": lives, "drops": drops, "matches": matches, "closed": closed, "attached": attached, "position_cov": positionCov, "moved": moved, "repositioned": haloFlagRaw(raws)})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-replay-flag-geometry.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
