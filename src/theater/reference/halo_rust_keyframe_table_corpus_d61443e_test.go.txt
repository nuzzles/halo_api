package grammar

import (
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
)

func TestHaloRustKeyframeTableCorpusD61443e(t *testing.T) {
 root := "/Users/simbleau/git/halo_api/experiments/films"
 rows := []any{}
 films := 0
 err := filepath.WalkDir(root, func(path string, entry os.DirEntry, err error) error {
  if err != nil { return err }
  if entry.IsDir() || entry.Name() != "film.json" { return nil }
  manifest, err := os.ReadFile(path); if err != nil { return err }
  var meta struct { Chunks []struct { File string `json:"file"`; Kind int `json:"chunk_type"` } }
  if err = json.Unmarshal(manifest, &meta); err != nil { return err }
  folder := filepath.Dir(path)
  var bootstrap []byte
  for _, c := range meta.Chunks { if c.Kind == 1 { bootstrap, err = os.ReadFile(filepath.Join(folder,c.File)); if err != nil { return err }; break } }
  reg, err := ParseRegistryChunk(bootstrap); if err != nil { return err }
  identity, err := ReadFilmIdentity(bootstrap); if err != nil { return err }
  films++
  ctx := ContexteParDefaut(); ctx.Profil.Grammaire.ControleDeCorruption = identity.ControleDeCorruption
  for _, c := range meta.Chunks {
   if c.Kind != 2 { continue }
   filename := filepath.Join(folder,c.File)
   data, err := os.ReadFile(filename); if err != nil { return err }
   rel, err := filepath.Rel(root, filename); if err != nil { return err }
   for pi, pk := range WalkPackets(data) {
    if pk.Type != PacketTypeKeyframe { continue }
    records, stop := WalkKeyframeRecords(pk.Payload(data),reg,ctx)
    rows = append(rows,map[string]any{"file":rel,"packet":pi,"offset":pk.Start,"size":pk.Size,"check":identity.ControleDeCorruption,"records":records,"stop":stop.String()})
   }
  }
  return nil
 })
 if err != nil { t.Fatal(err) }
 if films != 32 || len(rows) != 451 { t.Fatalf("corpus: %d films, %d keyframes",films,len(rows)) }
 raw, err := json.Marshal(rows); if err != nil { t.Fatal(err) }
 if err = os.WriteFile("/private/tmp/halo-keyframe-table-corpus-d61443e.json",raw,0600); err != nil { t.Fatal(err) }
}
