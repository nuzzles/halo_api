package replay
import("testing";"os";"encoding/json";"math/rand")
func TestHaloRustReplayBounds(t *testing.T){
 rng:=rand.New(rand.NewSource(41773));rows:=[]any{}
 for run:=0;run<1024;run++{
  tracks:=[]Track{};n:=[]int{0,1,199,200,201,400,1000}[run%7]
  for k:=0;k<1+run%4;k++{tr:=Track{Slot:uint32(k),Points:[]Point{}};for i:=0;i<n;i++{p:=Point{X:float32(rng.Intn(10000)-5000)/100,Y:float32(rng.Intn(10000)-5000)/100,Z:float32(rng.Intn(1000)-500)/100};if run%3==0{p.Z=0};if run%5==0{p.X=1;p.Y=1;p.Z=1};if i==0&&run%2==0{p.X=100000};if i==1&&run%4==0{p.Y=-100000};if i==2&&run%8==0{p.Z=100000};tr.Points=append(tr.Points,p)};tracks=append(tracks,tr)}
  b,r:=boundsOf(tracks);rows=append(rows,map[string]any{"tracks":tracks,"bounds":b,"rejected":r})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-bounds.json",raw,0600);e!=nil{t.Fatal(e)}
}
