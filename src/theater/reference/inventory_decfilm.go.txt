package main

import (
 "bytes"
 "encoding/json"
 "go/ast"
 "go/parser"
 "go/printer"
 "go/token"
 "os"
)

func main() {
 fs := token.NewFileSet()
 file, err := parser.ParseFile(fs, os.Args[len(os.Args)-1], nil, 0)
 if err != nil { panic(err) }
 rows := []any{}
 show := func(n ast.Node) string {
  var b bytes.Buffer
  if err := printer.Fprint(&b, fs, n); err != nil { panic(err) }
  return b.String()
 }
 for _, decl := range file.Decls {
  switch d := decl.(type) {
  case *ast.FuncDecl:
   if ast.IsExported(d.Name.Name) {
    target := ""
    unchanged := false
    if len(d.Body.List)==1 {
     if ret,ok:=d.Body.List[0].(*ast.ReturnStmt);ok && len(ret.Results)==1 {
      if call,ok:=ret.Results[0].(*ast.CallExpr);ok {
       target=show(call.Fun)
       params:=[]string{}
       for _,field:=range d.Type.Params.List {for _,name:=range field.Names {params=append(params,name.Name)}}
       unchanged=len(params)==len(call.Args)
       for i,arg:=range call.Args {id,ok:=arg.(*ast.Ident);if !ok||i>=len(params)||id.Name!=params[i] {unchanged=false}}
      }
     }
    }
    rows = append(rows, map[string]any{"name":d.Name.Name,"kind":"func","line":fs.Position(d.Pos()).Line,"signature":show(d.Type),"body":show(d.Body),"target":target,"single_return_call":target!="","arguments_forwarded_unchanged":unchanged})
   }
  case *ast.GenDecl:
   for _, spec := range d.Specs {
    switch s := spec.(type) {
    case *ast.TypeSpec:
     if ast.IsExported(s.Name.Name) {
      rows = append(rows,map[string]any{"name":s.Name.Name,"kind":"type","line":fs.Position(s.Pos()).Line,"target":show(s.Type),"alias":s.Assign.IsValid()})
     }
    case *ast.ValueSpec:
     for i,name := range s.Names {
      if ast.IsExported(name.Name) {
       target := ""
       if i<len(s.Values) { target=show(s.Values[i]) }
       rows=append(rows,map[string]any{"name":name.Name,"kind":d.Tok.String(),"line":fs.Position(s.Pos()).Line,"target":target})
      }
     }
    }
   }
  }
 }
 b,err := json.MarshalIndent(rows,"","  ")
 if err!=nil {panic(err)}
 os.Stdout.Write(b)
}
