package replay

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"log/slog"
 "levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
 "levelup/go-api/internal/games/halo_infinite/film/internal/facts/fallback"
	"os"
	"testing"
)

func TestHaloRustFactsLiveObjectives(t *testing.T) {
	old := slog.Default()
	defer slog.SetDefault(old)
	rows := []any{}
	for i := 0; i < 128; i++ {
		recs := []types.StatRecord{}
		deaths := []Death{}
		for r := 0; r < 2; r++ {
			for j := 0; j <= 30; j++ {
				at := r*60000 + j*1000
				if i%3 == 0 {
					at = (2-r)*60000 + j*1000
				}
				comps := map[int]types.StatValue{0: {A: int64(j)}}
				for _, c := range []int{1, 2, 3, 19, 20, 21, 22, 23, 24, 25} {
					v := int64(j)
					if i%3 == 1 {
						v *= 17
					}
					comps[c] = types.StatValue{A: v, B: v, C: v, D: v, HasC: true, HasD: true}
				}
				recs = append(recs, types.StatRecord{Slot: 10, Round: r, TimeMS: at, Comps: comps})
				if j > 0 {
					deaths = append(deaths, Death{XUID: 100, TimeMS: int64(at)})
				}
			}
		}
		bursts := []int{}
		for b := 0; b < i%65; b++ {
			bursts = append(bursts, b*1000)
		}

        h := &rustScoreSeriesHandler{records: []any{}}
        slog.SetDefault(slog.New(h))
        clock := replayClock{origin: uint64(i%3)*1000000, step: 100000, frames: 2000, fb: fallback.NouveauCompteur()}
        reg := IdentityRegistry{own: OwnerReport{DeathOffsetMS: int64(i%5-2)*1000}}
        tracks:=[]Track{{Slot:10,XUID:"100",StartFrame:0,EndFrame:1999}}
        for j:=0;j<2000;j++ { tracks[0].Points=append(tracks[0].Points,Point{T:j,X:float32(j%9),Y:0}) }
        identity:=objectives.RoundIdentity{}
        identityMode:=(i/3)%3
        if identityMode==1 {identity=objectives.FlatRoundIdentity(map[int]string{10:"100"})}
        if identityMode==2 {identity=objectives.FlatRoundIdentity(map[int]string{})}
        gauges:=[]grammar.ManagedPropertyRead{}
        for j:=0;j<200;j++ { gauges=append(gauges,grammar.ManagedPropertyRead{Slot:50,TimestampUS:uint64(j)*1000000,Field:grammar.ManagedPropertyScalar,Tag:3,Value:uint64(8388607+j*100),HasValue:true}) }
        in:=FlagInput{Scanned:i%7!=0,Records:recs,Bursts:bursts,Identity:identity,Spawns:[]FlagSpawn{{Team:0,X:0,Y:0},{Team:1,X:20,Y:0}},Gauge:gauges,GaugeScanned:i%4!=0}

        reg.own.SlotXUID=map[uint32]uint64{10:100}
        reads:=[]types.NavpointRadialRead{{Slot:50,TMS:0,Q:130},{Slot:50,TMS:1000,Q:180},{Slot:50,TMS:2000,Q:254}}
        changes:=[]types.HeldWeaponChange{{Slot:10,TimestampUS:1000000,Family:bombHeldFamily},{Slot:10,TimestampUS:2000000,Previous:bombHeldFamily}}
        opt:=Options{Flag:in,Vip:VipInput{Scanned:i%3!=0,Records:recs},Skull:SkullInput{Scanned:i%4!=0,Records:recs,Identity:identity},Bomb:BombInput{CarryScanned:i%5!=0,Scanned:i%6!=0,Reads:reads},WeaponChanges:changes,Deaths:deaths,Fallbacks:clock.fb,FilmClockOriginUS:1234567}
        if i%2==0{opt.Score=&ScoreInput{}}
        opt.Zone.Scanned=i%2!=0
        opt.Zone.Reads=gauges
        opt.Labels.FlagReturnZone=FlagReturnZone{RadiusM:1.3,ResetSeconds:30,SoloSeconds:3.1}
        opt.Objectives=[]objectives.IdentifiedEvent{{NamedEvent:objectives.NamedEvent{TimeMS:7000,Stat:"bomb_detonations",Side:"A"},XUID:"100"}}
        doc:=ReplayDocument{Tracks:tracks,FrameCount:clock.frames,FrameIntervalMS:100,Coverage:&Coverage{Equipment:&EquipmentCoverage{}},Objectives:[]ObjectiveAction{{TimeMS:7000,Stat:"bomb_detonations"}},FlagReturnZone:&FlagReturnZone{RadiusM:99}}
        before,_:=json.Marshal(doc)
        a:=assemblage{doc:doc,opt:opt,reg:reg,origin:clock.origin,step:clock.step,clock:scoreClock{originMS:0,intervalMS:100,frames:clock.frames},equipes:teamPublication{byXUID:map[uint64]int{100:0}}}
        a.poserObjectifsVivants()
        rows=append(rows,map[string]any{"records":recs,"deaths":deaths,"bursts":bursts,"identity_mode":identityMode,"spawns":in.Spawns,"gauges":gauges,"scanned":in.Scanned,"gauge_scanned":in.GaugeScanned,"vip_scanned":opt.Vip.Scanned,"skull_scanned":opt.Skull.Scanned,"carry_scanned":opt.Bomb.CarryScanned,"arming_scanned":opt.Bomb.Scanned,"score_read":opt.Score!=nil,"zone_scanned":opt.Zone.Scanned,"bomb_reads":reads,"changes":changes,"identified":opt.Objectives,"clock":map[string]any{"origin_us":clock.origin,"step_us":clock.step,"frames":clock.frames,"death_offset_ms":reg.DeathOffsetMS()},"before":json.RawMessage(before),"after":a.doc,"fallbacks":clock.fb.Rapport(),"logs":h.records})

	}
	data, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-facts-live-objectives.json", data, 0600); e != nil {
		t.Fatal(e)
	}
}
