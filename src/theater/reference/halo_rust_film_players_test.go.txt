func TestHaloRustFilmPlayers(t *testing.T) {
	haloRustFilmAssembly(t, []struct{ Folder, Map string }{{"maps/01-bazaar-idle", "bazaar"}, {"maps/02-aquarius", "aquarius"}, {"bandit/01-evo", "recharge"}, {"ranked-arena/02-oddball", "recharge"}, {"appearance/01-cadet-blue", "high ground"}, {"appearance/02-cadet-brick", "high ground"}}, "/private/tmp/halo-film-players.json")
}

// Facility Aetheria metadata identifies fo11_blank; all catalog entries for that
// module share these bounds. Corpo supplies the quantization entry only.
func TestHaloRustFilmRaid(t *testing.T) {
	haloRustFilmAssembly(t, []struct{ Folder, Map string }{{"raids/01-hour-long-raid", "corpo"}}, "/private/tmp/halo-film-raid.json")
}

func haloRustFilmAssembly(t *testing.T, samples []struct{ Folder, Map string }, output string) {
	catalog, err := profile.LoadMapQuantCatalog("/private/tmp/halo-levelup-v75/data/titles/halo_infinite/reference/map_quant_bounds.json")
	if err != nil {
		t.Fatal(err)
	}
	var abilityCatalog struct {
		Palettes        []AbilityPalette
		ImpulseFamilies []string `json:"impulse_families"`
		ChargeFamilies  []string `json:"charge_families"`
	}
	catalogBytes, err := os.ReadFile("/private/tmp/halo-replay-ability-catalog.json")
	if err != nil {
		t.Fatal(err)
	}
	if err = json.Unmarshal(catalogBytes, &abilityCatalog); err != nil {
		t.Fatal(err)
	}
	var equipmentCatalog struct { Families map[uint32]string; WeaponKeys map[uint32]string `json:"weapon_keys"`; ObjectiveLabels map[uint32]Label `json:"objective_labels"`; ObjectiveFamilies map[uint32]string `json:"objective_families"` }
	equipmentBytes, err := os.ReadFile("/private/tmp/halo-replay-equipment-catalog.json")
	if err != nil { t.Fatal(err) }
	if err = json.Unmarshal(equipmentBytes, &equipmentCatalog); err != nil { t.Fatal(err) }
	rows := []any{}
	for _, sample := range samples {
		root := filepath.Join("/Users/simbleau/git/halo_api/experiments/films", sample.Folder)
		manifest, err := os.ReadFile(filepath.Join(root, "film.json"))
		if err != nil {
			t.Fatal(err)
		}
		var meta struct {
			Chunks []struct {
				Index int    `json:"index"`
				Kind  int    `json:"chunk_type"`
				Start int    `json:"start_time_offset_ms"`
				File  string `json:"file"`
			}
		}
		if err = json.Unmarshal(manifest, &meta); err != nil {
			t.Fatal(err)
		}
		var chunks source.MemoryChunks
		var metadata []types.ChunkMeta
		for _, c := range meta.Chunks {
			d, e := os.ReadFile(filepath.Join(root, c.File))
			if e != nil {
				t.Fatal(e)
			}
			chunks = append(chunks, d)
			metadata = append(metadata, types.ChunkMeta{Index: c.Index, ChunkType: c.Kind, StartMS: c.Start})
		}
		film, err := source.Load(chunks, metadata)
		if err != nil {
			t.Fatal(err)
		}
		entry, err := catalog.Lookup(sample.Map)
		if err != nil {
			t.Fatal(err)
		}
		opt := Options{MapQuant: &entry}
		fc := grammar.NewFilmContextForMap(film, &entry, nil)
		installWorldObjectPrecision(fc, sample.Folder, opt.Fallbacks)
		in, err := scanFilmInputs(sample.Folder, film, fc, opt)
		if err != nil {
			t.Fatal(err)
		}
		in.applyTo(&opt)
		records, truncated := objectives.StatRecordsCtx(context.Background(), film, "")
		instants := []types.DeathInstant{}
		for _, d := range in.Deaths {
			instants = append(instants, types.DeathInstant{XUID: strconv.FormatUint(d.XUID, 10), TimeMS: int(d.TimeMS)})
		}
		opt.StatborgIdentity = objectives.ResolveRoundIdentity(records, instants).CompletedByLines(records, nil).CompletedByElimination(records, nil).CompletedByRoundResidue(records, nil)
		opt.Score = &ScoreInput{Records: records, Truncated: truncated}
		a := assemblage{matchID: sample.Folder, pos: in.Positions, fire: in.Fire, opt: opt}
		if !a.ouvrir("halo-infinite") {
		rows = append(rows, map[string]any{"folder": sample.Folder, "map": entry, "empty": true})
			continue
		}
		a.poserLesPistes()
		a.poserLesEquipesEtLeRoster()
		vclock := replayClock{origin: a.origin, step: a.step, frames: a.doc.FrameCount, fb: fallback.NouveauCompteur()}
		vtracks, vcov, vstats := buildVehicleTracks(in.Vehicles, a.sorted, a.reg, vclock)
		vcycles := buildVehicleCycles(vtracks, a.step, &vcov)
		if vtracks == nil {
			vtracks = []VehicleTrack{}
		}
		if vcycles == nil {
			vcycles = []VehicleCycle{}
		}
		vfallbacks := map[string]int{}
		for _, f := range vclock.fb.Rapport() {
			vfallbacks[string(f.Nom)] = f.Declenchements
		}
		vehicles := map[string]any{"tracks": vtracks, "cycles": vcycles, "coverage": vcov, "fallbacks": vfallbacks, "rides": map[string]any{"episodes": vstats.episodes, "named": vstats.nommes, "lost": vstats.perdus, "by_event": vstats.parEvenement, "by_nearest_event": vstats.parEvenementProche, "by_geometry": vstats.parGeometrie, "gap_fallbacks": vstats.repli, "seats": vstats.sieges, "rejected": vstats.ecartes, "borders_two": vstats.bornes2, "borders_one": vstats.bornes1, "borders_zero": vstats.bornes0, "film": map[string]any{"reads": vstats.film.lectures, "named": vstats.film.nommees, "outside_life": vstats.film.horsVie, "not_drawable": vstats.film.nonDessinable, "published": vstats.film.publies, "closed_by_read": vstats.film.fermetureParLecture, "closed_by_return": vstats.film.fermetureParRetour, "closed_by_life": vstats.film.fermetureParVie}}}
		shots, orphans, shotCov := buildShots(a.sorted, a.fire, a.origin, a.step, a.reg.IndexParSlot())
		before := len(shots)
		shots = keepShotsOfPublishedTracks(shots, a.doc.Tracks)
		shotCov.Unpublished = countUnpublished(before, len(shots))
		shotCov.Attached = len(shots)
		if shots == nil {
			shots = []Shot{}
		}
		orphanRows := []any{}
		for _, o := range orphans {
			reason := "NoSlot"
			if o.reason == reasonOutOfWindow {
				reason = "OutOfWindow"
			}
			orphanRows = append(orphanRows, map[string]any{"event": o.ev, "reason": reason})
		}
		projectiles, pubByRaw, cuts := buildProjectiles(in.Projectiles, a.origin, a.step)
		if projectiles == nil {
			projectiles = []Projectile{}
		}
		if pubByRaw == nil {
			pubByRaw = map[int]int{}
		}
		grenades, grenadeCov := buildGrenades(a.sorted, in.Grenades, a.origin, a.step, a.reg.IndexParSlot(), in.Projectiles, pubByRaw)
		beforeGrenades := len(grenades)
		grenades = keepGrenadesOfPublishedTracks(grenades, a.doc.Tracks)
		grenadeCov.Unpublished = countUnpublished(beforeGrenades, len(grenades))
		grenadeCov.Attached = len(grenades)
		if grenades == nil {
			grenades = []Grenade{}
		}
		loadouts := keepLoadoutsOfPublishedTracks(buildLoadouts(in.Loadouts, a.origin, a.step), a.doc.Tracks)
		if loadouts == nil {
			loadouts = []Loadout{}
		}
		builtReads := buildGrenadeReads(in.Inventory, in.InventoryDeltas, a.origin, a.step)
		beforeReads := append([]GrenadeRead{}, builtReads...)
		reads := keepGrenadeReadsOfPublishedTracks(builtReads, a.doc.Tracks)
		if reads == nil {
			reads = []GrenadeRead{}
		}
		readDoc := ReplayDocument{Coverage: &Coverage{}, GrenadeReads: reads}
		attachGrenadeReadCoverage(&readDoc, beforeReads, in.InventoryDeltaAmmoRefused)
		combatVehicleCoverage := vcov
		combatDoc := ReplayDocument{Vehicles: vtracks, Tracks: a.doc.Tracks, Shots: append([]Shot{}, shots...), Coverage: &Coverage{Shots: shotCov, Vehicles: &combatVehicleCoverage, Verdict: map[string]string{}}}
		attachVehicleShots(&combatDoc, orphans, a.reg, vclock)
		var vehicleShotVerdict any
		if v, ok := combatDoc.Coverage.Verdict["shots"]; ok {
			vehicleShotVerdict = v
		}
		combatShots := map[string]any{"shots": combatDoc.Shots, "orphans": orphanRows, "coverage": combatDoc.Coverage.Shots}
		placementClock := vclock
		placementClock.families = equipmentCatalog.Families
		placements, placementCov := buildEquipmentPlacements(equipmentInputs{Raw:a.opt.Placements,Stats:a.opt.PlacementStats,Positions:a.sorted,Census:a.opt.Pads.Powerups.Keyframes,Spawns:a.opt.SpawnEvents,SpawnStats:a.opt.SpawnStats,Lives:a.reg.Vies(),Changes:a.opt.EquipmentChanges},placementClock)
		if placements == nil { placements = []EquipmentPlacement{} }
		placementLayer := map[string]any{"placements":placements,"coverage":placementCov,"manifest_fallbacks":placementCov.ByCause[CausePoseManifeste]}
		weaponChanges, weaponChangeCov := buildWeaponChanges(a.opt.WeaponChanges, a.origin, a.step)
		if weaponChanges == nil { weaponChanges = []WeaponChange{} }
		weaponChangeLayer := map[string]any{"changes":weaponChanges,"coverage":weaponChangeCov}
		pickupJudge := newPickupOriginJudge(a.opt, a.pos, placements)
		pickups, pickupCov := buildPickups(a.opt.Pickups, placementClock, pickupInputs{occupant:a.reg.XUIDNumAt,st:a.opt.PickupStats,weaponKeys:equipmentCatalog.WeaponKeys,judge:pickupJudge})
		if pickups == nil { pickups = []Pickup{} }
		pickupLayer := map[string]any{"pickups":pickups,"coverage":pickupCov}

		closedByDeath := a.reg.TracesCloturesParMort(a.doc.Tracks, a.origin, a.step)
		episodes, nonBinary := buildEquipmentEpisodes(a.sorted, in.CamoStates, a.origin, a.step, a.doc.Tracks, closedByDeath)
		if episodes == nil { episodes = []EquipmentEpisode{} }
		episodeLayer := map[string]any{"episodes": episodes, "non_binary": nonBinary, "coverage": equipmentCoverage(episodes, a.doc.Tracks, closedByDeath)}
		grappleLines, grappleCov := buildGrappleLines(in.GrappleReads, entry, a.origin, a.step, a.doc.Tracks)
		if grappleLines == nil { grappleLines = []GrappleLine{} }
		grappleLayer := map[string]any{"lines": grappleLines, "coverage": grappleCov}
		builtEquipment, equipmentCov := buildEquipmentChanges(in.EquipmentChanges, in.EquipmentChangeStats, a.origin, a.step)
		publishedEquipment := keepEquipmentChangesOfPublishedTracks(builtEquipment, a.doc.Tracks)
		if publishedEquipment == nil { publishedEquipment = []EquipmentChange{} }
		equipmentLayer := map[string]any{"changes": publishedEquipment, "coverage": equipmentCov}
		translocations, translocationCov := buildTranslocations(in.Translocations, a.doc.Tracks, a.origin, a.step)
		if translocations == nil { translocations = []Translocation{} }
		translocationLayer := map[string]any{"jumps": translocations, "coverage": translocationCov}
		builtInventory, droppedInventory := buildInventory(in.Inventory, a.origin, a.step)
		publishedInventory := keepInventoryOfPublishedTracks(builtInventory, a.doc.Tracks)
		inventoryMarked := markInventoryDeadReadings(publishedInventory, in.Deaths, a.reg, vclock)
		var inventoryCoverage any
		if in.Inventory != nil { inventoryCoverage = buildInventoryCoverage(in.Inventory, builtInventory, publishedInventory, droppedInventory) }
		if publishedInventory == nil { publishedInventory = []Inventory{} }
		inventoryLayer := map[string]any{"reads": publishedInventory, "coverage": inventoryCoverage, "marked_dead": inventoryMarked}
		rawAbilities := buildAbilityReads(in.AbilityRanks, in.Inventory, a.origin, a.step)
		cleanAbilities, abilityNoise := rejectAbilityScanNoise(rawAbilities)
		publishedAbilities := keepAbilitiesOfPublishedTracks(cleanAbilities, a.doc.Tracks)
		abilityCoverage := buildAbilityCoverage(rawAbilities, cleanAbilities, publishedAbilities, abilityNoise)
		abilityPalette := classifyAbilityPalette(publishedAbilities, abilityCatalog.Palettes)
		abilityLabels := abilityLabelsUsed(publishedAbilities, abilityPalette)
		if abilityLabels == nil {
			abilityLabels = map[string]Label{}
		}
		var paletteOutput any
		if abilityPalette != nil {
			paletteOutput = map[string]any{"id": abilityPalette.ID, "markers": abilityPalette.Markers, "ranks": abilityPalette.Ranks, "families": abilityPalette.Families}
		}
		if publishedAbilities == nil {
			publishedAbilities = []AbilityRead{}
		}
		abilityImpulses, abilityImpulseCov := buildAbilityImpulses(abilityImpulseInputs{reads: in.AbilityImpulses, stats: in.AbilityImpulseStats, ranks: in.AbilityRanks, lives: a.reg.Vies(), palette: abilityPalette, measured: abilityCatalog.ImpulseFamilies}, a.doc.Tracks, a.origin, a.step)
		abilityCharges, abilityChargeCov := buildAbilityCharges(abilityChargeInputs{reads: in.AbilityCharges, stats: in.AbilityChargeStats, ranks: in.AbilityRanks, lives: a.reg.Vies(), palette: abilityPalette, measured: abilityCatalog.ChargeFamilies}, a.doc.Tracks, a.origin, a.step)
		if abilityImpulses == nil {
			abilityImpulses = []AbilityImpulse{}
		}
		if abilityCharges == nil {
			abilityCharges = []AbilityCharge{}
		}
		var impulseCov, chargeCov any
		if in.AbilityImpulseStats.Scanned {
			impulseCov = abilityImpulseCov
		}
		if in.AbilityChargeStats.Scanned {
			chargeCov = abilityChargeCov
		}
		abilityLayer := map[string]any{"abilities": map[string]any{"reads": publishedAbilities, "coverage": abilityCoverage, "palette": paletteOutput, "labels": abilityLabels}, "impulses": abilityImpulses, "impulse_coverage": impulseCov, "charges": abilityCharges, "charge_coverage": chargeCov}
		objectiveObjects:=equipmentCatalog.ObjectiveLabels
		pads,padPicks,padCov,groundObjects:=buildWeaponPads(in.Pads,a.sorted,placementClock,padCatalogs{ObjectiveObjects:objectiveObjects,EquipmentFamilies:equipmentCatalog.Families})
		dating:=datePadPickups(pads,padPicks,pickups)
		items,itemCov:=buildGroundWeaponItems(groundObjects,a.opt.WeaponChanges,a.sorted,placementClock)
		if pads==nil{pads=[]WeaponPad{}};if padPicks==nil{padPicks=[]PadPickup{}};if items==nil{items=[]GroundWeapon{}}
		groundLayer:=map[string]any{"pads":pads,"pickups":padPicks,"coverage":padCov,"dating":dating,"items":map[string]any{"weapons":items,"coverage":itemCov}}
        chunkClock:=map[int]int{};for _,m:=range metadata{chunkClock[m.Index]=m.StartMS}
        radialScan,err:=grammar.ScanNavpointRadial(fc,chunkClock);if err!=nil{t.Fatal(err)};if radialScan.Reads==nil{radialScan.Reads=[]types.NavpointRadialRead{}}
        carrierMarks,err:=grammar.ScanCarrierMarks(film);if err!=nil{t.Fatal(err)}
		if carrierMarks.Marks==nil{carrierMarks.Marks=[]grammar.CarrierMark{}};if carrierMarks.KeyframeUS==nil{carrierMarks.KeyframeUS=[]uint64{}}
		objectiveLives,objectiveCov:=buildObjectiveObjects(in.Pads.Weapons,equipmentCatalog.ObjectiveLabels,equipmentCatalog.ObjectiveFamilies,vclock)
		if objectiveLives==nil{objectiveLives=[]ObjectiveObjectLife{}}
		objectiveLayer:=map[string]any{"lives":objectiveLives,"coverage":objectiveCov}
		skullCarries,skullCov:=buildSkullCarries(SkullCarryScan{Scanned:sample.Folder=="ranked-arena/02-oddball",Records:records,Identity:opt.StatborgIdentity},matchClock{origin:a.origin,step:a.step,frames:a.doc.FrameCount,deathOffsetMS:a.reg.DeathOffsetMS()},carrierPresenceOf(a.doc.Tracks,a.unnamed.deduced))
        if skullCarries==nil{skullCarries=[]SkullCarry{}}
        skullLayer:=map[string]any{"carries":skullCarries,"coverage":skullCov}
        scoreFB:=fallback.NouveauCompteur()
		scoreTimeline,scoreCov:=buildScoreTimeline(a.opt.Score,a.opt.Deaths,replayScoreClock(&a.doc,a.interval,sample.Folder),scoreFB)
		scoreLayer:=map[string]any{"timeline":scoreTimeline,"coverage":scoreCov,"round_zero_fallbacks":scoreFB.Compte(fallback.NomReplayMancheZeroDecretee)}
		rows = append(rows, map[string]any{"radial_scan":radialScan,"skull_layer":skullLayer,"carrier_marks":carrierMarks,"objective_object_layer":objectiveLayer,"score_layer":scoreLayer,"ground_layer":groundLayer,"pickup_layer": pickupLayer, "weapon_change_layer": weaponChangeLayer, "placement_layer": placementLayer, "episode_layer": episodeLayer, "grapple_layer": grappleLayer, "equipment_layer": equipmentLayer, "translocation_layer": translocationLayer, "inventory_layer": inventoryLayer, "ability_layer": abilityLayer, "combat_shots": combatShots, "combat_vehicle_coverage": combatVehicleCoverage, "vehicle_shot_verdict": vehicleShotVerdict, "vehicles": vehicles, "vehicle_scan": map[string]any{"positions": len(in.Vehicles.Positions), "creations": len(in.Vehicles.Creations), "aims": len(in.Vehicles.Aims), "events": len(in.Vehicles.Events)}, "grenade_reads": map[string]any{"reads": reads, "coverage": readDoc.Coverage.GrenadeReads}, "loadouts": loadouts, "projectiles": map[string]any{"projectiles": projectiles, "published_by_raw": pubByRaw, "coverage": ProjectileCoverage{Tracks: len(in.Projectiles), Published: len(projectiles), Truncated: cuts}}, "grenades": map[string]any{"grenades": grenades, "coverage": grenadeCov}, "shots": map[string]any{"shots": shots, "orphans": orphanRows, "coverage": shotCov}, "folder": sample.Folder, "map": entry, "tracks": a.doc.Tracks, "roster": a.doc.Roster, "identity": a.reg.Section, "bounds": a.doc.Bounds, "origin_ms": a.doc.OriginMs, "clock": map[string]any{"origin_us": a.origin, "step_us": a.step, "frame_count": a.doc.FrameCount}, "frame_interval_ms": a.interval, "duration_ms": a.doc.DurationMS, "teams": a.equipes.couverture(a.viesTotal, a.viesNommees, a.viesSlotAmbigu, a.doc.Roster), "seats": a.siegeCov})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(output, raw, 0600); err != nil {
		t.Fatal(err)
	}
}
