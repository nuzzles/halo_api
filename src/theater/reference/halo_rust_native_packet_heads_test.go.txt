package grammar

import (
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
 "levelup/go-api/internal/games/halo_infinite/film/internal/source"
 "levelup/go-api/internal/games/halo_infinite/film/types"
)

// Direct native results before gameplay publication filters or time sorting.
func TestHaloRustNativePacketHeads(t *testing.T) {
 root:="/Users/simbleau/git/halo_api/experiments/films"
 paths,err:=filepath.Glob(filepath.Join(root,"*/*/chunk-*-type-2.bin"));if err!=nil { t.Fatal(err) }
 rows:=[]any{}
 for _,path:=range paths {
  data,err:=os.ReadFile(path);if err!=nil { t.Fatal(err) }
  relative,err:=filepath.Rel(root,path);if err!=nil { t.Fatal(err) }
  at:=0
  for ordinal,pk:=range source.Paquets(data,0) {
   offset:=at+16;at=offset+len(pk.Payload)
   if pk.Type!=0 || len(pk.Payload)<2 { continue }
   payload:=pk.Payload
   row:=map[string]any{"file":relative,"packet":ordinal,"offset":offset,"size":len(pk.Payload),"time":pk.TS}
   switch payload[0] {
   case 0xca: ev,ok:=decodeZoomHead(payload,pk.TS);row["family"]="zoom";row["event"]=ev;row["ok"]=ok
   case 0xc4: var stats types.BipedPickupStats;ev,ok:=decodeBipedPickup(payload,&stats);row["family"]="pickup";row["event"]=ev;row["ok"]=ok;row["stats"]=stats
   case 0xfa: ev,ok:=decodeTranslocHead(payload,pk.TS,nil);row["family"]="translocator";row["event"]=ev;row["ok"]=ok
   default: continue
   }
   rows=append(rows,row)
  }
 }
 raw,err:=json.Marshal(rows);if err!=nil { t.Fatal(err) }
 if err=os.WriteFile("/private/tmp/halo-native-packet-heads.json",raw,0600);err!=nil { t.Fatal(err) }
 t.Logf("%d captured native head reads",len(rows))
}
