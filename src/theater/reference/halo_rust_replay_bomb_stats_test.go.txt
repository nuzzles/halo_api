package replay
import("testing";"encoding/json";"os";"math/rand";"strconv";"levelup/go-api/internal/games/halo_infinite/film/internal/facts/objectives")
func TestHaloRustReplayBombStats(t *testing.T){
 rng:=rand.New(rand.NewSource(4135));rows:=[]any{}
 for run:=0;run<2048;run++{
  in:=BombStatsInput{DetonationsRead:run&1!=0,CarryRead:run&2!=0,KillsRead:run&4!=0,ArmingsRead:run&8!=0,FilmToMatchOffsetMS:(run%9-4)*1000,Objectives:[]objectives.IdentifiedEvent{},Armings:[]BombArming{},Kills:[]KillRef{},Carry:HeldObjectCarry{Events:[]HeldObjectEvent{},Periods:[]HeldObjectPeriod{},CarryMSByXUID:map[uint64]int{}}}
  for j:=0;j<run%25;j++ {
   x:=uint64(100+rng.Intn(5));if j%6==0{x=0};a:=rng.Intn(35)*1000-5000;b:=a+rng.Intn(20)*1000;open:=j%7==0;if open{b=HeldObjectOpenEndMS}
   in.Carry.Periods=append(in.Carry.Periods,HeldObjectPeriod{Slot:uint32(j%4),XUID:x,DebutMS:a,FinMS:b,Ouverte:open,FinParMort:j%3==0})
   if x!=0&&!open{in.Carry.CarryMSByXUID[x]+=b-a}
  }
  for j:=0;j<run%19;j++ {
   x:=strconv.Itoa(100+rng.Intn(5));if j%5==0{x=""};stat:="bomb_detonations";if j%7==0{stat="other"}
   in.Objectives=append(in.Objectives,objectives.IdentifiedEvent{NamedEvent:objectives.NamedEvent{TimeMS:rng.Intn(45)*1000,Stat:stat,Side:"A"},XUID:x})
   in.Armings=append(in.Armings,BombArming{TimeMS:rng.Intn(45)*1000})
   in.Kills=append(in.Kills,KillRef{KillerXUID:uint64(rng.Intn(6)+100),VictimXUID:uint64(rng.Intn(5)+100),TimeMS:int64(rng.Intn(50)*1000+[]int{0,150,151}[j%3])})
  }
  // Deliberately inconsistent aggregate: this layer must consume the supplied total.
  if run%11==0{in.Carry.CarryMSByXUID[999]=1234}
  stats,events:=BuildBombStats(in);if events==nil{events=[]BombEvent{}}
  rows=append(rows,map[string]any{"input":in,"output":map[string]any{"stats":stats,"events":events}})
 }
 probes:=[]any{};for _,k:=range []string{"bomb_armed","bomb_detonated","unknown",""}{s,c:=BombEventProvenance(k);probes=append(probes,[]string{k,s,c})}
 raw,e:=json.Marshal(map[string]any{"rows":rows,"provenance":probes});if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-bomb-stats.json",raw,0600);e!=nil{t.Fatal(e)}
}
