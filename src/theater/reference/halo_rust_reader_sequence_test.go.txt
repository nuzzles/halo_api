package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"math/rand"
	"os"
 "reflect"
	"testing"
)

func TestHaloRustReaderSequence(t *testing.T) {
	rng := rand.New(rand.NewSource(52141))
	rows := []any{}
	names := []string{"biped-spartan-ability-non-predicted-state-component", "biped-mobility-action-component", "unit-crouch-component", "unit-equipment-component"}
	allNames:=[]string{}
    for _,group:=range [][]string{haloRustHookNames,haloRustAbilityHookNames,haloRustObjectHookNames,haloRustManagedHookNames,haloRustEngineHookNames,haloRustPlayerHookNames,haloRustProbeHookNames,haloRustEquipmentHookNames}{allNames=append(allNames,group...)}
    for c := 0; c < 512+8*len(allNames); c++ {
		data := make([]byte, 128)
		rng.Read(data)
		if c%7 == 0 {
			data = data[:c%11]
		}
		start := c % 8
		name := names[c%4]
        if c >= 256 {name = []string{"biped-control-context-component", "biped-emp-timer-component", "managed-object-property-component", "object-multiplayer-properties-component"}[c%4]}
        if c >= 512 {name=allNames[(c-512)/8]}
		pa := ProfilDeBalayageParDefaut()
		pb := pa
		pa.Grammaire.CorpsAncrageCapacite = false
		pa.Grammaire.CorpsActionMobilite = false
		pb.Grammaire.CorpsAncrageCapacite = true
		pb.Grammaire.CorpsActionMobilite = true
		pb.Mouvement.MobilityActionExtraBits = 17
		pb.Mouvement.FullPrecision = c%3 == 0
		oa, ob := NouvelleObservation(), NouvelleObservation()
		obs := []*Observation{oa, ob}
		streams := []*[]any{}
		actions := [][][2]bool{{}, {}}
        ordered := [][]any{{}, {}}
		for i, o := range obs {
			idx := i
			stream := haloRustComponentHooks(o)
			streams = append(streams, stream)
			o.EtatMouvementHook = func(comp EtatMouvementComposant, slot uint32, v []uint64) {
				*stream = append(*stream, map[string]any{"kind": "movement_state", "component": comp.String(), "slot": slot, "values": v})
			}
			o.MobilityActionHook = func(a, b bool) { actions[idx] = append(actions[idx], [2]bool{a, b}) }
            observer:=reflect.ValueOf(o).Elem()
            for f:=0;f<observer.NumField();f++ {
                field:=observer.Field(f)
                if field.Kind()!=reflect.Func || !field.CanSet() || field.IsNil(){continue}
                original:=reflect.ValueOf(field.Interface());name:=observer.Type().Field(f).Name
                field.Set(reflect.MakeFunc(field.Type(),func(args []reflect.Value)[]reflect.Value {
                    before:=len(*stream);result:=original.Call(args)
                    if name=="MobilityActionHook" {ordered[idx]=append(ordered[idx],map[string]any{"kind":"mobility_action","value":actions[idx][len(actions[idx])-1]})}
                    for _,value:=range (*stream)[before:] {ordered[idx]=append(ordered[idx],map[string]any{"kind":"component","value":value})}
                    return result
                }))
            }

		}
		br := LecteurSur(data)
		br.PoserContexte(ContexteDeLecture{Profil: pa, Obs: oa})
		steps := []any{}
		restore := func() {}
		previous := pa
		var prevObs *Observation
		for j := 0; j < 6; j++ {
			switch j {
			case 1:
				restore = oa.neutraliserCaptures()
			case 2:
				previous = br.PoserProfil(pb)
				prevObs = br.PoserObservation(ob)
				if prevObs != oa {
					t.Fatal("observer swap")
				}
			case 3:
				br.PoserProfil(previous)
				br.PoserObservation(prevObs)
			case 4:
				restore()
			case 5:
				br.PoserContexte(ContexteDeLecture{Profil: pb, Obs: ob})
			}
			for i := range streams {
				*streams[i] = []any{}
                ordered[i] = []any{}
				actions[i] = [][2]bool{}
			}
			br.SetBitPos(start)
			_, _, ported := consumeByName(br, name, 35, 0)
			p := br.Profil()
			encoding := map[string]any{"bodies": map[string]any{"ability_anchor": p.Grammaire.CorpsAncrageCapacite, "mobility": p.Grammaire.CorpsActionMobilite, "mobility_extra_bits": p.Mouvement.MobilityActionExtraBits}, "index_bits": p.Mouvement.WorldObject.IndexW, "world_axis_bits": p.Mouvement.WorldObject.AxisW, "handle_bits": p.Mouvement.Traversal.IndexW, "traversal_axis_bits": p.Mouvement.Traversal.AxisW, "default_axis_bits": profile.LargeursAxeParDefautDuBuild(16), "region_axis_bits": map[string]any{}, "delta_axis_bits": [3]uint{p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth}, "full_precision": p.Mouvement.FullPrecision, "writer_absolute": p.Grammaire.GrammaireEcrivainI0, "delta_handle_tail": p.Mouvement.DeltaHasHandleTail}
			active := 0
			if j == 2 || j == 5 {
				active = 1
			}
			suppressed := j == 1 || j == 3
			steps = append(steps, map[string]any{"ordered":[]any{append([]any{},ordered[0]...),append([]any{},ordered[1]...)},"encoding": encoding, "active": active, "suppressed": suppressed, "end": br.BitPos(), "ported": ported, "observations": []any{append([]any{}, (*streams[0])...), append([]any{}, (*streams[1])...)}, "actions": []any{append([][2]bool{}, actions[0]...), append([][2]bool{}, actions[1]...)}})
		}
        // Removing the observer is different from neutralizing selected hooks.
        for i := range streams { *streams[i]=[]any{};ordered[i]=[]any{};actions[i]=[][2]bool{} }
        br.SetBitPos(start)
        detachedObserver:=br.PoserObservation(nil)
        if detachedObserver!=ob || br.BitPos()!=start {t.Fatal("detach changed receiver/cursor")}
        _,_,detachedOK:=consumeByName(br,name,35,0)
        detached:=map[string]any{"end":br.BitPos(),"ported":detachedOK,"ordered":[]any{append([]any{},ordered[0]...),append([]any{},ordered[1]...)}}
        if br.PoserObservation(detachedObserver)!=nil {t.Fatal("restore did not return nil")}
        rows = append(rows, map[string]any{"hex": hex.EncodeToString(data), "start": start, "name": name, "steps": steps,"detached":detached})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-reader-sequence.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
