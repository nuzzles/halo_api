package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"math/rand"
	"os"
	"path/filepath"
	"testing"
)

func TestHaloRustKFQComponents(t *testing.T) {
	root := "/Users/simbleau/git/halo_api/experiments/films/maps/01-bazaar-idle"
	var meta struct {
		Chunks []struct {
			Index int    `json:"index"`
			File  string `json:"file"`
		}
	}
	raw, e := os.ReadFile(filepath.Join(root, "film.json"))
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(raw, &meta); e != nil {
		t.Fatal(e)
	}
	var registry *Registry
	for _, c := range meta.Chunks {
		if c.Index == 0 {
			d, e := os.ReadFile(filepath.Join(root, c.File))
			if e != nil {
				t.Fatal(e)
			}
			registry, e = ParseRegistryChunk(source.Inflate(d))
			if e != nil {
				t.Fatal(e)
			}
		}
	}
	if registry == nil {
		t.Fatal("registry absent")
	}
	rows := []any{}
	rng := rand.New(rand.NewSource(414701))
	for _, arch := range registry.Archetypes {
		for ci, name := range arch.Components {
			for mode := 0; mode < 3; mode++ {
				reg := &Registry{Archetypes: make([]Archetype, 50)}
				for i := range reg.Archetypes {
					reg.Archetypes[i].Index = i
				}
				level := arch.Levels[ci]
				reg.Archetypes[arch.Index].Components = []string{name}
				reg.Archetypes[arch.Index].Levels = []uint32{level}
				v := KFQVariant{IDLowBits: 13, Preamble: mode, ExtraFields: mode == 2}
				w := &translocWriter{}
				w.put(mode, 0)
				if v.ExtraFields {
					w.put(32, 42)
				}
				w.put(1, 1)
				w.put(13, 1)
				w.put(2, 1)
				w.put(1, 0)
				w.put(1, 0)
				w.put(3, 1)
				w.put(6, 0)
				for j := 0; j < 256; j++ {
					b := uint64(0)
					if mode == 1 {
						b = uint64(rng.Intn(256))
					}
					w.put(8, b)
				}
				pay := w.pay
				if mode == 2 {
					pay = pay[:len(pay)/5]
				}
				world := NewWorld(reg)
				world.BindFull(0x40000001, uint32(arch.Index))
				walk := WalkPacketRecordsWithWorld(world, pay, v)
				rows = append(rows, map[string]any{"name": name, "level": level, "ti": arch.Index, "hex": hex.EncodeToString(pay), "variant": v, "walk": walk})
			}
		}
	}
	raw, e = json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-kfq-components.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
	t.Log("cases", len(rows))
}
