package grammar
import("testing";"os";"encoding/json";"encoding/hex";"math/rand")
func TestHaloRustCarrierMarks(t *testing.T){
 rng:=rand.New(rand.NewSource(4135));rows:=[]any{}
 for run:=0;run<256;run++{s:=CarrierMarkScan{};inputs:=[]any{}
  for part:=0;part<3;part++{w:=&translocWriter{};w.put(1,0)
   for i:=0;i<12;i++{w.put(32,uint64(0x40000000+512+i));ti:=35;if i%3==0{ti=42};w.put(32,uint64(ti));w.put(44,0)
    for j:=0;j<6;j++{w.put(rng.Intn(9),0);word:=rng.Uint32();if run%13!=0 && (i+j+run)%5!=0{word=[]uint32{0x00010005,0x0002000b,0x00040017,0x0008002f}[rng.Intn(4)]};w.put(32,uint64(word));w.put(32,uint64(rng.Uint32()))}
   }
   if part==2 && run%17==0{w.pay=[]byte{}}
   ts:=uint64(part*500000+run);s.KeyframeUS=append(s.KeyframeUS,ts);s.appendMarksOf(w.pay,ts);inputs=append(inputs,map[string]any{"hex":hex.EncodeToString(w.pay),"timestamp_us":ts})
  }
  if s.Marks==nil{s.Marks=[]CarrierMark{}};rows=append(rows,map[string]any{"inputs":inputs,"expected":s})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-carrier-marks.json",raw,0600);e!=nil{t.Fatal(e)}
}
