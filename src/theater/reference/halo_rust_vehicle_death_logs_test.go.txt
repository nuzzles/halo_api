func TestHaloRustVehicleDeathLogs(t *testing.T) {
 previous:=slog.Default();defer slog.SetDefault(previous)
 rng:=rand.New(rand.NewSource(412496));rows:=[]any{}
 for run:=0;run<512;run++ {
  st:=grammar.ObjectDeathStats{Records:map[uint32]int{},CleanRecords:map[uint32]int{},MaskDeclared:map[uint32]int{},MaskDeclaredDesync:map[uint32]int{}}
  for _,ti:=range []uint32{35,40,42} {
   if ti==40 && run%4==0 {continue}
   st.Records[ti]=rng.Intn(1000);st.CleanRecords[ti]=rng.Intn(1000);st.MaskDeclared[ti]=rng.Intn(1000)
   st.MaskDeclaredDesync[ti]=rng.Intn(500)+1
   if ti==40 && run%3==0 {st.MaskDeclaredDesync[ti]=0}
  }
  st.EventPackets=rng.Intn(1000);st.LocatedPackets=rng.Intn(1000)
  var retained any
  if run%3!=0 {st.CadreParDefaut=run%2==0;retained=st.CadreParDefaut}
  deaths:=[]types.ObjectDeath{};vehicleCount:=0
  for j:=0;j<run%17;j++ {ti:=[]uint32{35,40,42}[j%3];deaths=append(deaths,types.ObjectDeath{Slot:uint32(100+j),TypeIndex:ti});if ti==40 {vehicleCount++}}
  matchID:=[]string{"", "v41-test", "match-écriture"}[run%3]
  var buf bytes.Buffer;slog.SetDefault(slog.New(slog.NewJSONHandler(&buf,nil)))
  logVehicleDeathReads(matchID,len(deaths),vehicleCount,st)
  event:=map[string]any{};if err:=json.Unmarshal(buf.Bytes(),&event);err!=nil{t.Fatal(err)};delete(event,"time")
  rows=append(rows,map[string]any{"match_id":matchID,"deaths":deaths,"coverage":map[string]any{"records":st.Records,"clean_records":st.CleanRecords,"mask_declared":st.MaskDeclared,"mask_declared_desync":st.MaskDeclaredDesync},"events":st.EventPackets,"located":st.LocatedPackets,"retained":retained,"log":event})
 }
 raw,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)};if err=os.WriteFile("/private/tmp/halo-vehicle-death-logs.json",raw,0600);err!=nil{t.Fatal(err)}
}
