// Audit the reachable native named-type graph without executing decoder code.
// go run helper.go FILM_ROOT OUTPUT.json
package main

import (
	"bytes"
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"go/ast"
	"go/parser"
	"go/printer"
	"go/token"
	"os"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
)

type declaration struct {
	spec *ast.TypeSpec
	file *ast.File
	path string
}
type nativePackage struct {
	types map[string]declaration
	files []*ast.File
}
type edge struct {
	Target string `json:"target"`
	Via    string `json:"via"`
}
type node struct {
	ID         string   `json:"id"`
	File       string   `json:"file"`
	Line       int      `json:"line"`
	Definition string   `json:"definition"`
	Alias      bool     `json:"alias"`
	Methods    []string `json:"exported_methods"`
	Edges      []edge   `json:"edges"`
}

var fset = token.NewFileSet()
var moduleRoot, filmRoot string
var packages = map[string]*nativePackage{}
var nodes = map[string]*node{}
var external = map[string]bool{}
var hashes = map[string]string{}

func check(err error) {
	if err != nil {
		panic(err)
	}
}
func render(n ast.Node) string {
	var b bytes.Buffer
	check(printer.Fprint(&b, fset, n))
	return b.String()
}
func relative(path string) string {
	r, e := filepath.Rel(filmRoot, path)
	check(e)
	return filepath.ToSlash(r)
}
func load(dir string) *nativePackage {
	if p := packages[dir]; p != nil {
		return p
	}
	p := &nativePackage{types: map[string]declaration{}}
	entries, e := os.ReadDir(dir)
	check(e)
	for _, entry := range entries {
		if entry.IsDir() || !strings.HasSuffix(entry.Name(), ".go") || strings.HasSuffix(entry.Name(), "_test.go") {
			continue
		}
		path := filepath.Join(dir, entry.Name())
		f, e := parser.ParseFile(fset, path, nil, 0)
		check(e)
		p.files = append(p.files, f)
		for _, d := range f.Decls {
			g, ok := d.(*ast.GenDecl)
			if !ok || g.Tok != token.TYPE {
				continue
			}
			for _, s := range g.Specs {
				t := s.(*ast.TypeSpec)
				if _, ok := p.types[t.Name.Name]; ok {
					panic("duplicate type " + path + ":" + t.Name.Name)
				}
				p.types[t.Name.Name] = declaration{t, f, path}
			}
		}
	}
	packages[dir] = p
	return p
}
func importPath(f *ast.File, name string) string {
	for _, i := range f.Imports {
		path, e := strconv.Unquote(i.Path.Value)
		check(e)
		alias := filepath.Base(path)
		if i.Name != nil {
			alias = i.Name.Name
		}
		if alias == name {
			return path
		}
	}
	panic("unresolved import " + name)
}
func receiver(e ast.Expr) string {
	switch v := e.(type) {
	case *ast.Ident:
		return v.Name
	case *ast.StarExpr:
		return receiver(v.X)
	}
	panic("unsupported receiver " + render(e))
}
func visit(dir, name string) string {
	id := relative(dir) + "." + name
	if nodes[id] != nil {
		return id
	}
	p := load(dir)
	d, ok := p.types[name]
	if !ok {
		panic("unresolved type " + id)
	}
	typeVars := map[string]bool{}
	if d.spec.TypeParams != nil {
		for _, field := range d.spec.TypeParams.List {
			for _, name := range field.Names {
				typeVars[name.Name] = true
			}
		}
	}
	n := &node{ID: id, File: relative(d.path), Line: fset.Position(d.spec.Pos()).Line, Definition: render(d.spec.Type), Alias: d.spec.Assign.IsValid(), Methods: []string{}, Edges: []edge{}}
	nodes[id] = n
	raw, e := os.ReadFile(d.path)
	check(e)
	hashes[n.File] = fmt.Sprintf("%x", sha256.Sum256(raw))
	var walk func(ast.Expr, *ast.File, string)
	fields := func(fs *ast.FieldList, f *ast.File, via string) {
		if fs != nil {
			for _, field := range fs.List {
				walk(field.Type, f, via)
			}
		}
	}
	add := func(target, via string) { n.Edges = append(n.Edges, edge{target, via}) }
	walk = func(expr ast.Expr, f *ast.File, via string) {
		switch v := expr.(type) {
		case *ast.Ident:
			if typeVars[v.Name] {
				return
			}
			if _, ok := p.types[v.Name]; ok {
				add(visit(dir, v.Name), via)
				return
			}
			switch v.Name {
			case "bool", "string", "int", "int8", "int16", "int32", "int64", "uint", "uint8", "uint16", "uint32", "uint64", "uintptr", "float32", "float64", "complex64", "complex128", "byte", "rune", "any", "error":
				return
			}
			panic("unresolved identifier " + id + ":" + v.Name)
		case *ast.SelectorExpr:
			qualifier, ok := v.X.(*ast.Ident)
			if !ok {
				panic("unsupported selector " + render(v))
			}
			path := importPath(f, qualifier.Name)
			if strings.HasPrefix(path, "levelup/go-api/") {
				add(visit(filepath.Join(moduleRoot, strings.TrimPrefix(path, "levelup/go-api/")), v.Sel.Name), via)
			} else {
				target := path + "." + v.Sel.Name
				external[target] = true
				add(target, via)
			}
		case *ast.StarExpr:
			walk(v.X, f, via)
		case *ast.ArrayType:
			walk(v.Elt, f, via)
		case *ast.MapType:
			walk(v.Key, f, via)
			walk(v.Value, f, via)
		case *ast.ChanType:
			walk(v.Value, f, via)
		case *ast.Ellipsis:
			walk(v.Elt, f, via)
		case *ast.ParenExpr:
			walk(v.X, f, via)
		case *ast.StructType:
			for _, field := range v.Fields.List {
				label := "embedded"
				if len(field.Names) > 0 {
					names := []string{}
					for _, name := range field.Names {
						names = append(names, name.Name)
					}
					label = strings.Join(names, ",")
				}
				walk(field.Type, f, via+"/"+label)
			}
		case *ast.InterfaceType:
			fields(v.Methods, f, via+"/interface")
		case *ast.FuncType:
			if v.TypeParams != nil {
				for _, field := range v.TypeParams.List {
					for _, name := range field.Names {
						typeVars[name.Name] = true
					}
				}
				fields(v.TypeParams, f, via+"/constraints")
			}
			fields(v.Params, f, via+"/parameters")
			fields(v.Results, f, via+"/results")
		default:
			panic("unsupported type expression " + fmt.Sprintf("%T", expr) + " " + render(expr))
		}
	}
	fields(d.spec.TypeParams, d.file, "constraints")
	walk(d.spec.Type, d.file, "definition")
	for _, f := range p.files {
		for _, decl := range f.Decls {
			fn, ok := decl.(*ast.FuncDecl)
			if !ok || fn.Recv == nil || !fn.Name.IsExported() || receiver(fn.Recv.List[0].Type) != name {
				continue
			}
			n.Methods = append(n.Methods, fn.Name.Name+": "+render(fn.Type))
			walk(fn.Type, f, "method/"+fn.Name.Name)
			path := fset.Position(f.Pos()).Filename
			raw, e := os.ReadFile(path)
			check(e)
			hashes[relative(path)] = fmt.Sprintf("%x", sha256.Sum256(raw))
		}
	}
	sort.Strings(n.Methods)
	sort.Slice(n.Edges, func(i, j int) bool {
		a, b := n.Edges[i], n.Edges[j]
		if a.Via == b.Via {
			return a.Target < b.Target
		}
		return a.Via < b.Via
	})
	return id
}
func main() {
	if len(os.Args) != 3 {
		panic("FILM_ROOT OUTPUT.json")
	}
	var e error
	filmRoot, e = filepath.Abs(os.Args[1])
	check(e)
	moduleRoot = filepath.Clean(filepath.Join(filmRoot, "../../../.."))
	facade := load(filepath.Join(filmRoot, "decfilm"))
	roots := []string{}
	// Public facade declarations include alias targets and all function signatures.
	// Signature expressions are inventoried using synthetic named function types.
	for _, f := range facade.files {
		for _, decl := range f.Decls {
			switch v := decl.(type) {
			case *ast.GenDecl:
				if v.Tok != token.TYPE {
					continue
				}
				for _, s := range v.Specs {
					t := s.(*ast.TypeSpec)
					if t.Name.IsExported() {
						roots = append(roots, visit(filepath.Join(filmRoot, "decfilm"), t.Name.Name))
					}
				}
			case *ast.FuncDecl:
				if v.Recv != nil || !v.Name.IsExported() {
					continue
				}
				name := "signature_" + v.Name.Name
				facade.types[name] = declaration{&ast.TypeSpec{Name: &ast.Ident{Name: name, NamePos: v.Name.Pos()}, Type: v.Type}, f, fset.Position(f.Pos()).Filename}
				roots = append(roots, visit(filepath.Join(filmRoot, "decfilm"), name))
			}
		}
	}
	sort.Strings(roots)
	rows := []*node{}
	for _, n := range nodes {
		rows = append(rows, n)
	}
	sort.Slice(rows, func(i, j int) bool { return rows[i].ID < rows[j].ID })
	ext := []string{}
	for name := range external {
		ext = append(ext, name)
	}
	sort.Strings(ext)
	out, e := json.MarshalIndent(map[string]any{"reference_commit": "43a01721e8a02c87c955e175936f0ccf8dd97a81", "scope": "Reachable named declarations through public facade types/functions, all struct fields and exported receiver method signatures. Standard-library and external types are explicit leaves. Interface/any dynamic values, exported variable initializers, array-length constants, private methods and generic instantiation behavior are outside this syntactic graph. Not a behavior or Rust-parity claim.", "roots": roots, "types": rows, "external_leaves": ext, "source_sha256": hashes}, "", "  ")
	check(e)
	check(os.WriteFile(os.Args[2], append(out, '\n'), 0600))
}
