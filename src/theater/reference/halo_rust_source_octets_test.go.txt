package grammar

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustSourceOctets(t *testing.T) {
	rng := rand.New(rand.NewSource(4175))
	rows := []any{}
	for n := 0; n < 96; n++ {
		d := make([]byte, n)
		rng.Read(d)
		if n%3 == 0 {
			for i := range d {
				d[i] = 0xff
			}
		}
		data := []int{}
		for _, b := range d {
			data = append(data, int(b))
		}
		reads := []any{}
		for p := -65; p <= n*8+65; p++ {
			reads = append(reads, []any{p, source.OctetAuBit(d, p), source.U64LEAuBit(d, p)})
		}
		patterns := []any{}
		for p := 0; p+64 <= n*8; p++ {
			target := source.BitsAt(d, p, 64)
			at, ok := source.ChercherMotif64(d, target)
			patterns = append(patterns, []any{target, at, ok})
		}
		for i := 0; i < 16; i++ {
			target := rng.Uint64()
			at, ok := source.ChercherMotif64(d, target)
			patterns = append(patterns, []any{target, at, ok})
		}
		integers := []any{}
		for p := 0; p+8 <= len(d); p++ {
			integers = append(integers, []any{p, source.U16LE(d, p), source.U32LE(d, p), source.U64LE(d, p), source.U32BE(d, p)})
		}
		rows = append(rows, map[string]any{"integers": integers, "data": data, "reads": reads, "patterns": patterns})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-source-octets.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
