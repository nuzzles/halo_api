package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"os"
	"path/filepath"
	"testing"
)

func TestHaloRustGroundDirectory(t *testing.T) {
	var prior struct {
		Cases []struct {
			Hex   string
			Known []uint32
		}
	}
	raw, e := os.ReadFile("/private/tmp/halo-ground-keyframes.json")
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(raw, &prior); e != nil {
		t.Fatal(e)
	}
	rows := []any{}
	for run := 0; run < 128; run++ {
		c := prior.Cases[run*2]
		pay, _ := hex.DecodeString(c.Hex)
		known := map[uint32]bool{}
		for _, k := range c.Known {
			known[k] = run%4 != 1
		}
		if run%4 == 0 {
			known = map[uint32]bool{}
		}
		packet := func(kind uint16, stamp uint64) []byte {
			h := make([]byte, 16)
			binary.LittleEndian.PutUint16(h, kind)
			binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
			binary.LittleEndian.PutUint64(h[8:], stamp)
			return append(h, pay...)
		}
		data := append(packet(3, 123), packet(2, 456)...)
		data = append(data, packet(2, 789)...)
		files := map[string]string{}
		mode := (run / 4) % 8
		if mode >= 2 {
			files["chunk_00.bin"] = "290000001b000000"
		}
		if mode >= 3 {
			files["chunk_01.bin"] = hex.EncodeToString(data)
		}
		if mode == 4 {
			delete(files, "chunk_00.bin")
		}
		if mode == 5 {
			files["chunk_03.bin"] = hex.EncodeToString(data)
		}
		if mode == 6 {
			files["chunk_01.bin"] = "ff"
		}
		if mode == 7 {
			files["chunk_02.bin"] = hex.EncodeToString(data[:len(data)-3])
		}
		dir := t.TempDir()
		if mode == 0 {
			dir = filepath.Join(dir, "missing")
		}
		for name, h := range files {
			d, _ := hex.DecodeString(h)
			if e = os.WriteFile(filepath.Join(dir, name), d, 0600); e != nil {
				t.Fatal(e)
			}
		}
		records, err := ScanFilmKeyframeGroundWeapons(dir, known)
		if records == nil {
			records = []KeyframeGroundWeapon{}
		}
		kind := ""
		if err != nil {
			kind = "load"
			if err == ErrNoReadableFilmChunk {
				kind = "unreadable"
			}
		}
		rows = append(rows, map[string]any{"files": files, "missing": mode == 0, "known": known, "records": records, "error": kind})
	}
	raw, e = json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-ground-directory.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
