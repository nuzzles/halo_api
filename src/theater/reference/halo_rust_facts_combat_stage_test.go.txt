 rng:=rand.New(rand.NewSource(42021));rows:=[]any{}
 tags:=[]uint32{0xb0171062,0xc0e34c44,0x3b2567d4,0x9212e428,0,1}
 for run:=0;run<1024;run++{
  pos:=[]grammar.BipedPosition{};throws:=[]grammar.GrenadeThrow{};owner:=map[uint32]int{};proj:=[]types.ProjectileTrack{};published:=[]Track{};slots:=[]uint32{}
