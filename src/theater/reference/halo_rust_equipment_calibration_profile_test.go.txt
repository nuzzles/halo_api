package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustEquipmentCalibrationProfile(t *testing.T) {
	cat, err := profile.LoadMapQuantCatalog("/private/tmp/halo-levelup-v75/data/titles/halo_infinite/reference/map_quant_bounds.json")
	if err != nil {
		t.Fatal(err)
	}
	entry, err := cat.Lookup("bazaar")
	if err != nil {
		t.Fatal(err)
	}
	wr := entry.Range()
	rng := rand.New(rand.NewSource(373763))
	rows := []any{}
	for run := 0; run < 256; run++ {
		p := ProfilDeBalayageParDefaut()
		p.MPP = mppCandidates()[run%63]
		iw := uint(run % 4)
		axes := [3]uint{uint(7 + run%13), uint(8 + run%11), uint(9 + run%9)}
		if run%5 == 0 {
			axes = [3]uint{31, 32, 31}
		}
		region := uint32(run/4) % (1 << iw)
		lg := profile.PrecisionDescriptor{IndexW: iw, AxisW: axes, Region: region}
		p.PoserLargeursObjetDuMonde(lg)
		chunks := source.MemoryChunks{make([]byte, 8)}
		meta := []types.ChunkMeta{{Index: 0, ChunkType: 1}}
		inputs := []string{}
		spans := map[types.EquipmentLifeKey][]EquipmentLifeSpan{}
		band := map[uint32]bool{512: true}
		packet := func(pay []byte, ts uint64) []byte {
			h := make([]byte, 16)
			binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
			binary.LittleEndian.PutUint64(h[8:], ts)
			return append(h, pay...)
		}
		for c := 0; c < 3; c++ {
			chunk := []byte{}
			for j := 0; j < 20; j++ {
				pay := make([]byte, 512)
				rng.Read(pay)
				put := func(at, n int, v uint64) {
					for k := 0; k < n; k++ {
						m := byte(1 << uint(7-(at+k)%8))
						pay[(at+k)/8] &= ^m
						if v&(uint64(1)<<uint(n-1-k)) != 0 {
							pay[(at+k)/8] |= m
						}
					}
				}
				start := j % 8
				gen := uint32(j % 4)
				put(start, 3, 1)
				put(start+3, 13, 512)
				put(start+16, 2, uint64(gen))
				put(start+18, 6, 37)
				br := LecteurSur(pay)
				br.PoserContexte(ContexteDeLecture{Profil: p})
				br.SetBitPos(start + 24)
				consumeDefaultStateTI37(br)
				at := br.BitPos()
				put(at, 1, 0)
				put(at+1, 1, 0)
				put(at+2, 3, 1)
				put(at+5, 6, 0)
				at += 11
				posAt := at
				put(at, 2, 0)
				put(at+2, int(iw), uint64(region))
				at += 2 + int(iw)
				for _, n := range axes {
					q := uint64(rng.Uint32())%((uint64(1)<<n)-2) + 1
					put(at, int(n), q)
					at += int(n)
				}
				put(at, 2, 0)
				at += 2
				pay = pay[:(at+7)/8]
				xyz, ok := decodeWorldObjectPos(pay, posAt, &wr, lg)
				if !ok {
					t.Fatal("position")
				}
				ts := uint64(c*1000000 + j*10000 + 1)
				if run%7 != 0 || j < 4 {
					key := types.EquipmentLifeKey{Slot: 512, Gen: gen}
					spans[key] = append(spans[key], EquipmentLifeSpan{T0US: ts, T1US: ts + 20000, First: xyz, Points: 3})
				}
				delta := make([]byte, (27+projPosBits(lg)+7)/8)
				for k := 0; k < 27; k++ {
					var v uint64
					switch {
					case k == 0:
						v = 1
					case k >= 1 && k < 14:
						v = (512 >> uint(13-k)) & 1
					case k >= 14 && k < 16:
						v = (uint64(gen) >> uint(15-k)) & 1
					case k == 20:
						v = 1
					}
					if v != 0 {
						delta[k/8] |= 1 << uint(7-k%8)
					}
				}
				for k := 0; k < projPosBits(lg); k++ {
					if pay[(posAt+k)/8]&(1<<uint(7-(posAt+k)%8)) != 0 {
						delta[(27+k)/8] |= 1 << uint(7-(27+k)%8)
					}
				}
				chunk = append(chunk, packet(delta, ts)...)
				if run%11 == 0 && j%3 == 0 {
					pay = pay[:len(pay)-4]
				}
				chunk = append(chunk, packet(pay, ts)...)
			}
			chunks = append(chunks, chunk)
			meta = append(meta, types.ChunkMeta{Index: c + 1})
			inputs = append(inputs, hex.EncodeToString(chunk))
		}
		components := 1
		if run > 0 && run%17 == 0 {
			components = 0
		}
		if run > 0 && run%23 == 0 {
			band = map[uint32]bool{}
		}
		if run > 0 && run%29 == 0 {
			spans = map[types.EquipmentLifeKey][]EquipmentLifeSpan{}
		}
		film, err := source.Load(chunks, meta)
		if err != nil {
			t.Fatal(err)
		}
		fc := NewFilmContext(film)
		fc.bal = p
		fc.regLu = true
		fc.reg = &Registry{Archetypes: make([]Archetype, 38)}
		fc.reg.Archetypes[37] = Archetype{Index: 37, Components: make([]string, components), Levels: []uint32{0}}
		before := fc.ProfilDeBalayage()
		cal, ok := CalibrateMPPWidthsOf(fc, &wr, band, spans)
		if fc.ProfilDeBalayage().MPP != before.MPP {
			t.Fatal("profile not restored")
		}
		tracks, err := ScanWorldObjectsForBand(fc, &wr, band)
		if err != nil {
			t.Fatal(err)
		}
		if tracks == nil {
			tracks = []types.ProjectileTrack{}
		}
		scoreRows := []any{}
		for _, w := range mppCandidates() {
			if n := cal.ByWidths[w]; n > 0 {
				scoreRows = append(scoreRows, map[string]any{"key": w, "value": n})
			}
		}
		spanRows := []any{}
		for gen := uint32(0); gen < 4; gen++ {
			k := types.EquipmentLifeKey{Slot: 512, Gen: gen}
			if s, ok := spans[k]; ok {
				spanRows = append(spanRows, map[string]any{"key": k, "spans": s})
			}
		}
		rows = append(rows, map[string]any{"components": components, "empty_band": len(band) == 0, "tracks": tracks, "map": entry, "index": iw, "region": region, "axes": axes, "chunks": inputs, "spans": spanRows, "conclusive": ok, "calibration": map[string]any{"Widths": cal.Widths, "Agree": cal.Agree, "Runner": cal.Runner, "RunnerAgree": cal.RunnerAgree, "Anchors": cal.Anchors, "Chunks": cal.Chunks, "Lives": cal.Lives, "ByWidths": scoreRows}})
	}
	data, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-equipment-calibration-profile.json", data, 0600); err != nil {
		t.Fatal(err)
	}
}
