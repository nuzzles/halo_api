package replay
import("testing";"os";"encoding/json";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/types";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustReplayPickups(t *testing.T){
 rng:=rand.New(rand.NewSource(7531));rows:=[]any{}
 for run:=0;run<512;run++{
  pickups:=[]types.BipedPickup{};for i:=0;i<run%65;i++{pickups=append(pickups,types.BipedPickup{Slot:uint32(rng.Intn(8)),TimestampUS:uint64(rng.Intn(30))*100000,CatalogID:uint32(rng.Intn(8)),Class:uint8(rng.Intn(8))})};st:=types.BipedPickupStats{MultiEvent:run%4,RefusedNoRef:run%3,RefusedNoCatalog:run%2,RefusedOffBand:run%5}
  equipment:=map[uint32]string{1:"grenade",2:"equipment",5:"shared_equipment"};weapons:=map[uint32]string{3:"weapon_a",4:"weapon_b",5:"shared_weapon"};clk:=replayClock{origin:1000000,step:uint64(run%4)*100000,frames:3,families:equipment}
  occupant:=func(slot uint32,ts uint64)uint64{if (uint64(slot)+ts/100000)%3==0{return 0};return uint64(slot)*100000+ts/100000+1};in:=pickupInputs{st:st,weaponKeys:weapons};if run%2==0{in.occupant=occupant};points:=[]MapSpawnPoint{{X:0,Kind:"equipment"},{X:1,Kind:"grenade"}};positions:=[]grammar.BipedPosition{};for _,p:=range pickups{positions=append(positions,grammar.BipedPosition{Slot:p.Slot,TimestampUS:p.TimestampUS,X:float32(p.Slot),HasWorld:true})};placements:=[]EquipmentPlacement{{T0:0,UntilMax:8,End:"seen",Origin:"dropped",X:2},{T0:2,End:"open",Origin:"dropped",X:3}}
  if run%3!=0{in.judge=newPickupOriginJudge(Options{SpawnPoints:points,SpawnPointsState:"established"},positions,placements)}
  out,cov:=buildPickups(pickups,clk,in);if out==nil{out=[]Pickup{}}
  rows=append(rows,map[string]any{"input":pickups,"stats":st,"equipment":equipment,"weapons":weapons,"clock":map[string]any{"origin_us":clk.origin,"step_us":clk.step,"frames":clk.frames},"occupant":run%2==0,"origin":run%3!=0,"pickups":out,"coverage":cov})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-pickups.json",raw,0600);e!=nil{t.Fatal(e)}
}
