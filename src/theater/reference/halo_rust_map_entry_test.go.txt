package grammar

import (
 "encoding/json"
 "levelup/go-api/internal/games/halo_infinite/film/internal/profile"
 "math"
 "os"
 "testing"
)

func TestHaloRustMapEntry(t *testing.T) {
 rows:=[]any{}
 widths:=[]uint{0,1,2,3,8,26,31,32,63,127}
 for i:=0;i<256;i++ {
  e:=profile.MapQuantEntry{Module:"synthetic",Min:[3]float32{float32(i)-128,-0.25,-100},Max:[3]float32{float32(i)+1,0.5,200},AxisWidths:[3]uint{widths[i%len(widths)],widths[(i+3)%len(widths)],widths[(i+6)%len(widths)]},Region:uint32(i),RegionIndexBits:widths[(i/10)%len(widths)]}
  if i%4==0 {e.AxisWidths=[3]uint{13,14,15}}
  if i%7==0 {e.Min[0]=math.Float32frombits(0x80000000)}
  r:=e.Range(); rb:=[3][2]uint32{}
  for j:=0;j<3;j++ {rb[j]=[2]uint32{math.Float32bits(r[j].Min),math.Float32bits(r[j].Max)}}
  p:=e.PrecisionAbsolue(); l:=e.Layout()
  rows=append(rows,map[string]any{"entry":e,"index":e.EffectiveRegionIndexBits(),"precision":map[string]any{"index_bits":p.IndexW,"axis_bits":p.AxisW,"region":p.Region},"layout":map[string]any{"gate_bits":l.GateBits,"axis_widths":l.AxisW,"region":l.Region},"valid":l.Valid(),"range_bits":rb})
 }
 raw,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)};if err=os.WriteFile("/private/tmp/halo-map-entry.json",raw,0600);err!=nil {t.Fatal(err)}
}
