package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustNavpointRadialScan(t *testing.T) {
	rng := rand.New(rand.NewSource(4112))
	rows := []any{}
	for run := 0; run < 512; run++ {
		names := []string{"managed-navpoint-radial-progress", "managed-navpoint-radial-progress", "managed-navpoint-radial-progress"}
		if run%4 == 0 {
			names[1] = "unsupported"
		}
		if run%7 == 0 {
			names[2] = ""
		}
		arch := Archetype{Index: 12, Components: names, Levels: []uint32{0, 0, 0}}
		reg := &Registry{Archetypes: make([]Archetype, 13)}
		reg.Archetypes[12] = arch
		sc := &NavpointRadialScan{Blocked: map[int]int{}}
		prof := ProfilDeBalayageParDefaut()
		prof.Grammaire.ControleDeCorruption = run%3 == 0
		walk := navpointRadialWalk{prof: prof, arch: arch, reg: reg, sc: sc}
		walk.obs = walk.install()
        emissions := []any{}
        original := walk.obs.NavpointHook
        walk.obs.NavpointHook = func(f NavpointField, values []uint64) {
            emissions = append(emissions,map[string]any{"keyframe":walk.key,"field":f,"values":append([]uint64{},values...)})
            original(f,values)
        }
		inputs := []any{}
		band := map[uint32]bool{512: true, 513: true}
		for part := 0; part < 3; part++ {
			w := &translocWriter{}
			w.put(run%8, 0)
			for j := 0; j < 8; j++ {
				slot := 512 + rng.Intn(3)
				w.put(1, 1)
				w.put(13, uint64(slot))
				w.put(2, uint64(rng.Intn(4)))
				w.put(2, 0)
				n := 1 + rng.Intn(4)
				w.put(3, uint64(n))
				for k := 0; k < n; k++ {
					w.put(6, uint64(k))
				}
				for k := 0; k < n; k++ {
					w.put(8, uint64(rng.Intn(256)))
				}
				w.put(rng.Intn(7), 0)
			}
			if part == 2 && run%5 == 0 {
				w.pay = w.pay[:len(w.pay)-1]
			}
			ms := int32(part*1000 + run)
			walk.scanPayload(w.pay, band, ms)
			inputs = append(inputs, map[string]any{"key": false, "ms": ms, "hex": hex.EncodeToString(w.pay)})
		}
		w := &translocWriter{}
		w.put(1, 0)
		for j := 0; j < 4; j++ {
			w.put(32, uint64(0x40000200+j))
			w.put(32, 12)
			w.put(44, 0)
			n1 := 0
			if j%2 == 0 {
				n1 = 1
			}
			w.put(32, uint64(n1))
			if n1 > 0 {
				w.put(1, uint64(run%2))
				if run%2 != 0 {
					w.put(8, uint64(run%256))
				}
				if prof.Grammaire.ControleDeCorruption {
					w.put(32, 123)
				}
			}
			n2 := 1
			if (j+run)%5 == 0 {
				n2 = 0
			}
			w.put(32, uint64(n2))
			if n2 > 0 {
				for k := 0; k < 3; k++ {
					if names[k] == "" {
						continue
					}
					w.put(8, uint64(rng.Intn(256)))
					if prof.Grammaire.ControleDeCorruption {
						w.put(1, 0)
					}
				}
			}
		}
		if run%11 == 0 {
			w.pay = w.pay[:len(w.pay)-2]
		}
		walk.scanKeyframe(w.pay, int32(run))
		inputs = append(inputs, map[string]any{"key": true, "ms": run, "hex": hex.EncodeToString(w.pay)})
		pks := []FilmPacket{{Type: 2, Start: 0, Size: len(w.pay), TimestampUS: 1000000}, {Type: PacketTypeDelta, Start: 0, Size: len(w.pay), TimestampUS: 2000000}, {Type: 2, Start: 0, Size: len(w.pay), TimestampUS: 3000000}, {Type: 7, Start: 0, Size: 0, TimestampUS: 4000000}}
		if run%5 == 0 {
			pks[1].Type = 7
		}
		starts := map[int]int{}
		var start any
		if run%4 != 0 {
			starts[9] = run * 1000
			start = run * 1000
		}
		walk.scanChunk(w.pay, pks, band, starts, 9)
		inputs = append(inputs, map[string]any{"chunk": true, "packets": pks, "start": start, "hex": hex.EncodeToString(w.pay)})
		if sc.Reads == nil {
			sc.Reads = []types.NavpointRadialRead{}
		}
		rows = append(rows, map[string]any{"names": names, "check": prof.Grammaire.ControleDeCorruption, "inputs": inputs, "output": sc, "closure": sc.KeyframeExpvarPairs(),"emissions":emissions})
	}

	// Context propagation: the same direct dispatch consumes simulation-state but
	// must stop the record when its inherited completion policy is disabled.
	for run := 0; run < 512; run++ {
		names := []string{"managed-navpoint-radial-progress", "simulation-state", "managed-navpoint-radial-progress"}
		arch := Archetype{Index: 12, Components: names, Levels: []uint32{0, 0, 0}}
		reg := &Registry{Archetypes: make([]Archetype, 13)}
		reg.Archetypes[12] = arch
		prof := ProfilDeBalayageParDefaut()
		prof.Grammaire.SimStateComplet = run%2 != 0
		sc := &NavpointRadialScan{Blocked: map[int]int{}}
		walk := navpointRadialWalk{prof: prof, arch: arch, reg: reg, sc: sc}
		walk.obs = walk.install()
        emissions := []any{}
        original := walk.obs.NavpointHook
        walk.obs.NavpointHook = func(f NavpointField, values []uint64) {
            emissions = append(emissions,map[string]any{"keyframe":walk.key,"field":f,"values":append([]uint64{},values...)})
            original(f,values)
        }
		w := &translocWriter{}
		w.put(run%8, 0)
		for j := 0; j < 4; j++ {
			w.put(1, 1)
			w.put(13, uint64(512+j%2))
			w.put(2, 1)
			w.put(2, 0)
			w.put(3, 3)
			w.put(6, 0)
			w.put(6, 1)
			w.put(6, 2)
			w.put(8, uint64((run+j)%256))
			w.put(1, 0)
			w.put(8, uint64((run+j+17)%256))
		}
		if run%7 == 0 {
			w.pay = w.pay[:len(w.pay)-1]
		}
		walk.scanPayload(w.pay, map[uint32]bool{512: true, 513: true}, int32(run))
		if sc.Reads == nil {
			sc.Reads = []types.NavpointRadialRead{}
		}
		rows = append(rows, map[string]any{"names": names, "check": false, "simulation_complete": prof.Grammaire.SimStateComplet, "inputs": []any{map[string]any{"key": false, "ms": run, "hex": hex.EncodeToString(w.pay)}}, "output": sc, "closure": sc.KeyframeExpvarPairs(),"emissions":emissions})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-navpoint-radial-scan.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
