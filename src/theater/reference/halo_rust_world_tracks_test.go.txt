package grammar
import("testing";"os";"encoding/json";"encoding/hex";"math/rand";"sort";"levelup/go-api/internal/games/halo_infinite/film/internal/profile";"levelup/go-api/internal/games/halo_infinite/film/types")
func TestHaloRustWorldTracks(t *testing.T){
 rng:=rand.New(rand.NewSource(4142));cases:=[]any{};tracks:=[]any{}
 for run:=0;run<512;run++{
  lg:=profile.PrecisionDescriptor{IndexW:uint(1+run%2),AxisW:[3]uint{uint(13+run%5),uint(14+run%5),uint(15+run%5)},Region:uint32(run%2)};wr:=profile.Vec3Range{{Min:-100,Max:300},{Min:-200,Max:400},{Min:-300,Max:500}};band:=map[uint32]bool{512:true,513:true,515:true};w:=&translocWriter{}
  for i:=0;i<12;i++{w.put(rng.Intn(8),0);w.put(1,1);w.put(13,uint64(512+rng.Intn(4)));w.put(2,uint64(rng.Intn(4)));w.put(2,0);w.put(3,2);w.put(6,0);if i%3==0{w.put(6,18)}else{w.put(6,1)};w.put(2,0);w.put(int(lg.IndexW),uint64(rng.Intn(1<<lg.IndexW)));for _,n:=range lg.AxisW{w.put(int(n),uint64(rng.Intn(1<<n)))};w.put(2,0)}
  for cut:=0;cut<4;cut++{pay:=w.pay;if cut>0{pay=pay[:rng.Intn(len(pay))]};raw:=scanProjectileRecords(pay,band,&wr,lg);records:=[]any{};for _,r:=range raw{records=append(records,map[string]any{"slot":r.slot,"generation":r.gen,"sample":r.ProjectileSample})};cases=append(cases,map[string]any{"hex":hex.EncodeToString(pay),"widths":lg.AxisW,"index_bits":lg.IndexW,"region":lg.Region,"records":records})}
  input:=[]types.ProjectileSample{};for i:=0;i<128;i++{ts:=uint64(rng.Intn(12))*125000;input=append(input,types.ProjectileSample{TimestampUS:ts,Chunk:rng.Intn(3),X:float32(rng.Intn(4)),Y:float32(rng.Intn(2)),Z:float32(rng.Intn(2)),AtRest:rng.Intn(12)==0})};pts:=append([]types.ProjectileSample{},input...);sort.Slice(pts,func(i,j int)bool{return lessSample(pts[i],pts[j])});out:=[]types.ProjectileTrack{};for _,seg:=range splitLives(pts){out=append(out,types.ProjectileTrack{Slot:512,Gen:1,Pts:seg})};sort.Slice(out,func(i,j int)bool{return lessTrack(out[i],out[j])});tracks=append(tracks,map[string]any{"input":input,"tracks":out})
 }
 d,e:=json.Marshal(map[string]any{"cases":cases,"tracks":tracks});if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-world-tracks.json",d,0600);e!=nil{t.Fatal(e)}
}
