package replay

import (
 "encoding/json"
 "math/rand"
 "os"
 "testing"
 "levelup/go-api/internal/games/halo_infinite/film/internal/grammar"
)

func TestHaloRustDeathContext(t *testing.T) {
 rng := rand.New(rand.NewSource(410924))
 rows := []any{}
 for run:=0;run<512;run++ {
  offset:=int64(run%7-3)*1000
  lives:=[]lifeSpan{}
  exported:=[]any{}
  bridge:=map[uint32]uint64{}
  positions:=[]grammar.BipedPosition{}
  teams:=map[uint64]int{1:0,2:0,3:0,4:0,5:1}
  for i:=0;i<5;i++ {
   slot:=uint32(i+10); x:=uint64(i+1)
   // The second life reuses a slot under a different identity.
   if i==4 {slot=10}
   from:=int64(10000000);to:=int64(15000000)
   if i==4 {from=16000000;to=20000000}
   if i==2 {to=12000000}
   if run%9==0 && i==3 {x=0}
   l:=lifeSpan{slot:slot,from:from,to:to,xuid:x};lives=append(lives,l)
   exported=append(exported,map[string]any{"slot":slot,"from":from,"to":to,"xuid":x,"bid":"","cause":"","named_by":""})
   if x!=0 {bridge[slot]=x}
   for j:=0;j<20;j++ {
    ts:=uint64(from+int64(rng.Intn(7001))*1000)
    positions=append(positions,grammar.BipedPosition{Slot:slot,TimestampUS:ts,X:float32(rng.Intn(20)),Y:float32(rng.Intn(20)),Z:100,HasWorld:run%11!=0})
   }
  }
  // Exact boundaries: one-second freshness, life endpoints, strict death order.
  positions=append(positions,grammar.BipedPosition{Slot:10,TimestampUS:14000000,X:0,Y:0,HasWorld:true},grammar.BipedPosition{Slot:11,TimestampUS:14000000,X:3,Y:4,HasWorld:true})
  journal:=[]MortDuJournal{}
  for _,ms:=range []int64{12000,14000,15000,15001,16000,18000,22000} {
   for x:=uint64(1);x<=5;x++ {journal=append(journal,MortDuJournal{VictimeXUID:x,TempsMS:ms-offset})}
  }
  if run%13==0 {delete(teams,1)}
  disagreement:=0;if run%17==0 {disagreement=1}
  if run%19==0 {bridge=map[uint32]uint64{}}
  reg:=IdentityRegistry{own:OwnerReport{lives:lives,SlotXUID:bridge,DeathOffsetMS:offset,IndexDisagreements:disagreement}}
  result:=ContextesDesMorts(EntreeContexteMorts{Positions:positions,Registre:reg,Journal:journal,Equipes:teams})
  out:=[]any{};deaths:=[]any{}
  for _,d:=range journal {deaths=append(deaths,map[string]any{"victim_xuid":d.VictimeXUID,"time_ms":d.TempsMS})}
  for _,c:=range result {out=append(out,map[string]any{"victim_xuid":c.VictimeXUID,"time_ms":c.TempsMS,"nearest_teammate_m":c.PlusProcheM,"visible":c.Visibles,"waiting":c.EnAttente,"out_of_sight":c.HorsDeVue,"total":c.Total})}
  rows=append(rows,map[string]any{"positions":positions,"lives":exported,"bridge":len(bridge)>0,"offset":offset,"disagreements":disagreement,"teams":teams,"journal":deaths,"expected":out})
 }
 raw,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-death-context.json",raw,0600);err!=nil {t.Fatal(err)}
}
