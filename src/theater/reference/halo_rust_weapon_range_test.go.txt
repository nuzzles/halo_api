package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"
	"testing"
)

func TestHaloRustWeaponRange(t *testing.T) {
	rows := []any{}
	for _, indices := range [][]int{{}, {1}, {2}, {1, 3}, {3, 1, 2}, {0, 1, 2, 3}} {
		root := t.TempDir()
		chunks := []any{}
		for _, index := range indices {
			payload := []byte{0xd2, 0, 0, 0}
			if index%2 == 0 {
				payload = make([]byte, 64)
				payload[0] = 0xc0
			}
			raw := make([]byte, 16)
			binary.LittleEndian.PutUint32(raw[4:], uint32(len(payload)))
			binary.LittleEndian.PutUint64(raw[8:], uint64(index*100))
			raw = append(raw, payload...)
			if err := os.WriteFile(fmt.Sprintf("%s/chunk_%02d.bin", root, index), raw, 0600); err != nil {
				t.Fatal(err)
			}
			chunks = append(chunks, map[string]any{"index": index, "hex": hex.EncodeToString(raw)})
		}
		for n := -1; n <= 4; n++ {
			shots, se := ScanFilmWeaponShots(root, n)
			damages, base, de := ScanFilmWeaponDamages(root, &Registry{}, n)
			rows = append(rows, map[string]any{"chunks": chunks, "last": n, "shots": shots, "shot_error": se != nil, "damages": damages, "base": base, "damage_error": de != nil})
		}
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-weapon-range.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
