package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustBipedContext(t *testing.T) {
	rng := rand.New(rand.NewSource(414141))
	rows := []any{}
	names := make([]string, 64)
	for id, name := range map[int]string{6: "simulation-state", 22: "unit-grenade-counts-component", 26: "unit-equipment-component", 28: "unit-active-camo-state-component", 48: "biped-desired-ability-set-component", 56: "biped-spartan-ability-energy-component", 57: "biped-spartan-ability-component", 59: "biped-spartan-ability-non-predicted-state-component"} {
		names[id] = name
	}
	for c := 0; c < 512; c++ {
		p := ProfilDeBalayageParDefaut()
		p.Grammaire.SimStateComplet = c%2 != 0
		p.Grammaire.CorpsAncrageCapacite = c%3 != 0
		p.Mouvement.FullPrecision = c%5 == 0
		arch := Archetype{Index: 35, Components: names, Levels: make([]uint32, 64)}
		for i := range arch.Levels {
			arch.Levels[i] = uint32(c % 4)
		}
		ids := []int{6, 22, 26, 28, 48, 56, 57, 59}
		if c%7 == 0 {
			ids = ids[1:]
		}
		at := 2 + c%8
		data := make([]byte, 512)
		if c%4 >= 2 {
			rng.Read(data)
		}
		data[at/8] &= ^(byte(1) << uint(7-at%8)) // closed simulation body
		if c%11 == 0 {
			data = data[:1+c%13]
		}
		obs := NouvelleObservation()
		walkHooks := haloRustComponentHooks(obs)
		obs.EtatMouvementHook = func(comp EtatMouvementComposant, slot uint32, values []uint64) {
			*walkHooks = append(*walkHooks, map[string]any{"kind": "movement_state", "component": comp.String(), "slot": slot, "values": values})
		}
		visited := []int{}
		g := grammaireRecord{arch: arch, prof: p, obs: obs}
		walkComponentsAt(data, at, len(data)*8, ids, g, func(id int) bool { visited = append(visited, id); return true })
		steps := []any{}
		attempts := []any{}
		pos := at
		for _, id := range ids {
			o := NouvelleObservation()
			hooks := haloRustComponentHooks(o)
			o.EtatMouvementHook = func(comp EtatMouvementComposant, slot uint32, values []uint64) {
				*hooks = append(*hooks, map[string]any{"kind": "movement_state", "component": comp.String(), "slot": slot, "values": values})
			}
			br := LecteurSur(data)
			br.PoserContexte(ContexteDeLecture{Profil: p, Obs: o})
			br.SetBitPos(pos)
			_, _, ok := consumeByName(br, names[id], 35, arch.Level(id))
			attempts = append(attempts, map[string]any{"id": id, "end": br.BitPos(), "status": ok, "in_bounds": br.BitPos() <= len(data)*8, "observations": *hooks})
			if !ok || br.BitPos() > len(data)*8 {
				break
			}
			pos = br.BitPos()
			steps = append(steps, map[string]any{"id": id, "end": pos, "observations": *hooks})
		}
		impulse := abilityImpulseScanner{st: &types.AbilityImpulseStats{Records: 1}, gram: grammaireRecord{arch: arch, prof: p}, i57idx: 57, i59idx: 59}
		impulseObs := NouvelleObservation()
		impulseObs.SpartanAbilityHook = func(tag, _, _ uint64, _ bool) { impulse.tag57, impulse.got57 = tag, true }
		impulseObs.AbilityNonPredictedHook = func(v AbilityNonPredictedState) { impulse.tag59, impulse.got59 = uint64(v.Tag), true }
		impulse.gram.obs = impulseObs
		impulse.account(data, at-2, len(data)*8, append([]int{0}, ids...), 512, 1, FilmPacket{TimestampUS: 1000})
		impulse.st.Scanned = true
		grapple := grappleScanner{st: &GrappleStats{Records: 1}, gram: grammaireRecord{arch: arch, prof: p}, i59idx: 59}
		grappleObs := NouvelleObservation()
		grappleObs.AbilityNonPredictedHook = func(v AbilityNonPredictedState) { grapple.last, grapple.got = v, true }
		grapple.gram.obs = grappleObs
		grapple.account(data, at-2, len(data)*8, append([]int{0}, ids...), 512, 1, FilmPacket{TimestampUS: 1000})
		encoding := map[string]any{"bodies": map[string]any{"ability_anchor": p.Grammaire.CorpsAncrageCapacite, "mobility": p.Grammaire.CorpsActionMobilite, "mobility_extra_bits": p.Mouvement.MobilityActionExtraBits}, "index_bits": p.Mouvement.WorldObject.IndexW, "world_axis_bits": p.Mouvement.WorldObject.AxisW, "handle_bits": p.Mouvement.Traversal.IndexW, "traversal_axis_bits": p.Mouvement.Traversal.AxisW, "default_axis_bits": profile.LargeursAxeParDefautDuBuild(16), "region_axis_bits": map[string]any{"0": p.Mouvement.WorldObject.AxisW, "1": p.Mouvement.WorldObject.AxisW}, "delta_axis_bits": [3]uint{p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth}, "full_precision": p.Mouvement.FullPrecision, "writer_absolute": false, "delta_handle_tail": false}
		rows = append(rows, map[string]any{"names": names, "ids": ids, "level": c % 4, "start": at, "hex": hex.EncodeToString(data), "encoding": encoding, "simulation": p.Grammaire.SimStateComplet, "visited": visited, "steps": steps, "attempts": attempts, "walk_observations": *walkHooks, "grapple": grapple.out, "grapple_stats": grapple.st, "impulses": impulse.out, "impulse_stats": impulse.st})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-biped-context.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
