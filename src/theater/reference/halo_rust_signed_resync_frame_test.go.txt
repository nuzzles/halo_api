 rows:=[]any{}

  data:=make([]byte,16);data[0]=0xc0;data[1]=0x04;data[4]=0x84;data[5]=0x08
  reg:=&Registry{Archetypes:[]Archetype{{Index:0,Components:[]string{"rewind"},Levels:[]uint32{0}},{Index:1,Components:[]string{"signed-scan"},Levels:[]uint32{0}}}}

   world:=NewWorld(reg);world.BindFull(1,0);world.BindFull(0,1)

 }}}
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-signed-resync-frame.json",raw,0600);e!=nil{t.Fatal(e)}
}
