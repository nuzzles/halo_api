func haloProjectileSample(s types.ProjectileSample) any { return map[string]any{"time":s.TimestampUS,"chunk":s.Chunk,"xyz":[3]uint32{math.Float32bits(s.X),math.Float32bits(s.Y),math.Float32bits(s.Z)},"rest":s.AtRest} }
func haloProjectileTracks(tracks []types.ProjectileTrack) any {if tracks==nil{return nil};out:=[]any{};for _,tr:=range tracks{pts:=[]any{};for _,p:=range tr.Pts{pts=append(pts,haloProjectileSample(p))};out=append(out,map[string]any{"slot":tr.Slot,"generation":tr.Gen,"points":pts})};return out}
