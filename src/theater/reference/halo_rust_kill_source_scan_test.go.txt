 ids:=damagetag.IDs();tags:=append(append([]uint32{},ids...),0,0xffff,0x10000,0xffffffff);labels:=[]any{}

 candidates:=func(cs []candidate,packet bool)[]any{out:=[]any{};for _,c:=range cs {var p any;if packet {p=map[string]int{"chunk":c.chunk,"packet":c.pidx}};out=append(out,map[string]any{"packet":p,"time_ms":c.ms,"bit":c.bit,"tag":c.tag,"victim":c.victim,"killer":c.killer,"category":c.cat})};return out}

  }

  for j,c:=range relaxed {c.chunk=i%3;c.pidx=i%9;c.ms=i*100;c.bit=int(j%4);all=append(all,c,c)}
 }

}
