// Additional oracle from d61443ef59268ad734355db8e9974f68db5ca6d0.
// Uses the actual dispatch and default instrument context. These results do not
// replace the older pinned fixtures or prove the assumed vehicle runtime gate.
func TestHaloRustM4bComponents(t *testing.T) {
	type row struct {
		Name string `json:"name"`
		Input string `json:"input"`
		Start int `json:"start"`
		End int `json:"end"`
		Ported bool `json:"ported"`
		Next uint64 `json:"next"`
	}
	names := []string{
		"personal-ai-data-component", "player-aim-assist-component",
		"player-desired-frame-configuration-component",
		"managed-object-looping-sound-component", "vehicle-type-physics-component",
		"vehicle-emp-timer-component", "halo-rust-unknown-component",
	}
	var out []row
	state := uint64(0x482395abc19)
	for _, name := range names {
		for sample := 0; sample < 512; sample++ {
			data := make([]byte, 128)
			for i := range data {
				state ^= state << 13
				state ^= state >> 7
				state ^= state << 17
				data[i] = byte(state)
				if sample < 8 { data[i] = 0 }
				if sample >= 8 && sample < 16 { data[i] = 255 }
			}
			start := sample % 8
			br := lecteurDInstrument(data)
			br.SetBitPos(start)
			_, _, ported := consumeByName(br, name, 0, 1)
			end := br.BitPos()
			next := br.ReadBits(8)
			out = append(out, row{name, hex.EncodeToString(data), start, end, ported, next})
		}
	}
	path := os.Getenv("HALO_RUST_M4B_OUT")
	if path == "" { t.Fatal("HALO_RUST_M4B_OUT is required") }
	f, err := os.Create(path)
	if err != nil { t.Fatal(err) }
	z := zlib.NewWriter(f)
	if err := json.NewEncoder(z).Encode(out); err != nil { t.Fatal(err) }
	if err := z.Close(); err != nil { t.Fatal(err) }
	if err := f.Close(); err != nil { t.Fatal(err) }
	t.Logf("wrote %d actual-native component cases", len(out))
}
