package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustVehicleI0(t *testing.T) {
	rng := rand.New(rand.NewSource(419540))
	rows := []any{}
	for run := 0; run < 512; run++ {
		gate := 4 + run%5
		region := uint32(0)
		if gate > 4 {
			region = uint32(run) & ((1 << uint(gate-4)) - 1)
		}
		widths := [3]uint{uint(8 + run%56), uint(8 + (run*3)%56), uint(8 + (run*7)%56)}
		lay := profile.I0Layout{GateBits: gate, Region: region, AxisW: widths}
		wr := profile.Vec3Range{{Min: -123.5, Max: 908.25}, {Min: 0.125, Max: 2777.75}, {Min: -109.75, Max: 141.5}}
		w := &translocWriter{}
		start := run % 8
		w.put(start, 0)
		w.put(4, 0)
		w.put(gate-4, uint64(region))
		for _, n := range widths {
			q := rng.Uint64()
			q &= (uint64(1) << n) - 1
			if uint32(q) == 0 || uint32(q) == uint32(1)<<n-1 {
				q = 1
			}
			w.put(int(n), uint64(q))
		}
		for variant := 0; variant < 5; variant++ {
			pay := append([]byte{}, w.pay...)
			at := start
			switch variant {
			case 1:
				pay = pay[:rng.Intn(len(pay)+1)]
			case 2:
				pay[start/8] |= 1 << uint(7-start%8)
			case 3:
				at = -1
			case 4:
				for bit := 0; bit < int(widths[0]); bit++ {
					p := start + gate + bit
					pay[p/8] &^= 1 << uint(7-p%8)
				}
			}
			v, ok := decodeBipedI0Pos(pay, at, lay, &wr)
			var point any
			if ok {
				point = v
			}
			rows = append(rows, map[string]any{"hex": hex.EncodeToString(pay), "at": at, "layout": map[string]any{"gate_bits": gate, "axis_widths": widths, "region": region}, "point": point})
		}
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-vehicle-i0.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
