package replay
import("testing";"os";"encoding/json";"math/rand";"fmt")
func TestHaloRustIdentityTables(t *testing.T){
 rng:=rand.New(rand.NewSource(7561));rows:=[]any{}
 for run:=0;run<2048;run++{
  film:=FilmPlayerTable{Seats:[]FilmPlayerSeat{},Build:"HI_1_13_0",InterleavedVacant:run%11==0};refusals:=[]FilmTableRefusal{FilmTableRead,FilmTableNoRegistry,FilmTableNoSection,FilmTableUnknownBuild,FilmTableTruncated,FilmTableNotFound};if run%4==0{film.Refusal=refusals[run%len(refusals)]}
  for i:=0;i<run%17;i++{x:=uint64(rng.Intn(20));name:="";if i%3!=0{name=fmt.Sprintf("Player%d",x)};film.Seats=append(film.Seats,FilmPlayerSeat{FilmIndex:rng.Intn(24)-1,XUID:x,Gamertag:name})};film.Occupied=len(film.Seats);film.Vacant=32-film.Occupied
  repl:=PlayerIndexTable{ByXUID:map[uint64]int{},Readings:run%31,Disagreements:run%4};for i:=0;i<run%23;i++{repl.ByXUID[uint64(rng.Intn(24))]=rng.Intn(24)-1}
  out:=composerTableDIndex(IdentityInput{FilmTable:film,PlayerIndices:repl});guard,n:=injectiveOrEmpty(repl);methods:=map[uint64]string{};for x:=uint64(0);x<25;x++{methods[x]=string(out.voieDuLienDIndex(x))}
  rows=append(rows,map[string]any{"film":film,"replication":repl,"output":map[string]any{"table":out.table,"methods":out.voie,"names":out.noms,"coverage":out.couverture},"guard":guard,"collisions":n,"methods":methods})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-identity-tables.json",raw,0600);e!=nil{t.Fatal(e)}
}
