package killsource

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustKillIndexMotif(t *testing.T) {
	rng := rand.New(rand.NewSource(415201))
	rows := []any{}
	for c := 0; c < 512; c++ {
		mode := c % 8
		xs := []uint64{rng.Uint64(), rng.Uint64(), rng.Uint64(), rng.Uint64()}
		slots := []types.PlayerSlot{{XUID: xs[0], Gamertag: "Alpha"}, {XUID: xs[1], Gamertag: "Beta"}, {XUID: 0, Gamertag: "Zero"}, {XUID: xs[3], Gamertag: ""}}
		feed := map[string]uint64{"Gamma": xs[2], "": xs[3], "ZeroFeed": 0}
		if mode == 7 {
			feed["Replacement"] = xs[0]
		}
		chunks := source.MemoryChunks{}
		meta := []types.ChunkMeta{}
		inputs := []any{}
		for ch := 0; ch < 4; ch++ {
			data := []byte{}
			at := 0
			put := func(n int, v uint64) {
				for k := n - 1; k >= 0; k-- {
					if at/8 >= len(data) {
						data = append(data, 0)
					}
					data[at/8] |= byte((v>>uint(k))&1) << uint(7-at%8)
					at++
				}
			}
			put(16+c%8, 0)
			for j, x := range xs {
				if mode == 4 && j == 0 && ch != 3 {
					continue
				}
				pi := j + 3
				if mode == 1 && ch == 2 && j == 0 {
					pi = 17
				}
				if (mode == 2 && j < 2) || (mode == 3 && j < 3) {
					pi = 9
				}
				if ch == 0 || ch == 3 {
					pi = 31 - j
				} // excluded bootstrap and final chunk disagree deliberately
				put(5, uint64(pi))
				for b := 0; b < 8; b++ {
					put(8, (x>>uint(8*b))&255)
				}
				put(11, 0)
			}
			// A later occurrence is not allowed to replace the first match.
			put(5, 29)
			for b := 0; b < 8; b++ {
				put(8, (xs[2]>>uint(8*b))&255)
			}
			index := ch
			if mode == 6 && ch >= 2 {
				index++
			}
			chunks = append(chunks, data)
			meta = append(meta, types.ChunkMeta{Index: index})
			inputs = append(inputs, map[string]any{"index": index, "hex": hex.EncodeToString(data)})
			if mode == 5 && ch == 1 {
				break
			}
		}
		src, e := source.Load(chunks, meta)
		if e != nil {
			t.Fatal(e)
		}
		got := lireIndexParMotif(&film{src: src}, slots, &killFeed{xuidDe: feed})
		if mode <= 3 && mode > 0 && len(got.nomParIndex) != 0 {
			t.Fatal("conflicting motif published", c)
		}
		inslots := []any{}
		for _, s := range slots {
			inslots = append(inslots, map[string]any{"xuid": s.XUID, "name": s.Gamertag})
		}
		rows = append(rows, map[string]any{"mode": mode, "chunks": inputs, "slots": inslots, "feed": feed, "output": map[string]any{"names": got.nomParIndex, "readings": got.lectures, "disagreements": got.desaccords, "absent": got.absents}})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-kill-index-motif.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
