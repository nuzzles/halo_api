package replay
import("testing";"encoding/json";"os";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/types")
func TestHaloRustReplayHeldObject(t *testing.T){
 rng:=rand.New(rand.NewSource(4130));rows:=[]any{}
 for run:=0;run<2048;run++{
  events:=[]HeldObjectEvent{};deaths:=[]Death{}
  for j:=0;j<run%45;j++{events=append(events,HeldObjectEvent{TimeMS:rng.Intn(50)*1000-5000,Slot:uint32(10+rng.Intn(5)),XUID:999,Pickup:rng.Intn(3)!=0})}
  for j:=0;j<run%15;j++{deaths=append(deaths,Death{TimeMS:int64(rng.Intn(50)*1000-5000),XUID:uint64(100+rng.Intn(5))})}
  occupant:=func(slot uint32,ms int)uint64{if slot==14{return 0};return uint64(100+(int(slot)+ms/10000)%4)}
  if run%13==0{occupant=nil}
  out:=BuildHeldObjectCarry(events,occupant,deaths);if out.Events==nil{out.Events=[]HeldObjectEvent{}};if out.Periods==nil{out.Periods=[]HeldObjectPeriod{}}
  clock:=matchClock{origin:uint64(run%7)*1000000,step:[]uint64{0,100000,16667,1000000}[run%4],frames:[]int{-1,0,100,1000,10000}[run%5],deathOffsetMS:int64(run%7-3)*1000}
  if run%11==0{clock.deathOffsetMS=[]int64{-1<<63,1<<63-1,-1,0,1}[run%5]}
  presence:=carrierPresence{named:map[string][]presenceSpan{"100":{{10,80},{100,180}},"101":{{200,500}}}}
  if run%3==0{presence.unnamed=[]presenceSpan{{300,400}}}
  bombs,cov:=buildBombCarries(out,clock,presence);if bombs==nil{bombs=[]BombCarry{}}
  changes:=[]types.HeldWeaponChange{};for j:=0;j<run%17;j++{families:=[]uint32{0,bombHeldFamily,1};changes=append(changes,types.HeldWeaponChange{TimestampUS:rng.Uint64(),Slot:uint32(j%4),Family:families[rng.Intn(3)],Previous:families[rng.Intn(3)]})}
  filtered:=bombHeldEventsOf(changes,clock.deathOffsetMS);if filtered==nil{filtered=[]HeldObjectEvent{}}
  rows=append(rows,map[string]any{"events":events,"deaths":deaths,"bridge":run%13!=0,"output":out,"unknown":run%3==0,"clock":map[string]any{"origin_us":clock.origin,"step_us":clock.step,"frames":clock.frames,"death_offset_ms":clock.deathOffsetMS},"bombs":map[string]any{"carries":bombs,"coverage":cov},"changes":changes,"filtered":filtered})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-replay-held-object.json",raw,0600);e!=nil{t.Fatal(e)}
}
