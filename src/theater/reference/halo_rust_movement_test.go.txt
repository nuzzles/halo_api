 rng:=rand.New(rand.NewSource(594));runs:=[]any{};velocities:=[]any{}
 for i:=0;i<4096;i++ {dir:=uint64(rng.Intn(1<<19));scale:=uint64(i%1024);velocities=append(velocities,map[string]any{"dir":dir,"scale":scale,"vector":DecodeVelocity(dir,scale)})}

  sc:=&movementStateScanner{st:&types.MovementStateStats{},vit:map[uint32][]jumpVelSample{},vues:map[movementStateKey]types.MovementStateRead{}}
  input:=[]any{}
  for slot:=uint32(512);slot<515;slot++ {
   ts:=uint64(0)
   for i:=0;i<32;i++ {ts+=uint64(rng.Intn(400000));vz:=float64(rng.Intn(31)-10)/4;v:=jumpVelSample{ts:ts,vz:vz,chunk:run%7,paquet:i};sc.vit[slot]=append(sc.vit[slot],v);input=append(input,map[string]any{"slot":slot,"ts":ts,"vz":vz,"chunk":v.chunk,"packet":i});if i%7==0{sc.vit[slot]=append(sc.vit[slot],v);input=append(input,map[string]any{"slot":slot,"ts":ts,"vz":vz,"chunk":v.chunk,"packet":i})}}

  }

 if err=os.WriteFile("/private/tmp/halo-levelup-jump-derivation.json",data,0600);err!=nil{t.Fatal(err)}
}
