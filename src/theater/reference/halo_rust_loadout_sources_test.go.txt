package grammar

import (
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"os"
	"path/filepath"
	"testing"
)

func TestHaloRustLoadoutSources(t *testing.T) {
	var prior []struct {
		Files   map[string]string
		Missing bool
		Known   map[uint32]bool
	}
	raw, e := os.ReadFile("/private/tmp/halo-ground-directory.json")
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(raw, &prior); e != nil {
		t.Fatal(e)
	}
	rows := []any{}
	for run, c := range prior {
		dir := t.TempDir()
		if c.Missing {
			dir = filepath.Join(dir, "missing")
		}
		for name, h := range c.Files {
			b, _ := hex.DecodeString(h)
			if e = os.WriteFile(filepath.Join(dir, name), b, 0600); e != nil {
				t.Fatal(e)
			}
		}
		records, err := ScanFilmKeyframeLoadouts(dir, c.Known)
		if records == nil {
			records = []types.KeyframeLoadout{}
		}
		kind := ""
		if err != nil {
			kind = "load"
			if err == ErrNoReadableFilmChunk {
				kind = "unreadable"
			}
		}
		// Metadata order, duplicate numbers, missing prefix, role flags and partial buffers.
		chunks := source.MemoryChunks{[]byte{41, 0, 0, 0, 27, 0, 0, 0}}
		meta := []types.ChunkMeta{{Index: 0}}
		inputs := []any{}
		for i := 0; i < 3; i++ {
			b, _ := hex.DecodeString(c.Files["chunk_01.bin"])
			if i == 2 && len(b) > 3 {
				b = b[:len(b)-3]
			}
			idx := i + 1
			switch (run / 4) % 5 {
			case 1:
				if i == 1 {
					idx = 1
				}
			case 2:
				if i == 0 {
					idx = 2
				}
			case 3:
				if i == 1 {
					idx = 3
				}
			case 4:
				if i == 0 {
					idx = -1
				}
			}
			chunks = append(chunks, b)
			meta = append(meta, types.ChunkMeta{Index: idx, ChunkType: run % 3})
			inputs = append(inputs, map[string]any{"index": idx, "kind": run % 3, "hex": hex.EncodeToString(b)})
		}
		film, err := source.Load(chunks, meta)
		if err != nil {
			t.Fatal(err)
		}
		loaded, err := ScanKeyframeLoadouts(film, c.Known)
		if loaded == nil {
			loaded = []types.KeyframeLoadout{}
		}
		loadedErr := ""
		if err != nil {
			loadedErr = "unreadable"
		}
		rows = append(rows, map[string]any{"files": c.Files, "missing": c.Missing, "known": c.Known, "records": records, "error": kind, "chunks": inputs, "loaded": loaded, "loaded_error": loadedErr})
	}
	raw, e = json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-loadout-sources.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
