func TestHaloRustI0Layout(t *testing.T){
 rows:=[]any{};rng:=rand.New(rand.NewSource(72041))
 wire:=func(s i0Sample)any{return map[string]any{"slot":s.slot,"chunk":s.chunk,"packet":s.pkt,"bits":s.bits,"index_bit":s.indexBit}}
 for c:=0;c<512;c++{
 samples:=[]i0Sample{};input:=[]any{};for j:=0;j<c%200;j++{s:=i0Sample{slot:uint32(j%3),chunk:1+j/80,pkt:j/2,indexBit:uint8(j%3)};if c%3==0{s.chunk=1;s.pkt=j;s.slot=0};for k:=0;k<72;k++{on:=rng.Intn(2);if c%3==0&&(k<13||k>=19&&k<26||k>=32&&k<39||k>=45){on=0};if on==1{s.bits[k/64]|=1<<uint(63-k%64)}};samples=append(samples,s);input=append(input,wire(s))};rep:=profileI0(samples);bounds:=i0Boundaries(rep.FlipRate);if bounds==nil{bounds=[]int{}}
 pay:=make([]byte,40);rng.Read(pay);at:=c%8;put:=func(w int,v uint64){for k:=0;k<w;k++{bit:=at+k;mask:=byte(1<<uint(7-bit%8));pay[bit/8]&=^mask;if v&(uint64(1)<<uint(w-1-k))!=0{pay[bit/8]|=mask}};at+=w};put(1,1);put(13,512);put(2,uint64(c%4));put(2,0);put(3,2);put(6,0);put(6,1);put(4,0);if c%7==0{pay=pay[:6]};collected:=collectI0Samples(pay,NewSlotBand(map[uint32]bool{512:true,513:true}),c,19,nil);got:=[]any{};for _,s:=range collected{got=append(got,wire(s))}
 rows=append(rows,map[string]any{"samples":input,"records":rep.Records,"pairs":rep.Pairs,"rates":rep.FlipRate,"boundaries":bounds,"ones":rep.IndexBitOnes,"hex":hex.EncodeToString(pay),"chunk":c,"collected":got})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-i0-layout.json",b,0600);e!=nil{t.Fatal(e)}
}
