package replay
import("testing";"os";"encoding/json";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustFactsShots(t *testing.T){
 rng:=rand.New(rand.NewSource(42003));rows:=[]any{}
 for run:=0;run<1024;run++{
  pos:=[]grammar.BipedPosition{};events:=[]grammar.FireEvent{};owner:=map[uint32]int{};published:=[]Track{}
  for slot:=0;slot<8;slot++{if run%9!=0{owner[uint32(slot)]=[]int{-9223372036854775808,-1,0,255,256,4294967296,9223372036854775807}[slot%7]};if rng.Intn(3)!=0{published=append(published,Track{Slot:uint32(slot),StartFrame:100000,EndFrame:100001})}}
  for i:=0;i<run%80;i++{p:=grammar.BipedPosition{Slot:uint32(rng.Intn(8)),TimestampUS:uint64(rng.Intn(10))*200000,HasWorld:rng.Intn(4)!=0,X:float32(i)/317,Y:float32(i)/-313};pos=append(pos,p)}
  for i:=0;i<run%60;i++{e:=grammar.FireEvent{TimestampUS:uint64(rng.Intn(10))*200000+[]uint64{0,1,100000,120000,120001}[i%5],FilmIndex:[]int{-9223372036854775808,-1,0,255,256,4294967296,9223372036854775807,7}[rng.Intn(8)],WeaponID:rng.Uint64(),HasAim:rng.Intn(4)!=0,Aim:[3]float32{float32(rng.Intn(200)-100)/100,float32(rng.Intn(200)-100)/100,0},PacketIndex:i};if i%7==0{e.WeaponID=0};if i%13==0{e.Aim=[3]float32{1,0,0}};events=append(events,e)}
  origin:=uint64(run%3)*500000;step:=[]uint64{1,100000,1000000}[run%3]
  shots,orphans,cov:=buildShots(pos,events,origin,step,owner);before:=append([]Shot{},shots...);oc:=[]any{};for _,o:=range orphans{reason:="NoSlot";if o.reason==reasonOutOfWindow{reason="OutOfWindow"};oc=append(oc,map[string]any{"event":o.ev,"reason":reason})};rawCov:=cov;shots=keepShotsOfPublishedTracks(shots,published);cov.Unpublished=countUnpublished(len(before),len(shots));cov.Attached=len(shots);if shots==nil{shots=[]Shot{}}
  rows=append(rows,map[string]any{"positions":pos,"events":events,"owners":owner,"origin":origin,"step":step,"published":published,"raw":map[string]any{"shots":before,"orphans":oc,"coverage":rawCov},"output":map[string]any{"shots":shots,"orphans":oc,"coverage":cov}})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-facts-shots.json",raw,0600);e!=nil{t.Fatal(e)}
}
