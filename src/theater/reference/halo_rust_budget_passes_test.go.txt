func TestHaloRustBudgetPasses(t *testing.T) {
	old := slog.Default()
	defer slog.SetDefault(old)
	rows := []any{}
	for run := 0; run < 64; run++ {
		recs := []types.StatRecord{}
		for round := 0; round < 1+run%3; round++ {
			for slot := 6; slot <= 24; slot += 2 {
				for j := 0; j < 24; j++ {
					comps := map[int]types.StatValue{}
					for _, c := range []int{0, 2, 3, 12, 20, 21, 22, 23, 24} {
						v := int64(j)
						if run%3 != 0 {
							v *= int64(17 + (slot+c+run)%5)
						}
						comps[c] = types.StatValue{A: v, B: v / 2}
					}
					recs = append(recs, types.StatRecord{TimeMS: round*30000 + j*1000, Slot: slot, Round: round, Comps: comps})
				}
			}
		}
		if run%11 == 0 {
			recs = []types.StatRecord{}
		}
		lines := []types.PlayerLine{}
		for i := 0; i < 8; i++ {
			lines = append(lines, types.PlayerLine{XUID: fmt.Sprint(i + 1), Kills: 23, Deaths: 11, Assists: 23})
		}
		mode := []string{"flag", "zone", "vip", "bomb", "hill", "ball", ""}[run%7]
		h := &rustBudgetHandler{records: []any{}}
		slog.SetDefault(slog.New(h))
		events := NamedEventsFrom(recs, mode)
		if events == nil {
			events = []NamedEvent{}
		}
		eventLogs := h.records
		h.records = []any{}
		identity := SlotIdentityFrom(recs, lines)
		if identity == nil {
			identity = map[int]string{}
		}
		identityLogs := h.records
		h.records = []any{}
		cross := crossCheckFrom(recs, mode)
		if cross == nil {
			cross = map[int]map[string][2]int{}
		}
		rows = append(rows, map[string]any{"records": recs, "lines": lines, "mode": mode, "events": events, "event_logs": eventLogs, "identity": identity, "identity_logs": identityLogs, "cross": cross, "cross_logs": h.records})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-budget-passes.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}

// Exhaust the real one-million-event pass budget, not a reduced test budget.
func TestHaloRustBudgetExhaustion(t *testing.T) {
	old := slog.Default()
	defer slog.SetDefault(old)
	recs := []types.StatRecord{}
	for j := 0; j < 62503; j++ {
		v := int64(j * 16)
		recs = append(recs, types.StatRecord{TimeMS: j * 10, Slot: 10, Round: 0, Comps: map[int]types.StatValue{2: {A: v, B: v}, 3: {A: v}, 12: {A: v, B: v}}})
	}
	h := &rustBudgetHandler{records: []any{}}
	slog.SetDefault(slog.New(h))
	events := NamedEventsFrom(recs, "flag")
	if len(events) != 1000000 {
		t.Fatalf("expected exhaustion, got %d events", len(events))
	}
	digest := fnv.New64a()
	var word [8]byte
	for _, e := range events {
		for _, v := range []int{e.TimeMS, e.Slot, e.Comp} {
			binary.LittleEndian.PutUint64(word[:], uint64(v))
			digest.Write(word[:])
		}
		digest.Write([]byte(e.Stat))
		digest.Write([]byte{0})
		digest.Write([]byte(e.Side))
		digest.Write([]byte{0})
	}
	eventLogs := h.records
	h.records = []any{}
	identity := SlotIdentityFrom(recs, []types.PlayerLine{{XUID: "exhaustion", Kills: 1000000}})
	identityLogs := h.records
	h.records = []any{}
	cross := crossCheckFrom(recs, "flag")
	raw, e := json.Marshal(map[string]any{"event_count": len(events), "event_digest": digest.Sum64(), "event_logs": eventLogs, "identity": identity, "identity_logs": identityLogs, "cross": cross, "cross_logs": h.records})
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-budget-exhaustion.json", raw, 0600); e != nil {
		t.Fatal(e)
	}
}
