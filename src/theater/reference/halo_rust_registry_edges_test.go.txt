package grammar

import (
 "encoding/binary"
 "encoding/hex"
 "encoding/json"
 "os"
 "testing"
)

func TestHaloRustRegistryEdges(t *testing.T) {
 rows:=[]any{}
 for i:=0;i<256;i++ {
  data:=make([]byte,8+(i%4)*64*260+i%513)
  binary.LittleEndian.PutUint32(data,41)
  binary.LittleEndian.PutUint32(data[4:],27)
  for b:=0;b<i%4;b++ {
   n:=[]int{0,1,2,63,64}[((i/4)+b)%5]
   for s:=0;s<n;s++ {
    off:=8+(b*64+s)*260
    length:=[]int{1,17,255,256}[(i+s)%4]
    for k:=0;k<length;k++ {data[off+k]=byte(32+(i+s+k)%95)}
    binary.LittleEndian.PutUint32(data[off+256:],uint32(i)*0xffffff+uint32(s))
   }
   if i%7==0 && n<64 {data[8+(b*64+n)*260+259]=1}
   if i%11==0 && n>0 {data[8+b*64*260]=0xff}
  }
  reg,err:=ParseRegistryChunk(data);if err!=nil {t.Fatal(err)}
  rows=append(rows,map[string]any{"data":hex.EncodeToString(data),"fingerprint":RegistryFingerprint(reg),"slots":RegistryNamedSlots(reg),"blocks":len(reg.Archetypes),"truncated":reg.Truncated})
 }
 data,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-registry-edges.json",data,0644);err!=nil {t.Fatal(err)}
}
