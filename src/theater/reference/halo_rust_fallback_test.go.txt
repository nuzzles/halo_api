// Capture the native structured diagnostics independently of report aggregation.
type haloRustFallbackLog struct{ unknown []Declenchement }

func (h *haloRustFallbackLog) Enabled(context.Context, slog.Level) bool { return true }
func (h *haloRustFallbackLog) Handle(_ context.Context, r slog.Record) error {
	if r.Level == slog.LevelError {
		var d Declenchement
		r.Attrs(func(a slog.Attr) bool {
			if a.Key == "repli" {
				d.Nom = Nom(a.Value.String())
			}
			if a.Key == "declenchements" {
				d.Declenchements = int(a.Value.Int64())
			}
			return true
		})
		h.unknown = append(h.unknown, d)
	}
	return nil
}
func (h *haloRustFallbackLog) WithAttrs([]slog.Attr) slog.Handler { return h }
func (h *haloRustFallbackLog) WithGroup(string) slog.Handler      { return h }
func TestHaloRustFallback(t *testing.T) {
	previous := slog.Default()
	defer slog.SetDefault(previous)
	handler := &haloRustFallbackLog{}
	slog.SetDefault(slog.New(handler))
	table := Table()
	rows := []any{}
	rng := rand.New(rand.NewSource(941))
	for i := 0; i < 512; i++ {
		c := NouveauCompteur()
		ops := []Declenchement{}
		for j := 0; j < i%31; j++ {
			name := table[rng.Intn(len(table))].Nom
			if j%13 == 0 {
				name = "unknown"
			}
			n := rng.Intn(12) - 3
			ops = append(ops, Declenchement{Nom: name, Declenchements: n})
		}
		handler.unknown = []Declenchement{}
		c.Cumuler(ops)
		report := c.Rapport()
		rows = append(rows, map[string]any{"unknown_diagnostics": handler.unknown, "ops": ops, "report": report, "text": Texte(report)})
	}
	packages := []string{}
	conditions := [][]Condition{}
	for _, r := range table {
		packages = append(packages, r.Paquet())
		cs := []Condition{}
		for _, s := range r.Sites {
			cs = append(cs, r.ConditionEffective(s))
		}
		conditions = append(conditions, cs)
	}
	invalid := []any{}
	for i := 0; i < 512; i++ {
		r := table[i%len(table)]
		r.Sites = append([]Site{}, r.Sites...)
		if i%2 == 0 {
			r.Nom = "Bad_name"
		}
		if i%3 == 0 {
			r.Fait = "  "
		}
		if i%5 == 0 {
			r.DatePose = "2026-99-99"
		}
		if i%7 == 0 {
			r.DatePose = "bad"
		}
		if i%11 == 0 {
			r.Condition = "unknown"
		}
		if i%13 == 0 {
			r.Ordre = "unknown"
		}
		if i%17 == 0 {
			r.Sites = nil
		} else if len(r.Sites) > 0 {
			if i%3 == 0 {
				r.Sites[0].Ancre = ""
			}
			if i%5 == 0 {
				r.Sites[0].Condition = r.Condition
			}
		}
		if i%19 == 0 {
			r.CompteurBranche = !r.CompteurBranche
		}
		seen := map[Nom]bool{r.Nom: i%2 == 0}
		problems := verifierUneEntree(r, seen)
		if problems == nil {
			problems = []string{}
		}
		sort.Strings(problems)
		invalid = append(invalid, map[string]any{"entry": r, "duplicate": i%2 == 0, "problems": problems})
	}
	out := map[string]any{"invalid": invalid, "table": table, "families": Tranches(), "before_read": NbDevantLaLecture(), "problems": VerifierRegistre(), "packages": packages, "conditions": conditions, "cases": rows}
	b, e := json.Marshal(out)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-fallback.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
