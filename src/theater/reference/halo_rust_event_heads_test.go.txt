func TestHaloRustEventHeadsParity(t *testing.T) {
 rows:=[]any{}
 add:=func(data []byte, source string) {
  kind, present:=PacketHeadEventType(data);if !present{return}
  var domains []int
  switch kind {case 8:domains=[]int{2,3,7};case 9:domains=[]int{2,7,8};case 21:domains=[]int{4,8,7};case 22:domains=[]int{1,1,7};case 103:domains=[]int{0,0,7};default:return}
  refs:=[]any{};at:=9
  for _,domain:=range domains {start:=at;var ref guardedRef;if domain==1 {ref=readDom1Ref(data,at)}else{ref=readPlainRef(data,at,int(refDomWidth(domain)))};at=ref.EndBit
   refs=append(refs,map[string]any{"domain":domain,"start":start,"ref":ref})
  }
  if kind==9 {at+=3;hasCatalog:=readBitsAt(data,at,1)!=0;at++;if hasCatalog{at+=33}};if kind==8||kind==22{at+=6};if kind==21{at+=2};if at>len(data)*8{t.Fatalf("truncated reference source %s",source)}
  row:=map[string]any{"source":source,"kind":kind,"end":at,"refs":refs,"hex":hex.EncodeToString(data[:(at+7)/8])}

  if kind==9 {var st types.BipedPickupStats;ev,ok:=decodeBipedPickup(data,&st);row["pickup"]=ev;row["pickup_valid"]=ok;row["pickup_stats"]=st}
  if kind==21 {ev,ok:=decodeZoomHead(data,1000);row["zoom"]=ev;row["zoom_valid"]=ok}
  if kind==103 {ev,_,ok:=decodeEquipmentSpawnEvent(data);if !ok{t.Fatal("spawn refused")};row["spawn"]=ev}
  rows=append(rows,row)
 }
 rng:=rand.New(rand.NewSource(41))
 for _,kind:=range []int{8,9,21,22,103} {for i:=0;i<256;i++ {d:=make([]byte,64);if i!=0 {rng.Read(d)};d[0]=0xc0|byte(kind>>1);d[1]=(d[1]&0x7f)|byte(kind&1)<<7;add(d,"synthetic")}}
