package grammar

import (
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"math"
	"math/big"
	"os"
	"path/filepath"
	"testing"
)

func TestHaloRustMapRounding(t *testing.T) {
	tokens := []string{"0", "-0", "0.0", "-0.0", "1e-999", "-1e-999", "1e999", "-1e999", "3.4028234663852886e38", "3.4028235677973367e38", "3.4028235677973366e38"}
	// Exact rational midpoints of adjacent float32 values. Perturb by a power
	// of two far below a double-precision ULP, so parsing through float64 can
	// round twice. Rust never supplies these expectations.
	seeds := []uint32{0, 1, 2, 0x007ffffe, 0x007fffff, 0x00800000, 0x3f7fffff, 0x3f800000, 0x3f800001, 0x7f7ffffe}
	for exponent := uint32(0); exponent < 255; exponent++ {
		seeds = append(seeds, exponent<<23|0x345678)
	}
	for _, bits := range seeds {
		a := new(big.Rat).SetFloat64(float64(math.Float32frombits(bits)))
		b := new(big.Rat).SetFloat64(float64(math.Float32frombits(bits + 1)))
		midpoint := new(big.Rat).Quo(new(big.Rat).Add(a, b), big.NewRat(2, 1))
		epsilon := new(big.Rat).Quo(new(big.Rat).Sub(b, a), new(big.Rat).SetInt(new(big.Int).Lsh(big.NewInt(1), 64)))
		for _, delta := range []int64{-1, 0, 1} {
			value := new(big.Rat).Add(midpoint, new(big.Rat).Mul(epsilon, big.NewRat(delta, 1)))
			tokens = append(tokens, value.FloatString(220), new(big.Rat).Neg(value).FloatString(220))
		}
	}
	rows := []any{}
	path := filepath.Join(t.TempDir(), "catalog.json")
	for _, token := range tokens {
		input := `{"schemaVersion":1,"maps":{"a":{"min":[` + token + `,0,0]}}}`
		if err := os.WriteFile(path, []byte(input), 0600); err != nil {
			t.Fatal(err)
		}
		c, err := profile.LoadMapQuantCatalog(path)
		var bits uint32
		if err == nil {
			e, lookupErr := c.Lookup("a")
			if lookupErr != nil {
				t.Fatal(lookupErr)
			}
			bits = math.Float32bits(e.Min[0])
		}
		rows = append(rows, map[string]any{"token": token, "accepted": err == nil, "bits": bits})
	}
	raw, err := json.Marshal(rows)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile("/private/tmp/halo-map-rounding.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
