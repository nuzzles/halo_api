package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustMarchLoaded(t *testing.T) {
	rng := rand.New(rand.NewSource(419512))
	rows := []any{}
	for run := 0; run < 256; run++ {
		p := ProfilDeBalayageParDefaut()
		p.Grammaire.GenerationStricte = run%2 == 0
		p.Grammaire.ControleDeCorruption = run%3 == 0
		p.Grammaire.SimStateComplet = run%5 == 0
		p.MPP = profile.MPPWidths{Lead: 6 + run%7, Index: 2 + run%6}
		reg := &Registry{Archetypes: make([]Archetype, 41)}
		for i := range reg.Archetypes {
			reg.Archetypes[i].Index = i
			reg.Archetypes[i].Components = []string{}
			reg.Archetypes[i].Levels = []uint32{}
		}
		reg.Archetypes[3].Components = []string{"high-frequency"}
		reg.Archetypes[3].Levels = []uint32{1}
		for _, ti := range []int{35, 40} {
			reg.Archetypes[ti].Components = []string{"object-dead-state-component", "object-parent-state-component", "not-ported"}
			reg.Archetypes[ti].Levels = []uint32{0, 0, 0}
		}
		raw := make([]byte, 8)
		binary.LittleEndian.PutUint32(raw, 41)
		binary.LittleEndian.PutUint32(raw[4:], 27)
		chunks := source.MemoryChunks{raw}
		meta := []types.ChunkMeta{{Index: 0, ChunkType: 1}}
		inputs := []any{}
		packet := func(kind uint16, ts uint64, pay []byte) []byte {
			h := make([]byte, 16)
			binary.LittleEndian.PutUint16(h, kind)
			binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
			binary.LittleEndian.PutUint64(h[8:], ts)
			return append(h, pay...)
		}
		count := 3
		if run%13 == 0 {
			count = 0
		}
		for c := 0; c < count; c++ {
			data := []byte{}
			if (run+c)%7 != 0 {
				w := &translocWriter{}
				w.put(1, 0)
				for _, key := range [][3]uint64{{123, 1, 3}, {512, 1, 35}, {513, 2, 40}} {
					w.put(32, (key[1]<<30)|key[0])
					w.put(32, key[2])
					w.put(44, 0)
					w.put(32, 0)
					w.put(32, 0)
				}
				w.put(64, 0)
				data = append(data, packet(2, uint64((2-c)*1000), w.pay)...)
			}
			n := 8
			if run%11 == 0 {
				n = 0
			}
			for j := 0; j < n; j++ {
				w := &translocWriter{}
				w.put(1, 1)
				events := (run+j)%3 != 0
				if events {
					w.put(1, 1)
					w.put(10, 0)
					w.put(1, 1)
					w.put(13, 123)
					w.put(2, 1)
					w.put(1, 0)
					w.put(1, 0)
					w.put(3, 1)
					w.put(6, 0)
					w.put(8, uint64(j))
					if p.Grammaire.ControleDeCorruption {
						w.put(1, 0)
					}
				} else {
					w.put(1, 0)
				}
				w.put(1, 1)
				w.put(13, uint64(512+j%2))
				w.put(2, uint64(1+j%2))
				w.put(1, 0)
				w.put(1, 1)
				w.put(64, uint64(1+j%7))
				for k := 0; k < 20; k++ {
					w.put(64, rng.Uint64())
				}
				pay := w.pay
				if (run+j)%9 == 0 {
					pay = pay[:12]
				}
				data = append(data, packet(0, uint64(j%4*1000), pay)...)
			}
			index := c + 1
			if run%17 == 0 {
				index = c + 2
			}
			if run%19 == 0 && c == 1 {
				index = 1
			}
			chunks = append(chunks, data)
			meta = append(meta, types.ChunkMeta{Index: index})
			inputs = append(inputs, map[string]any{"index": index, "hex": hex.EncodeToString(data)})
		}
		film, e := source.Load(chunks, meta)
		if e != nil {
			t.Fatal(e)
		}
		fc := NewFilmContext(film)
		fc.reg = reg
		fc.regLu = true
		fc.bal = p
		fc.corr = p.Grammaire.ControleDeCorruption
		fc.corrLu = true
		facts, e := ScanMarchFacts(fc)
		if e != nil {
			t.Fatal(e)
		}
		if facts.Deaths == nil {
			facts.Deaths = []types.ObjectDeath{}
		}
		if facts.Occupancy == nil {
			facts.Occupancy = []types.VehicleOccupancy{}
		}
		cfg := fc.CadreDeBalayage()
		encoding := map[string]any{"index_bits": p.Mouvement.WorldObject.IndexW, "world_axis_bits": p.Mouvement.WorldObject.AxisW, "handle_bits": p.Mouvement.Traversal.IndexW, "traversal_axis_bits": p.Mouvement.Traversal.AxisW, "default_axis_bits": profile.LargeursAxeParDefautDuBuild(16), "region_axis_bits": map[string]any{}, "delta_axis_bits": [3]uint{p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth}, "full_precision": p.Mouvement.FullPrecision, "writer_absolute": p.Grammaire.GrammaireEcrivainI0, "delta_handle_tail": p.Mouvement.DeltaHasHandleTail}
		rows = append(rows, map[string]any{"chunks": inputs, "facts": facts, "base_low": cfg.IDLowBits, "base_id": cfg.IDBase, "extra": cfg.HasExtraFields, "strict": p.Grammaire.GenerationStricte, "simulation": p.Grammaire.SimStateComplet, "corruption": p.Grammaire.ControleDeCorruption, "mpp": [2]int{p.MPP.Lead, p.MPP.Index}, "position": encoding})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-march-loaded.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
