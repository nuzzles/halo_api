package replay
import("testing";"os";"encoding/json";"math/rand";"math")
func TestHaloRustGroundPads(t *testing.T){
 rng:=rand.New(rand.NewSource(7542));rows:=[]any{}
 for run:=0;run<512;run++{
  app:=[]gwPadApparition{};objs:=[]gwPickupObject{};members:=[]int{}
  for i:=0;i<run%129;i++{a:=gwPadApparition{Kind:[]string{"weapon","powerup"}[rng.Intn(2)],Family:[]string{"a","b"}[rng.Intn(2)],X:float32(rng.Intn(30))/10,Y:float32(rng.Intn(30))/10,Z:float32(rng.Intn(3))/10,TUS:uint64(rng.Intn(10))*1000000,Class:[]string{"spawned","dropped"}[rng.Intn(2)],HasDelta:rng.Intn(2)==0};app=append(app,a);status:=[]string{"dated","never","unknown"}[rng.Intn(3)];objs=append(objs,gwPickupObject{Appar:a,FamilyID:rng.Uint32(),Status:status,Picker:gwPickupHit{TUS:uint64(rng.Intn(10))*1000000,Found:status=="dated"}});members=append(members,i)}
  clusters,assign:=gwPadsClusterAssign(app);kept,src:=gwPadsKeep(clusters);gaps,missing:=gwPickupPadGaps(objs,members)
  cycleGaps:=[]float64{};for i:=0;i<run%12;i++{v:=float64(rng.Intn(100))/10;if run%3==0{v=10+v/100};cycleGaps=append(cycleGaps,v)};cycle:=gwPadsCycleFromGaps(cycleGaps)
  if clusters==nil{clusters=[]gwPadCluster{}};if gaps==nil{gaps=[]float64{}}
  counts:=padChainCounts{};rest,source:=gwAtRestOf(objs,&counts);cs,as:=gwPadsClusterAssign(rest);retain,orig:=gwPadsKeep(cs);memberMap:=map[int][]int{};for j,idx:=range as{memberMap[idx]=append(memberMap[idx],source[j])};pads:=[]WeaponPad{};pickups:=[]PadPickup{};cov:=GroundWeaponCoverage{};clock:=replayClock{origin:2000000,step:100000,frames:40};for i,c:=range retain{pad,picks:=gwBuildPad(c,objs,memberMap[orig[i]],clock,&cov);for j:=range picks{picks[j].Pad=len(pads)};pads=append(pads,pad);pickups=append(pickups,picks...)}
  layer:=map[string]any{"pads":pads,"pickups":pickups,"counts":map[string]any{"dropped":counts.dropped,"spawned":counts.spawned,"atRest":counts.atRest,"clusters":len(cs),"pads":len(pads),"occupancies":cov.Occupancies,"dated":cov.Dated,"unknown":cov.Unknown,"never":cov.Never,"cycles":cov.Cycles}}
  rows=append(rows,map[string]any{"layer":layer,"appearances":app,"clusters":clusters,"assignments":assign,"kept":kept,"source":src,"objects":objs,"members":members,"gaps":gaps,"missing":missing,"cycle_gaps":cycleGaps,"cycle":cycle,"cycle_bits":[]uint64{math.Float64bits(cycle.MedianS),math.Float64bits(cycle.P10S),math.Float64bits(cycle.P90S),math.Float64bits(cycle.SDS)}})
 }
 d,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-ground-pads.json",d,0600);e!=nil{t.Fatal(e)}
}

func TestHaloRustGroundPadCorpus(t *testing.T){
 raw,e:=os.ReadFile("/private/tmp/halo-ground-objects-corpus.json");if e!=nil{t.Fatal(e)}
 var films []struct{Folder string `json:"folder"`;Objects []gwPickupObject `json:"objects"`};if e=json.Unmarshal(raw,&films);e!=nil{t.Fatal(e)};rows:=[]any{}
 for _,f:=range films{
  origin:=uint64(0);if len(f.Objects)>0{origin=f.Objects[0].Appar.TUS};clock:=replayClock{origin:origin,step:100000,frames:10000}
  counts:=padChainCounts{};rest,source:=gwAtRestOf(f.Objects,&counts);cs,assign:=gwPadsClusterAssign(rest);keep,src:=gwPadsKeep(cs);members:=map[int][]int{};for j,a:=range assign{members[a]=append(members[a],source[j])};pads:=[]WeaponPad{};pickups:=[]PadPickup{};cov:=GroundWeaponCoverage{}
  for i,c:=range keep{p,ps:=gwBuildPad(c,f.Objects,members[src[i]],clock,&cov);for j:=range ps{ps[j].Pad=len(pads)};pads=append(pads,p);pickups=append(pickups,ps...)}
  rows=append(rows,map[string]any{"folder":f.Folder,"clock":map[string]any{"origin_us":origin,"step_us":100000,"frames":10000},"layer":map[string]any{"pads":pads,"pickups":pickups,"counts":map[string]any{"dropped":counts.dropped,"spawned":counts.spawned,"atRest":counts.atRest,"clusters":len(cs),"pads":len(pads),"occupancies":cov.Occupancies,"dated":cov.Dated,"unknown":cov.Unknown,"never":cov.Never,"cycles":cov.Cycles}}})
 }
 raw,e=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-ground-pad-corpus.json",raw,0600);e!=nil{t.Fatal(e)}
}
