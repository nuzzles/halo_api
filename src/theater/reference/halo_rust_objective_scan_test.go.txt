package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
	"levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
 "reflect"
	"testing"
)

func TestHaloRustObjectiveScan(t *testing.T) {
	rng := rand.New(rand.NewSource(411106))
	rows := []any{}
	for run := 0; run < 256; run++ {
		names := []string{"managed-objective-timers-component", "managed-objective-object-reference-component", "managed-objective-type-component", "managed-objective-progress-component", "managed-objective-required-progress-component", "managed-objective-state-component"}
		if run%4 == 0 {
			names[2] = "not-ported"
		}
		if run%7 == 0 {
			names[4] = ""
		}
		reg := &Registry{Archetypes: make([]Archetype, 13)}
		for i := range reg.Archetypes {
			reg.Archetypes[i].Index = i
		}
		reg.Archetypes[11] = Archetype{Index: 11, Components: names, Levels: make([]uint32, len(names))}
		p := ProfilDeBalayageParDefaut()
		p.Grammaire.ControleDeCorruption = run%3 == 0
		p.Grammaire.SimStateComplet = run%2 == 0
		p.Grammaire.LargeursCalibrees = map[string]int{}
		p.Grammaire.LargeursBouchon = map[string]int{}
		if run%5 == 0 {
			p.Grammaire.LargeursCalibrees[names[2]] = run % 18
		}
		if run%9 == 0 {
			p.Grammaire.LargeursBouchon[names[2]] = run % 7
		}
		raw := make([]byte, 8)
		binary.LittleEndian.PutUint32(raw, 41)
		binary.LittleEndian.PutUint32(raw[4:], 27)
		chunks := source.MemoryChunks{raw}
		meta := []types.ChunkMeta{{Index: 0, ChunkType: 1}}
		inputs := []any{}
		packet := func(kind uint16, ts uint64, pay []byte) []byte {
			h := make([]byte, 16)
			binary.LittleEndian.PutUint16(h, kind)
			binary.LittleEndian.PutUint32(h[4:], uint32(len(pay)))
			binary.LittleEndian.PutUint64(h[8:], ts)
			return append(h, pay...)
		}
		for c := 0; c < 3; c++ {
			pay := make([]byte, 8192)
			rng.Read(pay)
			put := func(at, n int, v uint64) {
				for j := 0; j < n; j++ {
					m := byte(1 << uint(7-(at+j)%8))
					pay[(at+j)/8] &= ^m
					if v&(uint64(1)<<uint(n-1-j)) != 0 {
						pay[(at+j)/8] |= m
					}
				}
			}
			pos := 1
			ctx := ContexteParDefaut()
			ctx.Profil = p
			for j := 0; j < 6; j++ {
				stride := 1
				if run%3 == 1 {
					stride = 2
				}
				put(pos, 32, 0x40000200+uint64(j*stride))
				ti := uint64(11)
				if j == 3 && run%5 == 0 && c == 1 {
					ti = 12
				}
				put(pos+32, 32, ti)
				put(pos+64, 44, 0)
				put(pos+108, 32, 0)
				n2 := uint64(1)
				if (run+j)%6 == 0 {
					n2 = 0
				}
				put(pos+140, 32, n2)
				tr := WalkKeyframeFullState(pay, pos, reg, ctx)
				pos = tr.EndBit
				if tr.DesyncAt >= 0 {
					pos += 32
				}
				if j%3 == 0 {
					pos += 7
				}
			}
			pay = pay[:(pos+7)/8]
			if run%11 == 0 {
				pay = pay[:len(pay)-3]
			}
			data := []byte{}
			if run%17 != 0 {
				data = append(data, packet(2, uint64((2-c)*1000), pay)...)
			}
			for j := 0; j < 3; j++ {
				w := &translocWriter{}
				w.put(run%8, 0)
				for r := 0; r < 7; r++ {
					w.put(1, 1)
					w.put(13, uint64(512+r))
					w.put(2, uint64(r%4))
					w.put(2, 0)
					n := 1 + r%7
					w.put(3, uint64(n))
					for k := 0; k < n; k++ {
						w.put(6, uint64(k))
					}
					for k := 0; k < n; k++ {
						w.put(64, rng.Uint64())
					}
				}
				if run%13 == 0 {
					w.pay = w.pay[:len(w.pay)-4]
				}
				data = append(data, packet(0, uint64(j%2*1000), w.pay)...)
			}
			chunks = append(chunks, data)
			meta = append(meta, types.ChunkMeta{Index: c + 1})
			inputs = append(inputs, map[string]any{"index": c + 1, "hex": hex.EncodeToString(data)})
		}
		if run%19 == 0 {
			reg.Archetypes = reg.Archetypes[:11]
		}
		if run%23 == 0 {
			chunks = chunks[:1]
			meta = meta[:1]
			inputs = []any{}
		}
		film, e := source.Load(chunks, meta)
		if e != nil {
			t.Fatal(e)
		}
		fc := NewFilmContext(film)
		fc.reg = reg
		fc.regLu = true
		fc.bal = p
		fc.corr = p.Grammaire.ControleDeCorruption
		fc.corrLu = true
		sc, scanErr := ScanObjectives(fc)
        emissions := []any{}
        if scanErr == nil {
            traced := ObjectiveScan{Slots:sc.Slots}
            w := objectiveWalk{prof:fc.ProfilDeBalayage(),arch:reg.Archetypes[11],reg:reg,sc:&traced}
            w.obs = w.install()
            original := w.obs.ObjectiveHook
            chunkNum, packetNum := 0, 0
            w.obs.ObjectiveHook = func(f ObjectiveField, values []uint64) {
                emissions = append(emissions,map[string]any{"chunk":chunkNum,"packet":packetNum,"keyframe":w.fromKeyframe,"field":f,"values":append([]uint64{},values...)})
                original(f,values)
            }
            band := observedSlotBand(fc.Film(), ObjectiveTypeIndex)
            for _, c := range fc.ChunkNumbers() {
                data,pks,ok:=fc.ChunkAt(c);if !ok {continue}
                for index,pk:=range pks {
                    chunkNum,packetNum=c,index
                    switch pk.Type {
                    case PacketTypeDelta: w.scanPayload(pk.Payload(data),band,pk.TimestampUS,&traced)
                    case PacketTypeKeyframe: w.scanKeyframe(pk.Payload(data),pk.TimestampUS,&traced)
                    }
                }
            }
            if !reflect.DeepEqual(sc,traced) {t.Fatal("instrumented scan changed outputs")}
        }

		reads := []any{}
		for _, r := range sc.Reads {
			reads = append(reads, map[string]any{"Slot": r.Slot, "TimestampUS": r.TimestampUS, "Field": r.Field.String(), "FieldCode": r.Field, "Value": r.Value, "ValueB": r.ValueB, "HasB": r.HasB, "Chained": r.Chained, "FromKeyframe": r.FromKeyframe})
		}
		sc.Reads = nil
		output := map[string]any{}
		b, _ := json.Marshal(sc)
		json.Unmarshal(b, &output)
		output["Reads"] = reads
		rows = append(rows, map[string]any{"names": names, "missing_arch": len(reg.Archetypes) < 12, "check": p.Grammaire.ControleDeCorruption, "simulation": p.Grammaire.SimStateComplet, "widths": map[string]any{"calibrated": p.Grammaire.LargeursCalibrees, "stubs": p.Grammaire.LargeursBouchon}, "chunks": inputs, "output": output, "error": scanErr != nil,"emissions":emissions})
	}
	b, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile("/private/tmp/halo-objective-scan.json", b, 0600); e != nil {
		t.Fatal(e)
	}
}
