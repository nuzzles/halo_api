package replay

import (
 "bytes"
 "encoding/json"
 "log/slog"
 "math/rand"
 "os"
 "reflect"
 "sort"
 "testing"
)

func TestHaloRustVehicleCoverageLogs(t *testing.T) {
 previous := slog.Default(); defer slog.SetDefault(previous)
 rng := rand.New(rand.NewSource(419245))
 rows := []any{}
 for run := 0; run < 513; run++ {
  var cov *VehicleCoverage
  if run > 0 {
   cov = &VehicleCoverage{}
   v := reflect.ValueOf(cov).Elem()
   for i := 0; i < v.NumField(); i++ { if v.Field(i).Kind() == reflect.Int { v.Field(i).SetInt(int64(rng.Intn(200))) } }
   cov.Scanned = run%2 == 0
   cov.Rides = []int{-1,0,1,2}[run%4]
   cov.RidesWithAim = []int{-1,0,1}[run%3]
   cov.WithChassis = []int{-1,0,1,2}[run/4%4]
   cov.FamilyResolved = []int{-1,0,1}[run/3%3]
   cov.UnknownChassis = map[string]int{}
   for i := 0; i < run%4; i++ { cov.UnknownChassis[[]string{"z", "0xFFFFFFFF", "a"}[i]] = run+i }
   if run == 1 { cov = &VehicleCoverage{} }
  }
  var buf bytes.Buffer
  slog.SetDefault(slog.New(slog.NewJSONHandler(&buf, nil)))
  logVehicleCoverage(cov)
  logs := []map[string]any{}
  dec := json.NewDecoder(&buf)
  for dec.More() { e := map[string]any{}; if err := dec.Decode(&e); err != nil {t.Fatal(err)}; delete(e, "time"); logs = append(logs,e) }
  // Only the native map traversal is unordered; retain all surrounding order.
  first, last := -1, -1
  for i, e := range logs { if _, ok := e["chassis"]; ok { if first < 0 {first=i}; last=i } }
  if first >= 0 { sort.Slice(logs[first:last+1],func(i,j int) bool { return logs[first+i]["chassis"].(string) < logs[first+j]["chassis"].(string) }) }
  rows = append(rows, map[string]any{"coverage":cov, "logs":logs})
 }
 raw,err:=json.Marshal(rows);if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/halo-vehicle-coverage-logs.json",raw,0600);err!=nil {t.Fatal(err)}
}
