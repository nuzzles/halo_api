package grammar

import (
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"levelup/go-api/internal/games/halo_infinite/film/internal/profile"
	"levelup/go-api/internal/games/halo_infinite/film/internal/source"
 "levelup/go-api/internal/games/halo_infinite/film/types"
	"math/rand"
	"os"
	"testing"
)

func TestHaloRustInventorySourceAvailability(t *testing.T) { haloRustInventorySources(t, false) }
func TestHaloRustInventoryDuplicateSources(t *testing.T) { haloRustInventorySources(t, true) }
func haloRustInventorySources(t *testing.T, duplicates bool) {
	rng := rand.New(rand.NewSource(42016))
	rows := []any{}
	for c := 0; c < 512; c++ {
		names := make([]string, 64)
		for id, name := range map[int]string{22: "unit-grenade-counts-component", 34: "weapon-state-type-info", 35: "weapon-state-ammo", 36: "weapon-state-rounds-inventory", 47: "biped-desired-grenade-set-component", 26: "unit-equipment-component", 28: camoComponentName, 48: "biped-desired-ability-set-component", 56: "biped-spartan-ability-energy-component", 57: "biped-spartan-ability-component", 59: "biped-spartan-ability-non-predicted-state-component"} {
			names[id] = name
		}
		arch := Archetype{Index: 35, Components: names, Levels: make([]uint32, 64)}
		ids := [][]int{{0, 22}, {0, 47}, {0, 34}, {0, 35, 36}, {0, 22, 34, 35, 36, 47}}[c%5]
		w := &bitWriter{}
		w.bits(0, c%8)
		w.bits(1, 1)
		w.bits(512, 13)
		w.bits(1, 2)
		w.bits(0, 2)
		w.bits(uint64(len(ids)), 3)
		for _, id := range ids {
			w.bits(uint64(id), 6)
		}
		w.bits(0, cliffLayout.GateBits)
		for _, width := range cliffLayout.AxisW {
			w.bits(1, int(width))
		}
		w.bits(0, 2)
		tail := make([]byte, 64)
		if c%8 >= 4 {
			rng.Read(tail)
		}
		for _, b := range tail {
			w.bits(uint64(b), 8)
		}
		raw := make([]byte, 16)
		binary.LittleEndian.PutUint16(raw, uint16(PacketTypeDelta))
		binary.LittleEndian.PutUint32(raw[4:], uint32(len(w.buf)))
		binary.LittleEndian.PutUint64(raw[8:], uint64(1000+c))
		raw = append(raw, w.buf...)
		inputs := source.MemoryChunks{nil, raw}
        var metadata []types.ChunkMeta
        var duplicate []byte
        if duplicates {
            // A later entry must never replace the first chunk's bytes. Include
            // empty, truncated, and equally sized conflicting packet payloads.
            duplicate = append([]byte(nil), raw...)
            for i := 16; i < len(duplicate); i++ { duplicate[i] ^= 0xff }
            switch c % 3 { case 0: duplicate = []byte{}; case 1: duplicate = duplicate[:17] }
            inputs = append(inputs, duplicate)
            metadata = []types.ChunkMeta{{Index:0}, {Index:1}, {Index:1}}
        }
        film, e := source.Load(inputs, metadata)
		if e != nil {
			t.Fatal(e)
		}
		arches := make([]Archetype, 36)
		arches[35] = arch
		fc := &FilmContext{film: film, chunks: []int{1}, chunksLus: true, slots: NewSlotBand(map[uint32]bool{512: true}), slotsLus: true, lay: cliffLayout, layLu: true, reg: &Registry{Archetypes: arches}, regLu: true, bal: ProfilDeBalayageParDefaut(), corrLu: true}
		anchors := []any{}
		walkDeltaBipedRecords(fc, fc.ChunkNumbers(), fc.BipedSlots(), cliffLayout, func(r deltaBipedRecord) {
			anchors = append(anchors, map[string]any{"i0": r.I0, "mask": r.Mask, "slot": r.Slot})
		})
		requested := [][]int{{1}, {1, 99}, {99}, {99, 1}}[c/128]
		fc.chunks = requested
		ranks, ast, ae := ScanAbilityRanks(fc)
		camo, cst, ce := ScanCamoStates(fc)
		em := []abilityEmission{}
		_, ee := walkAbilityEmissions(fc, func(e abilityEmission) { em = append(em, e) })
		if ae != nil || ee != nil {
			t.Fatalf("ability: %v %v", ae, ee)
		}

		charges, chargeStats, e := ScanAbilityCharges(fc)
		if e != nil {
			t.Fatal(e)
		}
		impulses, impulseStats, e := ScanAbilityImpulses(fc)
		if e != nil {
			t.Fatal(e)
		}
		grapple, grappleStats, e := ScanGrappleReads(fc)
		if e != nil {
			t.Fatal(e)
		}
		equipment, e := ScanUnitEquipment(fc)
		if e != nil {
			t.Fatal(e)
		}
		inventory, inventoryStats, e := ScanInventoryDeltas(fc)
		if e != nil {
			t.Fatal(e)
		}
		held, heldStats, e := ScanHeldWeaponChanges(fc, nil)
		if e != nil {
			t.Fatal(e)
		}
		p := fc.bal
		encoding := map[string]any{"bodies": map[string]any{"ability_anchor": p.Grammaire.CorpsAncrageCapacite, "mobility": p.Grammaire.CorpsActionMobilite, "mobility_extra_bits": p.Mouvement.MobilityActionExtraBits}, "index_bits": p.Mouvement.WorldObject.IndexW, "world_axis_bits": p.Mouvement.WorldObject.AxisW, "handle_bits": p.Mouvement.Traversal.IndexW, "traversal_axis_bits": p.Mouvement.Traversal.AxisW, "default_axis_bits": profile.LargeursAxeParDefautDuBuild(16), "region_axis_bits": map[string]any{"0": p.Mouvement.WorldObject.AxisW, "1": p.Mouvement.WorldObject.AxisW}, "delta_axis_bits": [3]uint{p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth, p.Mouvement.DeltaAxisWidth}, "full_precision": p.Mouvement.FullPrecision, "writer_absolute": false, "delta_handle_tail": false}
		rows = append(rows, map[string]any{"inventory": inventory, "inventoryStats": inventoryStats, "held": held, "heldStats": heldStats, "duplicate": func() any { if duplicates { return hex.EncodeToString(duplicate) }; return nil }(), "encoding": encoding, "charges": charges, "chargeStats": chargeStats, "impulses": impulses, "impulseStats": impulseStats, "grapple": grapple, "grappleStats": grappleStats, "equipment": equipment, "requested": requested, "hex": hex.EncodeToString(raw), "names": names, "anchors": anchors, "ranks": ranks, "emissions": em, "abilityStats": ast, "camo": camo, "camoStats": cst, "camoError": ce != nil})
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		t.Fatal(e)
	}
	path := "/private/tmp/halo-inventory-source-availability.json"
 if duplicates { path = "/private/tmp/halo-inventory-duplicate-sources.json" }
 if e = os.WriteFile(path, raw, 0600); e != nil {
		t.Fatal(e)
	}
}
