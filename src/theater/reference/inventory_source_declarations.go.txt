// Inventory declarations in exactly the pinned production source manifest.
// go run helper.go FILM_ROOT MANIFEST OUTPUT
package main
import("crypto/sha256";"encoding/hex";"encoding/json";"fmt";"go/ast";"go/parser";"go/token";"os";"path/filepath";"sort";"strconv")
type entry struct {Path string `json:"path"`;SHA string `json:"sha256"`}
type manifest struct {Commit string `json:"commit"`;Files []entry `json:"files"`}
type decl struct {Kind string `json:"kind"`;Name string `json:"name"`}
type file struct {Path string `json:"path"`;SHA string `json:"sha256"`;Package string `json:"package"`;Imports []string `json:"imports"`;Declarations []decl `json:"declarations"`;DocumentationOnly bool `json:"documentation_only"`}
func check(e error){if e!=nil{panic(e)}}
func main(){if len(os.Args)!=4{panic("expected FILM_ROOT MANIFEST OUTPUT")};raw,e:=os.ReadFile(os.Args[2]);check(e);var m manifest;check(json.Unmarshal(raw,&m));rows:=[]file{};seen:=map[string]bool{}
 for _,input:=range m.Files{if seen[input.Path]{panic("duplicate manifest path")};seen[input.Path]=true;b,e:=os.ReadFile(filepath.Join(os.Args[1],input.Path));check(e);sum:=sha256.Sum256(b);h:=hex.EncodeToString(sum[:]);if h!=input.SHA{panic("source hash mismatch: "+input.Path)};f,e:=parser.ParseFile(token.NewFileSet(),input.Path,b,0);check(e);r:=file{Path:input.Path,SHA:h,Package:f.Name.Name,Imports:[]string{},Declarations:[]decl{}}
  for _,im:=range f.Imports{v,e:=strconv.Unquote(im.Path.Value);check(e);r.Imports=append(r.Imports,v)}
  for _,d:=range f.Decls{switch v:=d.(type){case *ast.FuncDecl:kind:="function";if v.Recv!=nil{kind="method"};r.Declarations=append(r.Declarations,decl{kind,v.Name.Name});case *ast.GenDecl:if v.Tok==token.IMPORT{continue};for _,s:=range v.Specs{switch s:=s.(type){case *ast.TypeSpec:r.Declarations=append(r.Declarations,decl{"type",s.Name.Name});case *ast.ValueSpec:for _,n:=range s.Names{r.Declarations=append(r.Declarations,decl{v.Tok.String(),n.Name})};default:panic(fmt.Sprintf("unsupported spec %T",s))}};default:panic(fmt.Sprintf("unsupported declaration %T",d))}}
  sort.Strings(r.Imports);r.DocumentationOnly=len(r.Imports)==0&&len(r.Declarations)==0;rows=append(rows,r)
 };sort.Slice(rows,func(i,j int)bool{return rows[i].Path<rows[j].Path});out:=struct{Commit string `json:"reference_commit"`;Scope string `json:"scope"`;Files []file `json:"files"`}{m.Commit,"Pinned production manifest only; declaration inventory, not implementation or runtime proof",rows};b,e:=json.MarshalIndent(out,"","  ");check(e);check(os.WriteFile(os.Args[3],append(b,'\n'),0600))
}
