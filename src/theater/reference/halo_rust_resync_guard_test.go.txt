func TestHaloRustResyncGuard(t *testing.T){
 type result struct{ID uint32 `json:"id"`;End int `json:"end"`;Desync int `json:"desync"`}
 rows:=[]any{}
 for _,extra:=range []bool{false,true}{for _,count:=range []int{4095,4096,4097}{
  cfg:=DefaultFrameConfig();cfg.IDLowBits=11;cfg.HasExtraFields=extra
  reg:=&Registry{Archetypes:[]Archetype{{Index:0}}};world:=NewWorld(reg);world.BindFull(50,0);world.BindFull(51,0)
  var bw bitWriter
  for i:=0;i<count;i++{if extra{bw.bits(uint64(i),32)};bw.bit(1);bw.bits(uint64(50+i%2),11);bw.bits(0,2);bw.bit(0);bw.bits(0,4)}
  if extra{bw.bits(0,32)};bw.end()

  out:=[]result{};for _,r:=range records{out=append(out,result{r.ID,r.Trace.EndBit,r.DesyncAt})}
  rows=append(rows,map[string]any{"extra":extra,"count":count,"hex":hex.EncodeToString(bw.buf),"records":out,"calls":calls})
 }}
 raw,err:=json.Marshal(rows);if err!=nil{t.Fatal(err)};if err=os.WriteFile("/private/tmp/halo-resync-guard.json",raw,0600);err!=nil{t.Fatal(err)}
}
