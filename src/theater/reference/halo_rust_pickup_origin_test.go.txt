package replay
import("testing";"os";"encoding/json";"math/rand";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustPickupOrigin(t *testing.T){
 rng:=rand.New(rand.NewSource(7541)); rows:=[]any{}
 for run:=0;run<512;run++{
  points:=[]MapSpawnPoint{};pos:=[]grammar.BipedPosition{};placements:=[]EquipmentPlacement{};projection:=[]any{}
  states:=[]string{"","map_absent","not_established","established"};kinds:=[]string{"grenade","equipment","unknown",""};origins:=[]string{"dropped","deployed","unknown"};ends:=[]string{"seen","open","","other"}
  for i:=0;i<run%5;i++{points=append(points,MapSpawnPoint{X:float32(rng.Intn(5))/2,Y:float32(run%2)*float32(rng.Intn(3))/2,Z:float32(run%2)*float32(rng.Intn(3))/2,Kind:kinds[rng.Intn(4)]})}
  for i:=0;i<16;i++{pos=append(pos,grammar.BipedPosition{Slot:uint32(i%4),TimestampUS:uint64(rng.Intn(7))*100000,X:float32(rng.Intn(9))/2,Y:float32(run%2)*float32(rng.Intn(3))/2,Z:float32(run%2)*float32(rng.Intn(3))/2,HasWorld:rng.Intn(4)!=0})}
  for i:=0;i<8;i++{p:=EquipmentPlacement{T0:rng.Intn(7)-1,UntilMax:rng.Intn(9)-1,End:ends[rng.Intn(4)],Origin:origins[rng.Intn(3)],X:float32(rng.Intn(9))/2,Y:float32(run%2)*float32(rng.Intn(3))/2,Z:float32(run%2)*float32(rng.Intn(3))/2};placements=append(placements,p);projection=append(projection,map[string]any{"t0":p.T0,"until_max":p.UntilMax,"end":p.End,"origin":p.Origin,"x":p.X,"y":p.Y,"z":p.Z})}
  j:=newPickupOriginJudge(Options{SpawnPoints:points,SpawnPointsState:states[run%4]},pos,placements);queries:=[]any{}
  for slot:=uint32(0);slot<5;slot++{for _,ts:=range []uint64{0,99999,100000,100001,200000,500000,700000,700001,1<<63}{for frame:=-1;frame<10;frame++{origin:=j.origineDe(slot,ts,frame);kind:="";if origin=="spawner"{kind=j.kindAtteint};queries=append(queries,map[string]any{"slot":slot,"ts":ts,"frame":frame,"origin":origin,"kind":kind})}}}
  rows=append(rows,map[string]any{"state":states[run%4],"resolved_state":j.state,"points":points,"positions":pos,"placements":projection,"queries":queries})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-pickup-origin.json",raw,0600);e!=nil{t.Fatal(e)}
}
