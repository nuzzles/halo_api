package killsource
import("testing";"encoding/json";"encoding/hex";"os";"math/rand";"sort";"levelup/go-api/internal/games/halo_infinite/film/internal/grammar")
func TestHaloRustKillTimeline(t *testing.T){
 rng:=rand.New(rand.NewSource(183));rows:=[]any{}
 recs:=func(rs []grammar.KeyframeRec)[]any{out:=[]any{};for _,r:=range rs{out=append(out,map[string]any{"slot":r.Slot,"generation":r.Gen,"archetype":r.TI,"bit":r.Bit})};return out}
 for i:=0;i<256;i++{
  tl:=timeline{w:grammar.NewWorld(&grammar.Registry{})};for j:=0;j<i%10;j++ {rs:=[]grammar.KeyframeRec{};for k:=0;k<1+rng.Intn(7);k++{rs=append(rs,grammar.KeyframeRec{Slot:10+rng.Intn(20),Gen:1+rng.Intn(3),TI:[]int{35,35,8}[rng.Intn(3)],Bit:k*64})};tl.events=append(tl.events,keyframeEvent{ts:uint64(rng.Intn(10)*100),recs:rs})}
  input:=[]any{};for _,e:=range tl.events{input=append(input,map[string]any{"timestamp_us":e.ts,"records":recs(e.recs)})}
  sort.Slice(tl.events,func(i,j int)bool{return tl.events[i].ts<tl.events[j].ts});tl.preload();tl.buildGapWindows();tl.initSnap=tl.w.Snapshot();lo,hi:=tl.bipedRange();wins:=[]any{};for _,w:=range tl.gapWindows(){wins=append(wins,map[string]any{"slot":w.slot,"from":w.from,"to":w.to})}
  states:=[]any{};for _,ts:=range []uint64{0,99,100,299,500,900,1000,0,500}{if ts==0{tl.rewind()};w:=tl.advanceTo(ts);slots:=[]any{};for s:=0;s<40;s++{ti,ok:=w.ArchetypeForSlot(uint32(s));if !ok{continue};mask:=0;for g:=0;g<4;g++{if w.GenerationMatches(uint32(g<<30|s),true){mask|=1<<g}};slots=append(slots,map[string]any{"slot":s,"archetype":ti,"hard":w.HardBound(uint32(s)),"generation_mask":mask})};states=append(states,map[string]any{"time":ts,"slots":slots})}
  data:=make([]byte,128);rng.Read(data);put:=func(at,w int,v uint64){for j:=0;j<w;j++{a:=at+j;b:=byte(v>>uint(w-j-1)&1);data[a/8]=(data[a/8]&^(1<<uint(7-a%8)))|(b<<uint(7-a%8))}}
  for j:=0;j<5;j++{at:=i%8+j*173;put(at,32,uint64((1+j%3)<<30|10+j*2));put(at+32,32,35)}
  rows=append(rows,map[string]any{"input":input,"low":lo,"high":hi,"gaps":wins,"states":states,"data":hex.EncodeToString(data),"recovered":recs(keyframeRecs(data))})
 }
 b,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-kill-timeline.json",b,0600);e!=nil{t.Fatal(e)}
}
