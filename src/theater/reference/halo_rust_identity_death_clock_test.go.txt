func TestHaloRustIdentityDeathClock(t *testing.T){
 rng:=rand.New(rand.NewSource(7601));rows:=[]any{}

  ends:=[]int64{};deaths:=[]Death{};trueOffset:=int64(rng.Intn(1000000)-500000)

  rows=append(rows,map[string]any{"ends":ends,"deaths":deaths,"candidates":candidates,"refined":refined,"output":map[string]any{"offset_ms":off,"matched":n,"runner_up":second}})
 }
 raw,e:=json.Marshal(rows);if e!=nil{t.Fatal(e)};if e=os.WriteFile("/private/tmp/halo-identity-death-clock.json",raw,0600);e!=nil{t.Fatal(e)}
}
